eligible_war_for_international_alliance = {

	any_war_participant = {
		NOT = { THIS = ROOT }
		has_war_with = ROOT
		#does_not_share_same_international_alliance = yes
		save_temporary_scope_as = enemy_country
		eligible_war_goal_for_international_alliance = yes
		no_country_has_truce_with_international_alliance_member = yes
	}

}

eligible_war_goal_for_international_alliance = {

	OR = {
		
		#Country annexation
		custom_tooltip = {
			text = war_goal_target_state_in_alliance_annex_goal.tt
			OR = {
				war_participant_has_war_goal_of_type_against = { type = annex_country target = ROOT }
				war_participant_has_war_goal_of_type_against = { type = unification target = ROOT }
				war_participant_has_war_goal_of_type_against = { type = assisted_unification target = ROOT }
			}
		}
		
		#State annexations
		AND = {
			OR = { 
				war_participant_has_war_goal_of_type_against = { type = conquer_state target = ROOT }
				war_participant_has_war_goal_of_type_against = { type = return_state target = ROOT }
				war_participant_has_war_goal_of_type_against = { type = take_treaty_port target = ROOT }
			}
			ROOT = {
				custom_tooltip = {
					text = war_goal_target_state_in_alliance_geographical_limits.tt
					any_scope_state = {
						is_target_of_wargoal = scope:enemy_country
						OR = {
							AND = {
								owner ?= {
									OR = {
										has_modifier = nato_member
										has_modifier = nato_leader
									}
								}
								OR = {
									state_is_in_europe = yes
									state_is_in_north_america = yes
									region = sr:region_near_east
								}
							}
							AND = {
								owner ?= {
									OR = {
										has_modifier = warsaw_pact_member
										has_modifier = warsaw_pact_leader
									}
								}
								state_is_in_europe = yes
							}
							AND = {
								owner ?= {
									OR = {
										has_modifier = rio_pact_member
										has_modifier = rio_pact_leader
									}
								}
								OR = {
									state_is_in_north_america = yes
									state_is_in_south_america = yes
								}
							}
							AND = {
								owner ?= {
									OR = {
										has_modifier = csto_member
										has_modifier = csto_leader
									}
								}
								state_region ?= { is_homeland = cu:soviet }
							}
						}
					}
				}
			}
		}
		
		#Country autonomy
		custom_tooltip = {
			text = war_goal_target_state_in_alliance_subject_goal.tt
			OR = {
				war_participant_has_war_goal_of_type_against = { type = make_dominion target = ROOT }
				war_participant_has_war_goal_of_type_against = { type = make_semi_autonomous_colony target = ROOT }
				war_participant_has_war_goal_of_type_against = { type = make_protectorate target = ROOT }
				war_participant_has_war_goal_of_type_against = { type = make_satellite target = ROOT }
				war_participant_has_war_goal_of_type_against = { type = make_puppet target = ROOT }
				war_participant_has_war_goal_of_type_against = { type = make_mandate target = ROOT }
				war_participant_has_war_goal_of_type_against = { type = make_condominium target = ROOT }
			}
		}
		
		#Country politics
		war_participant_has_war_goal_of_type_against = { type = regime_change target = ROOT }
		war_participant_has_war_goal_of_type_against = { type = dominate_politics target = ROOT }
		
	}
	
}

no_country_has_truce_with_international_alliance_member = {

	custom_tooltip = {
		text = country_has_truce_with_international_alliance_member.tt
		any_country = {
			NOT = { THIS = ROOT }
			NOR = {
				has_truce_with = scope:enemy_country
				has_war_with = scope:enemy_country
			}
			shares_same_international_alliance = yes
		}
	}

}

is_in_an_international_alliance = {

	OR = {
	
		has_modifier = nato_member
		has_modifier = nato_leader
		
		has_modifier = warsaw_pact_member
		has_modifier = warsaw_pact_leader
		
		has_modifier = rio_pact_member
		has_modifier = rio_pact_leader
		
		has_modifier = csto_member
		has_modifier = csto_leader
		
	}

}

shares_same_international_alliance = {

	OR = {
		AND = {
			ROOT = {
				OR = { 
					has_modifier = nato_member
					has_modifier = nato_leader
				}
			}
			OR = { 
				has_modifier = nato_member
				has_modifier = nato_leader
			}
		}
		AND = {
			ROOT = {
				OR = { 
					has_modifier = warsaw_pact_member
					has_modifier = warsaw_pact_leader
				}
			}
			OR = { 
				has_modifier = warsaw_pact_member
				has_modifier = warsaw_pact_leader
			}
		}
		AND = {
			ROOT = {
				OR = { 
					has_modifier = rio_pact_member
					has_modifier = rio_pact_leader
				}
			}
			OR = { 
				has_modifier = rio_pact_member
				has_modifier = rio_pact_leader
			}
		}
		AND = {
			ROOT = {
				OR = { 
					has_modifier = csto_member
					has_modifier = csto_leader
				}
			}
			OR = { 
				has_modifier = csto_member
				has_modifier = csto_leader
			}
		}
	}

}

diplo_action_international_alliance_leader_check = {

	OR = {
		diplo_action_nato_leader_check = yes
		diplo_action_warsaw_pact_leader_check = yes
		diplo_action_rio_pact_leader_check = yes
		diplo_action_csto_leader_check = yes
	}

}

diplo_action_nato_leader_check = {
	
	#Leader
	scope:target_country = {
		has_modifier = nato_leader
	}
	NOT = { has_modifier = nato_member }
	
	#Conditions
	is_a_democracy = yes
	custom_tooltip = {
		text = eligible_geography_for_nato.tt
		OR = {
			country_is_in_europe = yes
			country_is_in_north_america = yes
			c:TUR ?= this
		}
	}
	
}

diplo_action_warsaw_pact_leader_check = {
	
	#Leader
	scope:target_country = { 
		has_modifier = warsaw_pact_leader
	}
	NOT = { has_modifier = warsaw_pact_member }
	
	#Conditions
	is_a_communist_government = yes
	custom_tooltip = {
		text = eligible_geography_for_warsaw_pact.tt
		country_is_in_europe = yes
	}
	
}

diplo_action_rio_pact_leader_check = {
	
	#Leader
	scope:target_country = { 
		has_modifier = rio_pact_leader
	}
	NOT = { has_modifier = rio_pact_member }
	
	#Conditions
	is_a_democracy = yes
	custom_tooltip = {
		text = eligible_geography_for_rio_pact.tt
		OR = {
			country_is_in_south_america = yes
			country_is_in_north_america = yes
		}
	}
	
}

diplo_action_csto_leader_check = {
	
	#Leader
	scope:target_country = {
		has_modifier = csto_leader
	}
	NOT = { has_modifier = csto_member }
	
	#Conditions
	OR = { 
		is_some_kind_of_democracy = yes
		is_a_republican_regime = yes
	}
	custom_tooltip = {
		text = eligible_geography_for_csto.tt
		capital ?= {
			state_region = { is_homeland = cu:soviet }	
		}
	}
	
}
