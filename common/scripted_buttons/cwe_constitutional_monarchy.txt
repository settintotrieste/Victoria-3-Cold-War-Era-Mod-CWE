je_seek_national_reconciliation_button = {
	name = "je_seek_national_reconciliation_button"
	desc = "je_seek_national_reconciliation_button_desc"

	visible = { 
		
		is_revolutionary = no
		is_secessionist = no
		
	}

	possible = {
	
		NOT = { has_modifier = national_reconciliation }
		
		trigger_if = {
			limit = { has_modifier = monarch_ig_landowners }
			ig:ig_landowners ?= {
				OR = { is_insurrectionary = no is_in_government = yes }
			}
		}
		trigger_if = {
			limit = { has_modifier = monarch_ig_devout }
			ig:ig_devout ?= {
				OR = { is_insurrectionary = no is_in_government = yes }
			}
		}
		trigger_if = {
			limit = { has_modifier = monarch_ig_rural_folk }
			ig:ig_rural_folk ?= {
				OR = { is_insurrectionary = no is_in_government = yes }
			}
		}
		trigger_if = {
			limit = { has_modifier = monarch_ig_industrialists }
			ig:ig_industrialists ?= {
				OR = { is_insurrectionary = no is_in_government = yes }
			}
		}
		trigger_if = {
			limit = { has_modifier = monarch_ig_petty_bourgeoisie }
			ig:ig_petty_bourgeoisie ?= {
				OR = { is_insurrectionary = no is_in_government = yes }
			}
		}
		trigger_if = {
			limit = { has_modifier = monarch_ig_trade_unions }
			ig:ig_trade_unions ?= {
				OR = { is_insurrectionary = no is_in_government = yes }
			}
		}
		
	}

	effect = {
	
		add_modifier = { 
			name = national_reconciliation 
			years = 1 
			is_decaying = yes 
		}
		
	}
	
	ai_chance = {
		base = 0
		
		#AI will only do this if country has high turmoil
		if = {
			limit = { 
				country_turmoil > 0.25
			}
			add = 10 
		}
		
	}
	
}

je_force_abdication_button = {
	name = "je_force_abdication_button"
	desc = "je_force_abdication_button_desc"

	visible = { 
		
		is_revolutionary = no
		is_secessionist = no
		
		NOT = { shares_same_monarch = yes }
		
		NOT = { active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy } }
		
		any_scope_character = {
			has_role = character_role_monarch
		}
		
	}

	possible = {
		
		custom_tooltip = {
			text = forced_abdication_timer.tt
			NOT = { has_variable = forced_abdication_timer }
		}
		
		trigger_if = {
			limit = { has_modifier = monarch_ig_landowners }
			ig:ig_landowners ?= {
				is_insurrectionary = yes
			}
		}
		trigger_if = {
			limit = { has_modifier = monarch_ig_devout }
			ig:ig_devout ?= {
				is_insurrectionary = yes
			}
		}
		trigger_if = {
			limit = { has_modifier = monarch_ig_rural_folk }
			ig:ig_rural_folk ?= {
				is_insurrectionary = yes
			}
		}
		trigger_if = {
			limit = { has_modifier = monarch_ig_industrialists }
			ig:ig_industrialists ?= {
				is_insurrectionary = yes
			}
		}
		trigger_if = {
			limit = { has_modifier = monarch_ig_petty_bourgeoisie }
			ig:ig_petty_bourgeoisie ?= {
				is_insurrectionary = yes
			}
		}
		trigger_if = {
			limit = { has_modifier = monarch_ig_trade_unions }
			ig:ig_trade_unions ?= {
				is_insurrectionary = yes
			}
		}
		
		government_has_legislative_super_majority = yes
		
	}

	effect = {
	
		#Domestic reaction
		every_scope_character = {
			limit = {
				OR = { 
					has_role = character_role_monarch
					has_role = character_role_viceroy
				}
			}
			interest_group ?= { save_scope_as = THIS_IG_scope }
			owner ?= {
				add_radicals = {
					value = very_large_radicals
					interest_group = scope:THIS_IG_scope
				}
			}
		}
	
		#Select new monarch
		remove_current_monarchs = yes
		trigger_event = { id = constitutional_monarchy_system.1 }
		
		set_variable = { 
			name = forced_abdication_timer
			years = 10 
		}
		
	}
	
	ai_chance = {
		base = 0
	}
	
}

je_enact_symbolic_monarchy_button = {
	name = "je_enact_symbolic_monarchy_button"
	desc = "je_enact_symbolic_monarchy_button_desc"

	visible = { 
		
		#Does not have Symbolic Monarchy
		NOT = { 
			active_law:lawgroup_governance_principles ?= {
				has_amendment = amendment_type:amendment_symbolic_monarchy
			}
		}
		
		#Cannot weaken overlord monarchy
		trigger_if = {
			limit = { is_subject = yes }
			NAND = {
				overlord ?= { is_a_monarchy = yes }
				shares_same_monarch = yes
			}
		}
		
	}

	possible = {
	
		#Legal requirements
		government_has_legislative_super_majority = yes
		
	}

	effect = {
	
		#Add Amendment
		active_law:lawgroup_governance_principles ?= {
			add_amendment = {
				type = amendment_symbolic_monarchy
				sponsor = PREV.ig:ig_petty_bourgeoisie 
			}
		}
		
	}
	
	ai_chance = {
		base = 0
		
		#AI will only do this if monarch is disliked
		if = {
			limit = { 
				NOT = { historical_railroading_active = yes }
				any_scope_character = {
					has_role = character_role_monarch
					popularity <= -50
				}
			}
			add = 10
		}
		
	}
	
}

je_enact_elective_monarchy_button = {
	name = "je_enact_elective_monarchy_button"
	desc = "je_enact_elective_monarchy_button_desc"

	visible = { 
		
		#Does not have Symbolic Monarchy
		NOT = { 
			active_law:lawgroup_governance_principles ?= {
				has_amendment = amendment_type:amendment_elective_monarchy
			}
		}
		
		#Cannot weaken overlord monarchy
		trigger_if = {
			limit = { is_subject = yes }
			NAND = {
				overlord ?= { is_a_monarchy = yes }
				shares_same_monarch = yes
			}
		}
		
	}

	possible = {
	
		#Federal states only
		has_law_or_variant = law_type:law_federal_state
	
		#Legal requirements
		government_has_legislative_super_majority = yes
		
	}

	effect = {
	
		#Add Amendment
		active_law:lawgroup_governance_principles ?= {
			add_amendment = {
				type = amendment_elective_monarchy
				sponsor = PREV.ig:ig_petty_bourgeoisie 
			}
		}
		
		#Generate elected monarchs
		if = {
			limit = { 
				active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy }
			}
			set_variable = {
				name = elective_monarchy_system_timer
				years = 5
			}
			random_scope_character = {
				limit = { 
					has_role = character_role_monarch
				}
				set_variable = { 
					name = elected_monarch_term
					years = 5
				}
			}
			generate_elective_monarchs_for_country = yes
		}
		
	}
	
	ai_chance = {
		base = 0
	}
	
}
