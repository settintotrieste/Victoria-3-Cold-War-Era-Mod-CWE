je_central_bank_give_independence_button = {
	name = "je_central_bank_give_independence_button"
	desc = "je_central_bank_give_independence_button_desc"

	visible = { 
		
		#Country does not already has an independent central bank
		NOT = { has_law_or_variant = law_type:law_independent_central_bank }
		
		#Not in a superstate currency union
		#NOT = { has_variable = superstate_currency_union }
		
	}

	possible = {
	
		#Cabinet agrees with this
		custom_tooltip = {
			text = has_no_low_cabinet_unity.tt
			NOT = { has_variable = low_cabinet_unity }
		}
		
		#Timer
		custom_tooltip = {
			text = central_bank_timer.tt
			NOT = { has_variable = central_bank_timer }
		}
		
	}

	effect = {
		
		#Make Central Bank independent
		set_variable = independent_central_bank
		
		#Only 1 time per year
		set_variable = { name = central_bank_timer years = 1 }
		
		#Domestic reaction
		every_interest_group = {
			limit = { 
				NOR = { 
					has_ideology = ideology:ideology_far_right_economy
					has_ideology = ideology:ideology_right_economy
					has_ideology = ideology:ideology_centre_economy
				}
			}
			hidden_effect = {
				remove_modifier = ig_little_angry_with_government
			}
			add_modifier = { name = ig_little_angry_with_government years = 1 is_decaying = yes }
		}
		
	}
	
	ai_chance = {
		base = 0
		
		if = {
			limit = { 
				
				#Government supports an independent monetary policy
				any_interest_group = { 
					leader ?= { is_ruler_of_own_country = yes }
					is_in_government = yes
					is_strongest_ig_in_government = yes
					OR = { 
						has_ideology = ideology:ideology_far_right_economy
						has_ideology = ideology:ideology_right_economy
						has_ideology = ideology:ideology_centre_economy
					}
				}
				
			}
			add = 10 
		}
		
	}
	
}

je_central_bank_increase_CBIR_button = {
	name = "je_central_bank_increase_CBIR_button"
	desc = "je_central_bank_increase_CBIR_button_desc"

	visible = {
		
	}

	possible = {
	
		#Country does not already have an independent central bank
		NOT = { has_law_or_variant = law_type:law_independent_central_bank }
	
		#Max
		NOT = { has_modifier = very_high_cb_interest_rate }
	
		#Cabinet agrees with this
		custom_tooltip = {
			text = has_no_low_cabinet_unity.tt
			NOT = { has_variable = low_cabinet_unity }
		}
		
		#Timer
		custom_tooltip = {
			text = central_bank_CBIR_timer.tt
			NOT = { has_variable = central_bank_CBIR_timer }
		}
		
	}

	effect = {
		
		#Increase CBIR
		if = {
			limit = { 
				has_modifier = very_low_cb_interest_rate
			}
			hidden_effect = { remove_modifier = very_high_cb_interest_rate }
			hidden_effect = { remove_modifier = high_cb_interest_rate }
			hidden_effect = { remove_modifier = medium_cb_interest_rate }
			add_modifier = { name = low_cb_interest_rate }
			hidden_effect = { remove_modifier = very_low_cb_interest_rate }
		}
		else_if = {
			limit = { 
				has_modifier = low_cb_interest_rate
			}
			hidden_effect = { remove_modifier = very_high_cb_interest_rate }
			hidden_effect = { remove_modifier = high_cb_interest_rate }
			add_modifier = { name = medium_cb_interest_rate }
			hidden_effect = { remove_modifier = low_cb_interest_rate }
			hidden_effect = { remove_modifier = very_low_cb_interest_rate }
		}
		else_if = {
			limit = { 
				has_modifier = medium_cb_interest_rate
			}
			hidden_effect = { remove_modifier = very_high_cb_interest_rate }
			add_modifier = { name = high_cb_interest_rate }
			hidden_effect = { remove_modifier = medium_cb_interest_rate }
			hidden_effect = { remove_modifier = low_cb_interest_rate }
			hidden_effect = { remove_modifier = very_low_cb_interest_rate }
		}
		else_if = {
			limit = { 
				has_modifier = high_cb_interest_rate
			}
			add_modifier = { name = very_high_cb_interest_rate }
			hidden_effect = { remove_modifier = high_cb_interest_rate }
			hidden_effect = { remove_modifier = medium_cb_interest_rate }
			hidden_effect = { remove_modifier = low_cb_interest_rate }
			hidden_effect = { remove_modifier = very_low_cb_interest_rate }
		}
		else = {
			remove_modifier = very_high_cb_interest_rate
			remove_modifier = high_cb_interest_rate
			add_modifier = { name = medium_cb_interest_rate }
			remove_modifier = low_cb_interest_rate
			remove_modifier = very_low_cb_interest_rate
		}
		
		#Only 1 time per month
		set_variable = { name = central_bank_CBIR_timer months = 1 }
		
	}
	
	ai_chance = {
		base = 0
		
		if = {
			limit = { 
				
				OR = {
					has_variable = very_high_core_inflation
					has_variable = high_core_inflation
				}
				
			}
			add = 10 
		}
		
	}
	
}

je_central_bank_increase_monetary_policy_button = {
	name = "je_central_bank_increase_monetary_policy_button"
	desc = "je_central_bank_increase_monetary_policy_button_desc"

	visible = {
		
	}

	possible = {
	
		#Country does not already have an independent central bank
		NOT = { has_law_or_variant = law_type:law_independent_central_bank }
	
		#Cabinet agrees with this
		custom_tooltip = {
			text = has_no_low_cabinet_unity.tt
			NOT = { has_variable = low_cabinet_unity }
		}
		
		#Not already max
		NOT = { has_modifier = law_very_expansionary_monetary_policy }
		
		#Not part of Bretton Woods System
		NOT = { has_journal_entry = je_bretton_woods_system_member }
		
	}

	effect = {
		
		if = {
			limit = { has_modifier = law_expansionary_monetary_policy }
			activate_law_law_very_expansionary_monetary_policy = yes
		}
		else_if = {
			limit = { has_modifier = law_neutral_monetary_policy }
			activate_law_law_expansionary_monetary_policy = yes
		}
		else_if = {
			limit = { has_modifier = law_deflationary_monetary_policy }
			activate_law_law_neutral_monetary_policy = yes
		}
		else_if = {
			limit = { has_modifier = law_very_deflationary_monetary_policy }
			activate_law_law_deflationary_monetary_policy = yes
		}
		else = {
			activate_law_law_neutral_monetary_policy = yes
		}
		
	}
	
	ai_chance = {
		base = 0
	}
	
}

je_central_bank_devalue_currency_system_button = {
	name = "je_central_bank_devalue_currency_system_button"
	desc = "je_central_bank_devalue_currency_system_button_desc"

	visible = {
		
	}

	possible = {
	
		#Cabinet agrees with this
		custom_tooltip = {
			text = has_no_low_cabinet_unity.tt
			NOT = { has_variable = low_cabinet_unity }
		}
		
		#Not already max
		NOT = { has_modifier = law_fiat_standard }
		
		#Timer
		custom_tooltip = {
			text = currency_system_timer.tt
			NOT = { has_variable = currency_system_timer }
		}
		
	}

	effect = {
		
		if = {
			limit = { has_modifier = law_semi_gold_standard }
			activate_law_law_fiat_standard = yes
		}
		else_if = {
			limit = { has_modifier = law_gold_standard }
			activate_law_law_semi_gold_standard = yes
		}
		else = {
			activate_law_law_fiat_standard = yes
		}
		
		set_variable = { name = currency_system_timer years = 5 }
		
	}
	
	ai_chance = {
		base = 0
		if = {
			limit = {
				NOT = { has_modifier = gold_standard_leader }
				has_technology_researched = tech_economy_3
				has_modifier = law_gold_standard
				NOT = { has_modifier = law_semi_gold_standard }
			}
			add = 10
		}
		if = {
			limit = {
				NOT = { has_modifier = gold_standard_leader }
				has_technology_researched = tech_economy_4
				has_modifier = law_semi_gold_standard
				NOT = { has_modifier = law_fiat_standard }
			}
			add = 10 
		}
	}
	
}
