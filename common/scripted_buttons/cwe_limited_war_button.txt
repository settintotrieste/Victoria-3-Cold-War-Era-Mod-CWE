je_limited_war_system_capitulate_button = {
	name = "je_limited_war_system_capitulate_button"
	desc = "je_limited_war_system_capitulate_button_desc"

	visible = { 
		
		any_country = {
			has_diplomatic_pact = { who = ROOT type = conduct_limited_war is_initiator = yes }
			NOT = { has_diplomatic_pact = { who = ROOT type = dominate_politics is_initiator = yes } }
		}
		
	}

	possible = {
		
	}

	effect = {
		
		#Enforce War Goal
		every_country = {
			limit = {
				has_diplomatic_pact = { who = ROOT type = conduct_limited_war is_initiator = yes }
				NOT = { has_diplomatic_pact = { who = ROOT type = dominate_politics is_initiator = yes } }
			}
			create_diplomatic_pact = { country = ROOT type = dominate_politics }
			remove_diplomatic_pact = { country = ROOT type = conduct_limited_war }
		}
		
	}
	
	ai_chance = {
		base = 0
		if = {
			limit = {
				OR = {
				
					#THIS is broke
					in_default = yes
					
					#THIS has no military
					AND = { 
						army_size = 0
						navy_size = 0
					}
					
				}
			}
			add = 10
		}
	}
	
}

je_limited_war_system_escalate_button = {
	name = "je_limited_war_system_escalate_button"
	desc = "je_limited_war_system_escalate_button_desc"

	visible = { 
		
		any_country = {
			has_diplomatic_pact = { who = ROOT type = conduct_limited_war is_initiator = yes }
			NOT = { has_war_with = ROOT }
		}
		
	}

	possible = {
		
	}

	effect = {
		
		#Enforce War Goal
		every_country = {
			limit = {
				has_diplomatic_pact = { who = ROOT type = conduct_limited_war is_initiator = yes }
				NOT = { has_war_with = ROOT }
			}
			save_scope_as = THIS_scope
			create_diplomatic_play = {
				type = dp_dominate_politics
				target_country = ROOT
				war = yes
				add_war_goal = {
					holder = scope:THIS_scope
					type = regime_change
					target_country = ROOT
				}
			}
		}
		
	}
	
	ai_chance = {
		base = 0
		if = {
			limit = { 
				any_country = {
					has_diplomatic_pact = { who = ROOT type = conduct_limited_war is_initiator = yes }
					country_rank < root.country_rank
				}
			}
			add = 10
		}
	}
	
}
