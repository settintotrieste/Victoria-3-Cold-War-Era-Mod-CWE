je_create_collective_presidency_button = {
	name = "je_create_collective_presidency_button"
	desc = "je_create_collective_presidency_button_desc"

	visible = { 
		
		#Does not have Collective HOS
		NOT = { 
			active_law:lawgroup_governance_principles ?= {
				has_amendment = amendment_type:amendment_rotating_presidency
			}
		}
		
	}

	possible = {
	
		#Presidential system
		OR = { 
			has_law_or_variant = law_type:law_HOS_president
			has_law_or_variant = law_type:law_HOS_nationalist_leader
			has_law_or_variant = law_type:law_HOS_religious_leader
			has_law_or_variant = law_type:law_HOS_communist_leader
		}
		
		#Not for Prime Ministers - Unless dictatorship
		OR = {
			NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			AND = { 
				has_law_or_variant = law_type:law_HOG_prime_minister
				NOT = { has_law_or_variant = law_type:law_democratic_legislature }
			}
		}
		
		#Has Term Limit
		trigger_if = {
			limit = {
				NOR = {
					has_journal_entry = je_successor_struggle_system
					has_journal_entry = je_power_struggle_system
				}
			}
			has_law_or_variant = law_type:law_yes_term_limit
		}
		
		#Legal requirements
		government_has_legislative_super_majority = yes
		
	}

	effect = {
	
		#Add Amendment
		active_law:lawgroup_governance_principles ?= {
			add_amendment = {
				type = amendment_rotating_presidency
				sponsor = PREV.ig:ig_petty_bourgeoisie 
			}
		}
		
	}
	
	ai_chance = {
		base = 0
	}
	
}

je_extend_term_limit_button = {
	name = "je_extend_term_limit_button"
	desc = "je_extend_term_limit_button_desc"

	visible = {
		
	}

	possible = {
	
		government_has_legislative_super_majority = yes
		
		in_election_campaign = no
		
		custom_tooltip = {
			text = extend_term_limit_timer.tt
			NOT = { has_variable = extend_term_limit_timer }
		}
		
	}

	effect = {
	
		#Add term limit
		custom_tooltip = {
			text = extend_term_limit.tt
			change_variable = {
				name = term_limit_max_var
				add = 4
			}
		}
		
		#Timer
		set_variable = { name = extend_term_limit_timer years = 1 }
		
		#Domestic effects
		every_interest_group = {
			limit = {
				is_in_government = no
				NOT = { is_same_interest_group_type = root.ruler.interest_group }
			}
			save_scope_as = ig_scope
			owner ?= {
				add_radicals = {
					value = medium_radicals
					interest_group = scope:ig_scope
				}
			}
		}
		
	}
	
	ai_chance = {
		base = 0
		
		#AI will only do this if the ruler is autocratic
		if = {
			limit = {
				ruler ?= {
					character_is_autocratic = yes
				}
			}
			add = 100
		}
		
	}
	
}
