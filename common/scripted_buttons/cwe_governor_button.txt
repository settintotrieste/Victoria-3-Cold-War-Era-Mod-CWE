je_governor_make_elected_button = {
	name = "je_governor_make_elected_button"
	desc = "je_governor_make_elected_button_desc"

	visible = { 
		NOT = { has_variable = elected_governor_system }
		has_law_or_variant = law_type:law_democratic_legislature
	}

	possible = {
	
		NOT = { has_law_or_variant = law_type:law_HOS_monarch }
		
		government_has_legislative_super_majority = yes
		in_election_campaign = no
		
		#Timer
		custom_tooltip = {
			text = governor_timer.tt
			NOT = { has_variable = fire_governor_timer }
		}
		
	}

	effect = {
	
		set_variable = elected_governor_system
	
		#PM IG is angry
		every_scope_character = {
			limit = {
				has_role = character_role_ig_leader
				character_is_democratic = yes
			}
			interest_group ?= {
				save_scope_as = THIS_IG_scope
			}
			owner ?= {
				add_loyalists = {
					value = medium_radicals
					interest_group = scope:THIS_IG_scope
				}
			}
		}
		every_scope_character = {
			limit = { 
				has_role = character_role_ig_leader
				interest_group ?= { is_in_government = yes }
				character_is_autocratic = yes
			}
			interest_group ?= {
				save_scope_as = THIS_IG_scope
			}
			owner ?= {
				add_radicals = {
					value = medium_radicals
					interest_group = scope:THIS_IG_scope
				}
			}
		}
		
		#Only 1 time per year
		set_variable = { name = fire_governor_timer years = 1 }
		
	}
	
	ai_chance = {
		base = 0
	}
	
}
