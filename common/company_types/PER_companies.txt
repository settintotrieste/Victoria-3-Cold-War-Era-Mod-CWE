		}

			has_company = company_type:company_national_iranian_oil
		}
		any_scope_state = {
			state_region = s:STATE_KHUZESTAN

		}
	}

	ai_construction_targets = {
		building_oil_rig = {
			level = 5
			state_trigger = {
