calculate_cabinet_unity = {

	#Initialise variables
	set_variable = {
		name = cabinet_unity_var
		value = 0
	}
	set_variable = {
		name = cabinet_unity_divisor_var
		value = 0
	}
	
	#Calculate Cabinet Loyalty
	every_scope_character = {
		limit = {
			character_is_key_state_officeholder = yes
			is_ruler_of_own_country = no
		}
		save_scope_as = cabinet_minister_scope
		owner = {
			change_variable = {
				name = cabinet_unity_var
				add = scope:cabinet_minister_scope.loyalty
			}
			change_variable = {
				name = cabinet_unity_divisor_var
				add = 1
			}
		}
	}
	
	if = {
		limit = {
			has_variable = cabinet_unity_var
			has_variable = cabinet_unity_divisor_var
			var:cabinet_unity_divisor_var > 0
		}
		change_variable = {
			name = cabinet_unity_var
			divide = var:cabinet_unity_divisor_var
		}
	}
	else = {
		set_variable = {
			name = cabinet_unity_var
			value = 100
		}
	}
	
	remove_variable = strong_cabinet_unity
	remove_variable = medium_cabinet_unity
	remove_variable = low_cabinet_unity
	if = {
		limit = { var:cabinet_unity_var > 66 }
		set_variable = strong_cabinet_unity
	}
	else_if = {
		limit = { var:cabinet_unity_var > 33 }
		set_variable = medium_cabinet_unity
	}
	else = {
		set_variable = low_cabinet_unity
	}
	
	calculate_cabinet_incompetence = yes

}

calculate_cabinet_incompetence = {

	set_variable = {
		name = cabinet_incompetence_var
		value = 0
	}
	
	every_scope_character = {
		limit = {
			
			#Ministers only
			character_is_key_state_officeholder = yes
			
			#Incompetent traits
			OR = {
				has_trait = alcoholic
				has_trait = opium_addiction
				has_trait = cocaine_addiction
				has_trait = grifter
				has_trait = senile
				has_trait = reckless
				has_trait = bandit
				has_trait = political_appointee
				has_trait = inept
				has_trait = inexperienced
				has_trait = social_bandit
			}
			
		}
		owner ?= {
			change_variable = {
				name = cabinet_incompetence_var
				add = 0.01
			}
		}
	}
	
	if = {
		limit = { has_modifier = cabinet_incompetence_modifier }
		remove_modifier = cabinet_incompetence_modifier
	}
	
	if = {
		limit = { 
			has_variable = cabinet_incompetence_var
		}
		
		clamp_variable = {
			name = cabinet_incompetence_var
			max = 1
			min = 0
		}
	
		#Apply effects
		if = {
			limit = { root.var:cabinet_incompetence_var >= 0.01 }
			add_modifier = {
				name = cabinet_incompetence_modifier
				multiplier = var:cabinet_incompetence_var
			}
		}
		else = {
			if = {
				limit = { has_modifier = cabinet_incompetence_modifier }
				remove_modifier = cabinet_incompetence_modifier
			}
		}
		
		#For display purposes only
		change_variable = {
			name = cabinet_incompetence_var
			multiply = 100
		}
		
	}

}

remove_non_government_characters_from_cabinet = {

	#Clear Ruler modifier
	every_scope_character = {
		limit = {
			has_modifier = THIS_is_ruler
			is_ruler_of_own_country = no 
		}
		remove_modifier = THIS_is_ruler
	}
	
	#Clear cabinet modifiers
	every_scope_character = {
		limit = {
			OR = { 
				has_role = character_role_cabinet_minister
				interest_group ?= { has_modifier = cabinet_minister_ig }
			}
			interest_group ?= { is_in_government = no }
		}
		remove_character_role = character_role_cabinet_minister
		interest_group = { remove_modifier = cabinet_minister_ig }
	}

}
