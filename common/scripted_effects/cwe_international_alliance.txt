diplo_action_international_alliance_join_effect = {

	#Add membership
	if = {
		limit = { diplo_action_nato_leader_check = yes }
		if = {
			limit = {
				NOT = { has_modifier = nato_member }
			}
			add_modifier = { name = nato_member }
		}
		add_journal_entry = { type = je_nato_system }
	}
	if = {
		limit = { diplo_action_warsaw_pact_leader_check = yes }
		if = {
			limit = {
				NOT = { has_modifier = warsaw_pact_member }
			}
			add_modifier = { name = warsaw_pact_member }
		}
		add_journal_entry = { type = je_warsaw_pact_system }
	}
	
	#Remove timer
	remove_variable = international_alliance_application_cooldown

}
