calculate_organised_crime_effects = {

	set_variable = {
		name = organised_crime_effect_var
		value = 0
	}
	
	every_scope_character = {
		limit = {
			character_committed_crimes = yes
		}
		owner ?= {
			change_variable = {
				name = organised_crime_effect_var
				add = 0.01
			}
		}
	}
	
	if = {
		limit = { has_modifier = organised_crime_effect_modifier }
		remove_modifier = organised_crime_effect_modifier
	}
	
	if = {
		limit = { 
			has_variable = organised_crime_effect_var
		}
		
		clamp_variable = {
			name = organised_crime_effect_var
			max = 1
			min = 0
		}
	
		#Apply effects
		if = {
			limit = { root.var:organised_crime_effect_var >= 0.01 }
			add_modifier = {
				name = organised_crime_effect_modifier
				multiplier = var:organised_crime_effect_var
			}
		}
		else = {
			if = {
				limit = { has_modifier = organised_crime_effect_modifier }
				remove_modifier = organised_crime_effect_modifier
			}
		}
		
		#For display purposes only
		change_variable = {
			name = organised_crime_effect_var
			multiply = 100
		}
		
	}

}
