generate_subsidiary_monarchs_for_country = {

	set_variable = {
		name = subsidiary_monarchs_quota
		value = 0
	}
	
	hidden_effect = {
		if = {
			limit = { has_variable = consolidated_princely_states }
			ordered_scope_state = {
				limit = { 
					is_incorporated = yes 
					is_largest_state_in_region = yes 
				}
				order_by = state_population
				position = 0
				check_range_bounds = no
				generate_subsidiary_monarchs_for_state = yes
			}
			ordered_scope_state = {
				limit = { 
					is_incorporated = yes 
					is_largest_state_in_region = yes 
				}
				order_by = state_population
				position = 1
				check_range_bounds = no
				generate_subsidiary_monarchs_for_state = yes
			}
			ordered_scope_state = {
				limit = { 
					is_incorporated = yes 
					is_largest_state_in_region = yes 
				}
				order_by = state_population
				position = 2
				check_range_bounds = no
				generate_subsidiary_monarchs_for_state = yes
			}
			ordered_scope_state = {
				limit = { 
					is_incorporated = yes 
					is_largest_state_in_region = yes 
				}
				order_by = state_population
				position = 3
				check_range_bounds = no
				generate_subsidiary_monarchs_for_state = yes
			}
			ordered_scope_state = {
				limit = { 
					is_incorporated = yes 
					is_largest_state_in_region = yes 
				}
				order_by = state_population
				position = 4
				check_range_bounds = no
				generate_subsidiary_monarchs_for_state = yes
			}
		}
		else = {
			every_scope_state = {
				limit = { 
					is_incorporated = yes 
					is_largest_state_in_region = yes 
				}
				generate_subsidiary_monarchs_for_state = yes
			}
		}
	}
	
	set_variable = {
		name = subsidiary_monarchs_count
		value = 0
	}
	every_scope_character = {
		limit = { has_role = character_role_subsidiary_monarch }
		owner ?= {
			change_variable = {
				name = subsidiary_monarchs_count
				add = 1
			}
		}
	}

}

generate_subsidiary_monarchs_for_state = {

	save_scope_as = state_in_question
	set_variable = { name = state_in_question days = 1 }
	ordered_scope_pop = {
		limit = {
			culture ?= {
				save_temporary_scope_as = colonial_culture_in_question_temp 
			}
			scope:state_in_question ?= { 
				state_region = { 
					is_homeland = scope:colonial_culture_in_question_temp
				}
			}
		}
		order_by = total_size
		position = 0
		check_range_bounds = no
		culture ?= { save_scope_as = colonial_culture_in_question }
	}
	owner ?= {
		change_variable = {
			name = subsidiary_monarchs_quota
			add = 1
		}
		if = {
			limit = {
				exists = scope:state_in_question
				NOT = {
					any_scope_character = {
						has_role = character_role_subsidiary_monarch
						home_state ?= { has_variable = state_in_question }
					}
				}
			}
			create_character = {
				role = magnate noble = yes
				culture = scope:colonial_culture_in_question
				on_created = {
					add_character_role = character_role_subsidiary_monarch
				}
				save_scope_as = new_subsidiary_monarch
			}
			scope:new_subsidiary_monarch ?= {
				set_home_state = scope:state_in_question
			}
		}
	}
	remove_variable = state_in_question

}

set_subsidiary_monarchs_amendment = {

	if = { 
		limit = {
			NOT = {
				active_law:lawgroup_governance_principles ?= {
					has_amendment = amendment_type:amendment_subsidiary_monarchy
				}
			}
		}
		active_law:lawgroup_governance_principles ?= {
			add_amendment = {
				type = amendment_subsidiary_monarchy
				sponsor = PREV.ig:ig_landowners
			}
		}
	}
	set_variable = { name = block_subsidiary_monarch_generation months = 2 }

}

upgrade_subsidiary_monarchs_to_elective_monarchs = {

	if = {
		limit = {
			active_law:lawgroup_governance_principles ?= {
				has_amendment = amendment_type:amendment_elective_monarchy
			}
		}
		every_scope_character = {
			limit = {
				has_role = character_role_subsidiary_monarch
				NOT = { has_role = character_role_monarch_elective }
			}
			remove_character_role = character_role_subsidiary_monarch
			add_character_role = character_role_monarch_elective
		}
	}

}

calculate_subsidiary_monarch_ratio = {

	#Calculate Subsidiary Monarchy ratio
	if = {
		limit = {
			has_variable = subsidiary_monarchs_count
		}
		set_variable = {
			name = subsidiary_monarch_state
			value = 0
		}
		every_scope_state = {
			limit = {
				is_incorporated = yes 
				is_largest_state_in_region = yes
			}
			owner ?= {
				change_variable = {
					name = subsidiary_monarch_state
					add = 1
				}
			}
		}
		set_variable = {
			name = subsidiary_monarch_ratio
			value = var:subsidiary_monarchs_count
		}
		change_variable = {
			name = subsidiary_monarch_ratio
			divide = var:subsidiary_monarch_state
		}
	}

}

initialise_subsidiary_monarch_effect = {

	if = {
		limit = {
			is_a_subsidiary_monarchy = yes
			scope:target ?= {
				is_a_monarchy = yes
			}
		}
		if = {
			limit = {
				scope:target ?= {
					NOT = {
						any_scope_character = {
							has_role = character_role_monarch
						}
					}
				}
			}
			random_scope_character = {
				limit = {
					has_role = character_role_subsidiary_monarch
					home_state ?= {
						owner = scope:target
					}
				}
				transfer_character = scope:target
				set_character_as_monarch = yes
				remove_character_role = character_role_subsidiary_monarch
			}
		}
		every_scope_character = {
			limit = {
				has_role = character_role_subsidiary_monarch
				home_state ?= {
					owner = scope:target
				}
			}
			remove_character_role = character_role_subsidiary_monarch
		}
		set_variable = {
			name = subsidiary_monarchs_count
			value = 0
		}
		every_scope_character = {
			limit = { has_role = character_role_subsidiary_monarch }
			owner ?= {
				change_variable = {
					name = subsidiary_monarchs_count
					add = 1
				}
			}
		}
	}

}
