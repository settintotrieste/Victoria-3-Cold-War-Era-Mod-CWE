change_to_parliamentary_elective = {
	clean_up_any_monarchs = yes
	
	if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_HOG_none
				has_law_or_variant = law_type:law_HOG_president_prime_minister
			}
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				any_scope_character = {
					has_role = character_role_regent
					is_ruler_of_own_country = no 
				}
			}
			every_scope_character = {
				limit = { 
					has_role = character_role_regent
					is_ruler_of_own_country = no 
				}
				remove_character_role = character_role_regent
				clean_up_regent_character = yes
			}
		}
		else_if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				any_scope_character = {
					has_role = character_role_monarch
					is_ruler_of_own_country = no 
				}
			}
			set_variable = { name = hide_ruler_change_notification days = 3 }
			random_scope_character = {
				limit = { has_role = character_role_monarch }
				set_character_as_ruler = yes
				set_character_as_monarch = yes
			}
		}
		every_scope_character = {
			limit = { 
				has_role = character_role_viceroy
			}
			remove_character_role = character_role_viceroy
			owner ?= { ruler ?= { set_character_as_viceroy = yes } }
		}
	}
	else = {
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				ruler ?= {
					has_role = character_role_monarch
				}
			}
			if = {
				limit = { any_scope_character = { has_role = character_role_regent } }
				random_scope_character = {
					limit = { has_role = character_role_regent }
					set_character_as_ruler = yes
				}
			}
			else = {
				ordered_scope_character = {
					limit = { 
						has_role_of_type = politician 
						NOT = { has_role = character_role_monarch }
					}
					order_by = popularity
					position = 0
					set_character_as_ruler = yes
				}
			}
			if = {
				limit = { exists = heir }
				heir ?= {
					disinherit_character = yes
				}
			}
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				ruler ?= {
					has_role = character_role_regent
				}
			}
			if = {
				limit = { exists = heir }
				heir ?= {
					disinherit_character = yes
				}
			}
		}
		if = {
			limit = {
				ruler ?= {
					has_role = character_role_viceroy
				}
			}
			ordered_scope_character = {
				limit = { 
					has_role_of_type = politician 
					NOT = { has_role = character_role_monarch }
					NOT = { has_role = character_role_viceroy }
				}
				order_by = popularity
				position = 0
				set_character_as_ruler = yes
			}
		}
	}
	
}

post_change_to_parliamentary_elective = {
	if = {
		limit = { is_foreign_governor_only_country = yes }
		trigger_event = { id = revolution_system.20 }
	}
}

change_to_dictatorial = {
	
	clean_up_any_monarchs = yes
	
	if = {
		limit = {
			ruler ?= { NOT = { has_role = character_role_monarch } }
		}
		ruler ?= { save_scope_as = previous_ruler_scope }
	}
	
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			has_law_or_variant = law_type:law_HOG_prime_minister
		}
		if = {
			limit = { exists = heir }
			heir ?= {
				disinherit_character = yes
			}
		}
	}
	else_if = {
		limit = {
			NOR = {
				ruler ?= { has_variable = dynastic_ruler_enabled }
				AND = {
					has_law_or_variant = law_type:law_HOS_monarch
					NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
					ruler_is_governor_general = no
					shares_same_monarch = no
				}
			}
		}
		if = {
			limit = { exists = heir }
			heir ?= {
				disinherit_character = yes
			}
		}
	}
	
}

post_change_to_dictatorial = {
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOG_prime_minister
			exists = scope:previous_ruler_scope
		}
		scope:previous_ruler_scope ?= { set_character_as_ruler = yes }
	}
	if = {
		limit = { is_foreign_governor_only_country = yes }
		trigger_event = { id = revolution_system.20 }
	}
}

change_to_hereditary = {
	
	#Set Monarch
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			ruler_is_governor_general = no
			is_a_colonial_government = no
			shares_same_monarch = no
		}
		set_variable = { name = hide_ruler_change_notification days = 3 }
		if = {
			limit = {
				any_scope_character = {
					has_role = character_role_monarch
				}
			}
			random_scope_character = {
				limit = {
					has_role = character_role_monarch
				}
				set_character_as_ruler = yes
			}
		}
		else = {
			ruler ?= { set_character_as_monarch = yes }
		}
		if = {
			limit = {
				ruler ?= { has_role = character_role_monarch }
				OR = {
					NOT = { exists = heir }
					heir ?= { NOT = { has_role = character_role_prince } }
				}
			}
			random_scope_character = {
				limit = {
					has_role = character_role_prince
					is_ruler_of_own_country = no
				}
				save_scope_as = THIS_scope
				owner ?= { set_heir = scope:THIS_scope }
			}
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
		}
		set_variable = { name = recently_became_monarchy days = 1 }
		if = {
			limit = {
				OR = {
					any_scope_character = {
						has_role = character_role_monarch
						is_female = yes
					}
					AND = {
						shares_same_monarch = yes
						overlord ?= {
							any_scope_character = {
								has_role = character_role_monarch
								is_female = yes
							}
						}
					}
				}
			}
			remove_variable = primogeniture_system
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			ruler_is_governor_general = no
			shares_same_monarch = no
			any_scope_character = {
				NOT = { has_role = character_role_regent }
				has_role = character_role_monarch
				is_eligible_to_become_monarch = yes
			}
		}
		trigger_event = { id = regency.3 popup = yes }
	}
	
}

post_change_to_hereditary = {
	
}

change_to_regency = {

	#Failsafe
	if = {
		limit = {
			exists = heir
			ruler ?= { NOT = { has_role = character_role_monarch } }
		}
		ruler ?= {
			if = {
				limit = { NOT = { has_role = character_role_regent } }
				set_character_as_regent = yes
			}
		}
	}
	else = {
		ruler ?= {
			if = {
				limit = { has_role = character_role_monarch }
				remove_character_role = character_role_monarch
			}
			if = {
				limit = { NOT = { has_role = character_role_regent } }
				set_character_as_regent = yes
			}
		}
		heir ?= {
			if = {
				limit = { NOT = { has_role = character_role_monarch } }
				set_character_as_monarch = yes
			}
		}
	}
	random_scope_character = {
		limit = { has_role = character_role_regent }
		set_character_as_ruler = yes
	}
	
	debug_log = "Regency Created in [THIS.GetCountry.GetNameNoFormatting]"

}
