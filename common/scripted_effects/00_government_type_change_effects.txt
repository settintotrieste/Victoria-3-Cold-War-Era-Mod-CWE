change_to_parliamentary_elective = {
	
	clean_up_any_monarchs = yes
	
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			has_law_or_variant = law_type:law_HOG_prime_minister
			ruler_is_governor_general = no
			shares_same_monarch = no
			ruler ?= { has_role = character_role_monarch }
		}
		set_variable = { name = hide_ruler_change_notification days = 3 }
		random_scope_character = {
			limit = { has_role = character_role_head_of_government }
			set_character_as_ruler = yes
		}
	}
	if = {
		limit = { exists = heir }
		heir ?= {
			disinherit_character = yes
		}
	}
	
	#Reset term limits for constitutional monarchies
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			has_law_or_variant = law_type:law_HOG_prime_minister
		}
		activate_law = law_type:law_no_term_limit
	}
	
	#Deal with change of parliamentary elective
	remove_modifier = starting_voting_power_modifier
	
}

post_change_to_parliamentary_elective = {
	if = {
		limit = { is_foreign_governor_only_country = yes }
		trigger_event = { id = revolution_system.20 }
	}
}

change_to_dictatorial = {
	
	clean_up_any_monarchs = yes
	
	ruler ?= { set_variable = { name = previous_ruler_scope days = 3 } }
	if = {
		limit = {
			ruler ?= { has_role = character_role_monarch }
		}
		set_variable = { name = hide_ruler_change_notification days = 3 }
		if = {
			limit = { any_scope_character = { has_role = character_role_head_of_government } }
			random_scope_character = {
				limit = { has_role = character_role_head_of_government }
				set_character_as_ruler = yes
			}
		}
		else_if = {
			limit = {
				OR = { 
					has_law_or_variant = law_type:law_HOG_prime_minister
					has_law_or_variant = law_type:law_HOG_none
				}
			}
			ordered_scope_character = {
				limit = {
					OR = { 
						has_role_of_type = politician
						has_role = character_role_regent
					}
					is_ruler_of_own_country = no
				}
				order_by = popularity
				position = 0
				check_range_bounds = no
				set_character_as_ruler = yes
			}
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			shares_same_monarch = yes
		}
		set_variable = { name = hide_ruler_change_notification days = 3 }
		ruler ?= {
			set_character_as_viceroy = yes
			set_character_as_ruler = yes
		}
	}
	if = {
		limit = { exists = heir }
		heir ?= {
			disinherit_character = yes
		}
	}
	
	#Reset term limits for constitutional monarchies
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			has_law_or_variant = law_type:law_HOG_prime_minister
		}
		activate_law = law_type:law_no_term_limit
	}
	
	#Deal with change of parliamentary elective
	remove_modifier = starting_voting_power_modifier
	
}

post_change_to_dictatorial = {
	if = {
		limit = {

		}
		random_scope_character = {

		}

		trigger_event = { id = revolution_system.20 }
	}
}

change_to_hereditary = {
	
	#Set Monarch
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			ruler_is_governor_general = no
			shares_same_monarch = no
			is_a_regency_government = no
		}
		set_variable = { name = hide_ruler_change_notification days = 3 }
		if = {
			limit = {
				any_scope_character = {
					has_role = character_role_monarch
				}
			}
			random_scope_character = {
				limit = {
					has_role = character_role_monarch
				}
				set_character_as_ruler = yes
			}
		}
		else = {
			ruler ?= { set_character_as_monarch = yes }
		}
		if = {
			limit = {
				ruler ?= { has_role = character_role_monarch }
				OR = {
					NOT = { exists = heir }
					heir ?= { NOT = { has_role = character_role_prince } }
				}
			}
			random_scope_character = {
				limit = {
					has_role = character_role_prince
					is_ruler_of_own_country = no
				}
				save_scope_as = THIS_scope
				owner ?= { set_heir = scope:THIS_scope }
			}
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
		}
		set_variable = { name = recently_became_monarchy days = 1 }
		if = {
			limit = {
				OR = {
					any_scope_character = {
						has_role = character_role_monarch
						is_female = yes
					}
					AND = {
						shares_same_monarch = yes
						overlord ?= {
							any_scope_character = {
								has_role = character_role_monarch
								is_female = yes
							}
						}
					}
				}
			}
			remove_variable = primogeniture_system
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			ruler_is_governor_general = no
			shares_same_monarch = no
			any_scope_character = {
				NOT = { has_role = character_role_regent }
				has_role = character_role_monarch
				is_eligible_to_become_monarch = yes
			}
		}
		trigger_event = { id = regency.3 popup = yes }
	}
	
	#Reset term limits for constitutional monarchies
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			has_law_or_variant = law_type:law_HOG_prime_minister
		}
		activate_law = law_type:law_no_term_limit
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy }
		}
		activate_law = law_type:law_yes_term_limit
	}
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			NOT = { active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy } }
		}
		activate_law = law_type:law_no_term_limit
	}
	
	#Deal with change of parliamentary elective
	remove_modifier = starting_voting_power_modifier
	
}

post_change_to_hereditary = {
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			ruler_is_governor_general = no
			shares_same_monarch = no
			ruler ?= {
				NOR = {
					has_role = character_role_monarch
					has_role = character_role_regent
				}
			}
		}
		random_scope_character = {
			limit = { has_role = character_role_regent }
			set_character_as_ruler = yes
		}
		random_scope_character = {
			limit = { has_role = character_role_monarch }
			set_character_as_ruler = yes
		}
	}
}

change_to_regency = {

	#Failsafe
	random_scope_character = {
		limit = { has_role = character_role_regent }
		set_character_as_ruler = yes
	}
	if = {
		limit = {
			ruler ?= {
				NOR = { 
					has_role = character_role_monarch
					has_role = character_role_regent 
				}
			}
		}
		ruler ?= {
			set_character_as_regent = yes
		}
		heir ?= {
			if = {
				limit = { NOT = { has_role = character_role_monarch } }
				set_character_as_monarch = yes
			}
		}
	}
	
	#Deal with change of parliamentary elective
	remove_modifier = starting_voting_power_modifier

}

	}
	
}

change_to_elective_monarchy = {

	#Set Monarch
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			ruler_is_governor_general = no
			is_a_colonial_government = no
			shares_same_monarch = no
		}
		set_variable = { name = hide_ruler_change_notification days = 3 }
		if = {
			limit = {
				any_scope_character = {
					has_role = character_role_monarch
				}
			}
			random_scope_character = {
				limit = {
					has_role = character_role_monarch
				}
				set_character_as_ruler = yes
			}
		}
		else = {
			random_scope_character = {
				limit = {
					OR = { 
						has_role = character_role_monarch_elective
						has_role = character_role_prince
					}
				}
				set_character_as_ruler = yes
				set_character_as_monarch = yes
			}
		}
		if = {
			limit = {
				ruler ?= { has_role = character_role_monarch }
				OR = {
					NOT = { exists = heir }
					heir ?= { NOT = { has_role = character_role_prince } }
				}
			}
			random_scope_character = {
				limit = {
					has_role = character_role_prince
					is_ruler_of_own_country = no
				}
				save_scope_as = THIS_scope
				owner ?= { set_heir = scope:THIS_scope }
			}
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
		}
		set_variable = { name = recently_became_monarchy days = 1 }
		if = {
			limit = {
				OR = {
					any_scope_character = {
						has_role = character_role_monarch
						is_female = yes
					}
					AND = {
						shares_same_monarch = yes
						overlord ?= {
							any_scope_character = {
								has_role = character_role_monarch
								is_female = yes
							}
						}
					}
				}
			}
			remove_variable = primogeniture_system
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			ruler_is_governor_general = no
			shares_same_monarch = no
			any_scope_character = {
				NOT = { has_role = character_role_regent }
				has_role = character_role_monarch
				is_eligible_to_become_monarch = yes
			}
		}
		trigger_event = { id = regency.3 popup = yes }
	}

	#Set term limits
	if = {
		limit = {
			has_law_or_variant = law_type:law_HOS_monarch
			NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			ruler_is_governor_general = no
			shares_same_monarch = no
		}
		activate_law = law_type:law_yes_term_limit
		if = {
			limit = { NOT = { active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy } } }
			active_law:lawgroup_governance_principles ?= {
				add_amendment = {
					type = amendment_elective_monarchy
					sponsor = PREV.ig:ig_petty_bourgeoisie 
				}
			}
		}
		heir ?= {
			if = {
				limit = { NOT = { has_role = character_role_monarch_elective } }
				add_character_role = character_role_monarch_elective
			}
			disinherit_character = yes
		}
		generate_elective_monarchs_for_country = yes
	}
	
	#Deal with change of parliamentary elective
	remove_modifier = starting_voting_power_modifier

}
