generate_governor_for_country = {

	#Generate Governor
	if = {
		limit = {
			NOT = {
				any_scope_character = {
					has_role = character_role_governor
				}
			}
		}
		if = {
			limit = { 
				has_a_republican_governor_HOG = yes
			}
			
			if = {
				limit = { has_variable = elected_governor_system }
				ordered_scope_character = {
					limit = {
						is_ruler_of_own_country = no
						has_role_of_type = politician
						character_has_legal_restrictions = no
					}
					order_by = popularity
					position = 0
					check_range_bounds = no
					set_character_as_governor = yes
					set_variable = { name = new_governor days = 1 }
				}
			}
			else = {
				overlord ?= { 
					save_scope_as = dominant_country
					random_primary_culture = { save_scope_as = dominant_country_culture }
					ruler ?= { interest_group = { save_scope_as = dominant_country_IG } }
				}
				create_character = {
					role = character_role_governor
					culture = scope:dominant_country_culture
					religion = scope:dominant_country.religion
					interest_group = scope:dominant_country_IG
					on_created = {
						#add_character_role = character_role_agitator
						set_variable = { name = new_governor days = 1 }
					}
				}
				random_scope_character = {
					limit = { has_variable = new_governor }
					remove_variable = new_governor
					set_home_country = scope:dominant_country
					set_home_state = scope:dominant_country.capital
					set_appropriate_character_ideology_from_overlord = yes
				}
			}
		}
	}
	else_if = {
		limit = { 
			any_scope_character = {
				has_role = character_role_governor
			}
		}
		if = {
			limit = { 
				has_a_republican_governor_HOG = yes
				NOT = {
					any_scope_character = {
						has_role = character_role_governor
						root.overlord = this.home_country
					}
				}
			}
			if = {
				limit = { has_variable = elected_governor_system }
				ordered_scope_character = {
					limit = {
						is_ruler_of_own_country = no
						has_role_of_type = politician
						character_has_legal_restrictions = no
					}
					order_by = popularity
					position = 0
					check_range_bounds = no
					set_character_as_governor = yes
					set_variable = { name = new_governor days = 1 }
				}
			}
			else = {
				overlord ?= { 
					save_scope_as = dominant_country
					random_primary_culture = { save_scope_as = dominant_country_culture }
					ruler ?= { interest_group = { save_scope_as = dominant_country_IG } }
				}
				create_character = {
					role = character_role_governor
					culture = scope:dominant_country_culture
					religion = scope:dominant_country.religion
					interest_group = scope:dominant_country_IG
					on_created = {
						#add_character_role = character_role_agitator
						set_variable = { name = new_governor days = 1 }
					}
				}
				random_scope_character = {
					limit = { has_variable = new_governor }
					remove_variable = new_governor
					set_home_country = scope:dominant_country
					set_home_state = scope:dominant_country.capital
					set_appropriate_character_ideology_from_overlord = yes
				}
			}
		}
	}
	
}
