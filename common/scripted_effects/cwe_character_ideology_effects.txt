add_interest_group_leader_weight_ideology_autocratic = {

	# Effect of Party Lean
	if = {
		limit = {
			is_in_government = yes
			#has_party = yes
			owner = {
				has_variable = very_radical_party
			}
		}
		add = 1000
	}
	# Effect of existing IG leader
	else_if = {
		limit = {
			is_in_government = yes
			#has_party = no
			owner = {
				government_legitimacy >= 50
			}
			leader ?= { has_ideology_autocratic = yes }
		}
		add = 500
	}
	
	# angry IGs are more likely to have autocratic opinions
	if = {
		limit = {
			ig_approval < -5
		}
		add = 25
	}
	if = {
		limit = {
			ig_approval < -10
		}
		add = 50
	}
	
	# IGs in power are more likely to have autocratic opinions
	if = {
		limit = {
			is_in_government = yes
		}
		add = 50
	}
	
	# more likely if you border a rival
	if = {
		limit = {
			owner = {
				any_rival_country = {
					any_neighbouring_state = {
						owner = root.owner
					}
				}
			}
		}
		add = 25
	}
	# more likely if you're the defender in a war
	if = {
		limit = {
			owner = {
				is_at_war = yes
				any_scope_war = {
					defender_warleader = { this = ROOT.owner }
				}
			}
		}
		add = 25
	}
	
	# Cycle of life
	if = {
		limit = {
			is_in_government = no
			owner = { is_a_democracy = yes }
		}
		add = 50
	}
	if = {
		limit = {
			is_in_government = no
			owner = { is_a_dictatorship = yes }
		}
		add = -50
	}
	
	# National Values
	if = {
		limit = {
			owner = { has_modifier = order_national_value }
		}
		add = 25
	}
	
	# World Tension
	if = {
		limit = {
			owner = { has_global_variable = very_high_world_tension }
		}
		add = 25
	}

}

				exists = ig:ig_armed_forces

					is_powerful = yes
					is_in_government = yes
				}
			}
		}
		add = 10

					}

	}
	if = {

		}
		add = 50
	}
