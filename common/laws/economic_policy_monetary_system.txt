#### Represents Financial & Industrial Regulations ####

law_independent_central_bank = {
	group = lawgroup_monetary_system
	
	icon = "gfx/interface/icons/law_icons/consumption_based_taxation.dds"
	
	can_enact = {
		
		#Not part of a Currency Union
		custom_tooltip = {
			text = je_bretton_woods_system_member.4.tt
			NOT = {
				any_scope_diplomatic_pact = {
					is_diplomatic_action_type = currency_union
					second_country = ROOT
				}
			}
		}
		
	}
	
	on_activate = {
		custom_tooltip = je_central_bank_effect.tt
	}
	
	modifier = {
		
	}
	
	possible_political_movements = {
		#law_independent_central_bank
		law_semi_independent_central_bank
		law_non_independent_central_bank
	}
	
	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
		
	}
	
}
