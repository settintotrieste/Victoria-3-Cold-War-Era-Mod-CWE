# group = this is the law_group a law belongs to
# icon = graphical icon shown in-game
# modifier = {} modifier on country for having adopted this law

#For Presidential Republics & dictatorships in general
law_HOG_none = {
	group = lawgroup_head_of_government
	
	icon = "gfx/interface/icons/law_icons/neo_absolutism.dds"
	
	can_impose = { can_impose_law_default = yes } #progressiveness = -50

	disallowing_laws = {
		
	}
	
	can_enact = {
		NOT = { 
			active_law:lawgroup_governance_principles ?= {
				has_amendment = amendment_type:amendment_symbolic_monarchy
			}
		}
	}
	
	on_activate = {
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				any_scope_character = {
					has_role = character_role_regent
					is_ruler_of_own_country = no 
				}
			}
			every_scope_character = {
				limit = { 
					has_role = character_role_regent
					is_ruler_of_own_country = no 
				}
				remove_character_role = character_role_regent
				clean_up_regent_character = yes
			}
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				any_scope_character = {
					has_role = character_role_monarch
					is_ruler_of_own_country = no 
				}
			}
			random_scope_character = {
				limit = { has_role = character_role_monarch }
				set_character_as_monarch = yes
			}
		}
	}

	modifier = {
		country_authority_mult = 0.50
		country_legitimacy_headofstate_add = 50
	}
	
	possible_political_movements = {
		#law_HOG_none
		law_HOG_prime_minister
		law_HOG_president_prime_minister
	}
	
	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
	}
	
}

#For Parliamentary Republic and Regimes
law_HOG_prime_minister = {
	group = lawgroup_head_of_government
	
	icon = "gfx/interface/icons/law_icons/parliamentary_republic.dds"
	
	can_impose = { can_impose_law_default = yes } #progressiveness = 0

	disallowing_laws = {
		
	}
	
	can_enact = {
		
	}
	
	on_activate = {
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				any_scope_character = {
					has_role = character_role_monarch
					is_ruler_of_own_country = yes 
				}
			}
			if = {
				limit = { any_scope_character = { has_role = character_role_regent } }
				random_scope_character = {
					limit = { has_role = character_role_regent }
					set_character_as_ruler = yes
				}
			}
			else_if = {
				limit = {
					any_scope_character = {
						has_role_of_type = magnate
						is_noble = yes
						is_adult = yes
					}
				}
				ordered_scope_character = {
					limit = {
						has_role_of_type = magnate
						is_noble = yes
						is_adult = yes
					}
					order_by = popularity
					position = 0
					set_character_as_ruler = yes
				}
			}
			else_if = {
				limit = {
					any_scope_character = {
						OR = { 
							has_role_of_type = general
							has_role_of_type = admiral
						}
					}
				}
				ordered_scope_character = {
					limit = {
						OR = { 
							has_role_of_type = general
							has_role_of_type = admiral
						}
					}
					order_by = commander_rank
					position = 0
					set_character_as_ruler = yes
				}
			}
			else = {
				create_character = {
					ruler = yes noble = yes
				}
			}
		}
	}

	modifier = {
		country_legitimacy_total_votes_add = 50
		country_legitimacy_ideological_incoherence_mult = -0.20
	}
	
	possible_political_movements = {
		law_HOG_none
		#law_HOG_prime_minister
		law_HOG_president_prime_minister
	}
	
	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
	}
	
}
