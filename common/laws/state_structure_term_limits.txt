#### Term Limits ####

law_no_term_limit = {
	group = lawgroup_term_limits
	
	icon = "gfx/interface/icons/law_icons/autocracy.dds"
	
	can_impose = { can_impose_law_default = yes } #progressiveness = -50
	
	on_activate = {
		
		#Remove Elective Monarchy
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
				ruler_is_governor_general = no
				shares_same_monarch = no
			}
			active_law:lawgroup_governance_principles ?= {
				random_scope_amendment = {
					limit = {
						type = amendment_type:amendment_elective_monarchy
					}
					remove_amendment = yes
				}
			}
		}
		
	}
	
	can_enact = {
		
	}
	
	modifier = {
		
	}
	
	possible_political_movements = {
		#law_no_term_limit
		law_yes_term_limit
	}
	
	pop_support = { value = 0 }
	
	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
	}
	
}

law_yes_term_limit = {
	group = lawgroup_term_limits
	
	icon = "gfx/interface/icons/law_icons/parliamentary_republic.dds"
	
	can_impose = { can_impose_law_default = yes } #progressiveness = -50
	
	on_activate = {
		
		#Reset Dynastic Republic
		if = {
			limit = { 
				is_a_dynastic_government = yes
			}
			custom_tooltip = {
				text = abolish_dynastic_republic.tt
				ruler ?= { remove_variable = dynastic_ruler_enabled }
				heir ?= { disinherit_character = yes }
			}
		}
		
		#Add Elective Monarchy
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
				ruler_is_governor_general = no
				shares_same_monarch = no
				NOT = { active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy } }
			}
			#Add Amendment
			active_law:lawgroup_governance_principles ?= {
				add_amendment = {
					type = amendment_elective_monarchy
					sponsor = PREV.ig:ig_petty_bourgeoisie 
				}
			}
			generate_elective_monarchs_for_country = yes
		}
		
	}
	
	can_enact = {
		
	}
	
	modifier = {
		
		#More agitation
		political_movement_radicalism_add = -0.10
		
	}
	
	possible_political_movements = {
		law_no_term_limit
		#law_yes_term_limit
	}
	
	pop_support = { value = 0 }
	
	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
	}
	
}
