legal_union = {
	groups= { general } requires_approval = yes
	can_use_obligations = yes

	potential = {
		
		#Has judiciary
		hidden_trigger = { 
			has_journal_entry = je_judiciary
			scope:target_country = { has_journal_entry = je_judiciary }
		}
		
		#THIS is not decentralized
		NOT = { is_country_type = decentralized }
		scope:target_country = { NOT = { is_country_type = decentralized } }
		
		#THIS and Target are not revolutionary
		is_revolutionary = no is_secessionist = no is_warlord_state = no
		scope:target_country = { is_revolutionary = no is_secessionist = no is_warlord_state = no } 
		
		#THIS and Target are not subjects
		is_sufficiently_independent = yes
		scope:target_country = {
			OR = {
				AND = {
					is_sufficiently_independent = yes
					country_rank < root.country_rank
				}
				is_direct_subject_of = ROOT
			}
		}
		
		#TARGET is not already in a legal union
		NOT = { any_scope_diplomatic_pact = { is_diplomatic_action_type = legal_union second_country = scope:target_country } }
		
	}

	possible = {
	
		#THIS is interested in the TARGET
		has_diplomatic_relevance = scope:target_country
		
		#Good relations
		OR = { 
		
			# THIS has good relations with TARGET
			custom_tooltip = {
				text = DIPLO_ACTION_RELATIONS_CORDIAL_OR_GREATER
				scope:target_country.relations:root >= relations_threshold:cordial
			}
			
			#TARGET is a subject of THIS
			scope:target_country = { 
				is_subject_of = scope:actor 
				is_not_sufficiently_independent = yes
			}
			
		}
		
		#Sufficient government similarity
		OR = {
		
			#THIS shares the same government type as TARGET
			custom_tooltip = {
				text = has_same_gov_type_tt
				OR = {
					AND = { is_some_kind_of_democracy = yes scope:target_country = { is_some_kind_of_democracy = yes } }
					AND = { is_a_communist_government = yes scope:target_country = { is_a_communist_government = yes } }
					AND = { is_a_nationalist_government = yes scope:target_country = { is_a_nationalist_government = yes } }
					AND = { is_a_theocratic_government = yes scope:target_country = { is_a_theocratic_government = yes } }
				}
			}
			
			#TARGET is a subject of THIS
			scope:target_country = { 
				is_direct_subject_of = ROOT
				OR = {
					is_a_colonial_subject_gov = yes
					is_subject_type = subject_type_special_administrative_region
					is_subject_type = subject_type_confederal_state
				}
			}
			
		}
		
	}
	
	accept_effect = {
	
		#Lobby
		add_lobby_appeasement_from_diplomacy_bidirectional = { 
			FIRST = root
			SECOND = scope:target_country 
			PRO_AMOUNT = 2
			ANTI_AMOUNT = -2
			FACTOR = appeasement_defensive_pact_formed
		}
	
		change_relations = {
			country = scope:target_country
			value = 25
		}
		
		#Harmonise judiciary of TARGET with THIS
		custom_tooltip = {
			text = legal_union_1.tt
			save_scope_as = legal_union_leader
			scope:target_country = { 
				save_scope_as = legal_union_member
				if = {
					limit = {
						scope:actor = {
							has_variable = judiciary_size
						}
					}
					set_variable = { 
						name = judiciary_size
						value = scope:actor.var:judiciary_size
					}
				}
				if = {
					limit = {
						scope:actor = {
							has_variable = judiciary_progressiveness
						}
					}
					set_variable = {
						name = judiciary_progressiveness
						value = scope:actor.var:judiciary_progressiveness
					}
				}
			}
		}
		
		#Harmonise laws of TARGET if TARGET is a confederal state of THIS
		if = {
			limit = {
				scope:target_country = { 
					is_direct_subject_of = scope:actor 
					is_subject_type = subject_type_confederal_state
				}
			}
			custom_tooltip = {
				text = harmonise_confederal_state_law.tt
				scope:target_country = { set_variable = THIS_in_confederal_legal_union }
			}
		}
		
	}
	
	pact = {
		cost = 5
		has_junior_participant = yes
		
		can_be_used_in_sway_offers = no
		
		max_target_involvement = 1000
		target_involvement_applies_to = mutual
		
		second_modifier = {
			country_pact_leverage_generation_add = 100
		}
		
		actor_can_break = {
			
		}
		
		target_can_break = {
			
			#Subjects cannot freely leave Currency Union
			NOT = {
				scope:target_country = { 
					is_direct_subject_of = scope:actor 
					is_sufficiently_independent = no 
				} 
			}
			
		}
		
		requirement_to_maintain = { 
		
			trigger = {
				
				#THIS and Target are not subjects
				is_sufficiently_independent = yes
				scope:target_country = { 
					OR = { 
						is_sufficiently_independent = yes
						is_direct_subject_of = ROOT
					}
				}
				
				#TARGET and THIS are not at war
				NOT = { has_war_with = scope:target_country }
				
				#More than 2 legal union leaders
				NOT = {
					any_scope_diplomatic_pact = { 
						is_diplomatic_action_type = legal_union 
						second_country = ROOT
						count > 1
					}
				}
				scope:target_country = {
					NOT = {
						any_scope_diplomatic_pact = { 
							is_diplomatic_action_type = legal_union 
							second_country = scope:target_country
							count > 1
						}
					}
				}
				
			}
			
		}
		
		monthly_effect = {
		
			#Harmonise judiciary of TARGET with THIS
			scope:target_country = { 
				if = {
					limit = {
						scope:actor = {
							has_variable = judiciary_size
						}
					}
					set_variable = { 
						name = judiciary_size
						value = scope:actor.var:judiciary_size
					}
				}
				if = {
					limit = {
						scope:actor = {
							has_variable = judiciary_progressiveness
						}
					}
					set_variable = {
						name = judiciary_progressiveness
						value = scope:actor.var:judiciary_progressiveness
					}
				}
			}
		
		}
		
		manual_break_effect = {
		
			change_relations = {
				country = scope:target_country
				value = -25
			}
			
			custom_tooltip = {
				text = legal_union_2.tt
				save_scope_as = legal_union_leader
				scope:target_country = { 
					save_scope_as = legal_union_member
				
					#Default is an even split
					set_variable = {
						name = judiciary_progressiveness
						value = 6
					}
					
					#Default Supreme Court size is 12
					set_variable = {
						name = judiciary_size
						value = 12
					}
					
				}
			}
			
			if = {
				limit = { 
					scope:target_country = { 
						is_direct_subject_of = scope:actor
						is_subject_type = subject_type_confederal_state
					} 
					has_variable = THIS_in_confederal_legal_union
				}
				remove_variable = THIS_in_confederal_legal_union
			}
			
		}
		
		auto_break_effect = {
			
			custom_tooltip = {
				text = legal_union_2.tt
				save_scope_as = legal_union_leader
				scope:target_country = { 
					save_scope_as = legal_union_member
				
					#Default is an even split
					set_variable = {
						name = judiciary_progressiveness
						value = 6
					}
					
					#Default Supreme Court size is 12
					set_variable = {
						name = judiciary_size
						value = 12
					}
					
				}
			}
			
			if = {
				limit = { 
					scope:target_country = { 
						is_direct_subject_of = scope:actor
						is_subject_type = subject_type_confederal_state
					} 
					has_variable = THIS_in_confederal_legal_union
				}
				remove_variable = THIS_in_confederal_legal_union
			}
			
		}
		
	}
	
	ai = {
		check_acceptance_for_will_break = yes
		check_acceptance_for_will_propose = yes

		evaluation_chance = {
			value = 0.25		
		}

		# This is the acceptance score for the senior part of the customs union (used for 'join customs union' action and checking )
		accept_score = {
			value = 0

			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_BASE"
				value = -200
			}

			if = {
				limit = { 
					has_strategy = ai_strategy_economic_imperialism
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ECONOMIC_IMPERIALISM"
					add = 50
				}
			}

			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_GDP"
				value = "scope:actor.ai_gdp_comparison(root)"
				min = -10
				max = 10
			}
		
			if = {
				limit = {
					"scope:actor.ai_ideological_opinion(root)" > 0
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_IDEOLOGICAL_OPINION"
					value = "scope:actor.ai_ideological_opinion(root)"
					multiply = 0.5
					max = 20
				}					
			}
			else_if = {
				limit = {
					"scope:actor.ai_ideological_opinion(root)" < 0
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_IDEOLOGICAL_OPINION_NEGATIVE"
					value = "scope:actor.ai_ideological_opinion(root)"
					multiply = 0.5
					min = -20
				}					
			}	
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
				if = {
					limit = {
						OR = {
							has_attitude = {
								who = scope:actor
								attitude = protective
							}
							has_attitude = {
								who = scope:actor
								attitude = domineering
							}							
						}
					}
					value = 40
				}
				if = {
					limit = {
						has_attitude = {
							who = scope:actor
							attitude = genial
						}
					}
					value = 30
				}				
				if = {
					limit = {
						OR = {
							has_attitude = {
								who = scope:actor
								attitude = conciliatory
							}	
							has_attitude = {
								who = scope:actor
								attitude = cooperative
							}							
						}
					}
					value = 20
				}
				if = {
					limit = {
						has_attitude = {
							who = scope:actor
							attitude = cautious
						}
					}
					value = -50
				}
				if = {
					limit = {
						OR = {
							has_attitude = {
								who = scope:actor
								attitude = antagonistic
							}
							has_attitude = {
								who = scope:actor
								attitude = belligerent
							}
							has_attitude = {
								who = scope:actor
								attitude = rebellious
							}
						}
					}
					value = -1000
				}
			}

			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_RELATIONS"
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:amicable
						"relations(scope:actor)" < relations_threshold:friendly
					}
					value = 20
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:friendly
					}
					value = 50
				}
			}

			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_INFAMY"
				if = {
					limit = {
						"scope:actor.infamy" >= infamy_threshold:infamous
						"scope:actor.infamy" < infamy_threshold:notorious
					}
					value = -10
				}
				if = {
					limit = {
						"scope:actor.infamy" >= infamy_threshold:notorious
						"scope:actor.infamy" < infamy_threshold:pariah
					}
					value = -25
				}
				if = {
					limit = {
						"scope:actor.infamy" >= infamy_threshold:pariah
					}
					value = -50
				}
			}
		}

		# This is the acceptance score for the junior part of the customs union
		junior_accept_score = {
			value = 0
			
			if = {
				limit = {
					scope:target_country = {
						is_subject = yes
						NOT = { is_subject_of = scope:actor }
					}
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE"
					value = -100
				}
			}
			else = {
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE"
					value = -50
				}
			}

			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_GDP"
				value = "scope:actor.ai_gdp_comparison(root)"
				multiply = 2
				min = -50
				max = 50
			}
			
			#Colonies love to enter into currency unions with its overlord
			if = {
				limit = { 
					scope:target_country = { 
						is_subject_of = scope:actor
						OR = {
							is_a_devolved_gov = yes
							is_subject_type = subject_type_confederal_state
						}
					}
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_SENIOR_PART"
					value = 100
					subtract = scope:target_country.liberty_desire
				}
			}

			if = {
				limit = {
					"scope:actor.ai_ideological_opinion(root)" > 0
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_IDEOLOGICAL_OPINION"
					value = "scope:actor.ai_ideological_opinion(root)"
					multiply = 0.5
					max = 20
				}					
			}
			else_if = {
				limit = {
					"scope:actor.ai_ideological_opinion(root)" < 0
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_IDEOLOGICAL_OPINION_NEGATIVE"
					value = "scope:actor.ai_ideological_opinion(root)"
					multiply = 0.5
					min = -20
				}					
			}

			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
				if = {
					limit = {
						has_attitude = {
							who = scope:actor
							attitude = loyal
						}
					}
					value = 50
				}
				if = {
					limit = {
						has_attitude = {
							who = scope:actor
							attitude = genial
						}
					}
					value = 40
				}				
				if = {
					limit = {
						OR = {
							has_attitude = {
								who = scope:actor
								attitude = conciliatory
							}	
							has_attitude = {
								who = scope:actor
								attitude = cooperative
							}
							has_attitude = {
								who = scope:actor
								attitude = protective
							}							
						}
					}
					value = 30
				}
				if = {
					limit = {
						has_attitude = {
							who = scope:actor
							attitude = cautious
						}
					}
					value = -50
				}
				if = {
					limit = {
						OR = {							
							has_attitude = {
								who = scope:actor
								attitude = antagonistic
							}
							has_attitude = {
								who = scope:actor
								attitude = domineering
							}
							has_attitude = {
								who = scope:actor
								attitude = belligerent
							}
							has_attitude = {
								who = scope:actor
								attitude = rebellious
							}
						}
					}
					value = -1000
				}
			}

			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_RELATIONS"
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:amicable
						"relations(scope:actor)" < relations_threshold:friendly
					}
					value = 20
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:friendly
					}
					value = 50
				}
			}

			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_INFAMY"
				if = {
					limit = {
						"scope:actor.infamy" >= infamy_threshold:infamous
						"scope:actor.infamy" < infamy_threshold:notorious
					}
					value = -10
				}
				if = {
					limit = {
						"scope:actor.infamy" >= infamy_threshold:notorious
						"scope:actor.infamy" < infamy_threshold:pariah
					}
					value = -25
				}
				if = {
					limit = {
						"scope:actor.infamy" >= infamy_threshold:pariah
					}
					value = -50
				}
			}

			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_RANK"

				if = {
					limit = { scope:actor = { country_rank = root.country_rank } }
					value = -20
				}

				if = {
					limit = { scope:actor = { country_rank < root.country_rank } }
					value = -50
				}
			}
			
		}
		
		will_break = {
			OR = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}
				has_attitude = {
					who = scope:target_country
					attitude = domineering
				}
				has_attitude = {
					who = scope:target_country
					attitude = belligerent
				}
				has_attitude = {
					who = scope:target_country
					attitude = rebellious
				}
				AND = {
					NOT = {
						has_law_or_variant = law_type:law_HOS_communist_leader
					}
					scope:target_country = {
						has_law_or_variant = law_type:law_HOS_communist_leader	
					}	
				}
			}
		}

		propose_score = {
			value = 10
		}

		propose_break_score = {
			value = 10
		}

		use_obligation_chance = {
			value = 25
		}

		owe_obligation_chance = {
			value = 0
		}
	}
	
}
