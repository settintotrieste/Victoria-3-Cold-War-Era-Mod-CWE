no_tolls = {
    kind = directed
    cost = 25

    relations_progress_per_day = 0.5
    relations_improvement_max = 20

	max_target_involvement = 500
	target_involvement_applies_to = source_country

	icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/no_tolls_treaties.dds"

	flags = {
		is_no_tolls
		can_be_enforced
		ai_consider_threaten_naval_hostilities
		can_be_renegotiated
		giftable
	}

	maintenance_paid_by = target_country
	usage_limit = once_per_side

    unlocked_by_technologies = {
        
    }

	visible = {
		has_any_strait_province = yes
		is_not_revolutionary_or_decentralised = yes
		is_sufficiently_independent = yes
	}

	possible = {
		has_any_strait_province = yes
		has_any_strait_control = yes
	}

    can_ratify = {
		NOT = {
			custom_tooltip = {
				text = DUPLICATE_ARTICLE_SAME_INPUTS_TREATY
				scope:source_country = {
					any_scope_treaty = {
						binds = scope:target_country
						hidden_trigger = {
							OR = {
								scope:treaty = {
									is_renegotiation = no
								}
								scope:treaty.amended_treaty != this
							}
						}
						any_scope_article = {
							has_type = no_tolls
							source_country = scope:source_country
						}
					}
				}
			}
		}
        custom_tooltip = {
            text = DUPLICATE_ARTICLE_SAME_INPUTS_TREATY
            NOT = {
				scope:treaty = {
					any_scope_article_option = {
						has_type = toll_exemption
						source_country = scope:source_country
					}
				}
            }
        }
    }

    on_entry_into_force = {}

	on_withdrawal = {}
	on_break = {
		scope:withdrawing_country = {
			change_relations = { country = scope:non_withdrawing_country value = -10 }
		}
	}

	ai = {

		treaty_categories = {
			economy
		}

		evaluation_chance = {
			value = 0

			if = {
				limit = {
					country_rank >= rank_value:unrecognized_major_power
				}
				add = {
					value = country_rank
					multiply = 0.025
				}
			}

			# Any country a strait controller is hurting looks for relief, whatever its rank
			add = {
				value = worst_strait_toll_grievance
				multiply = 0.005
			}
		}

		article_ai_usage = {
			request
		}

		proposal_weight = {
			value = 1

			# Pursued harder the more the strait controller is hurting us
			add = {
				value = "strait_toll_grievance_against(scope:source_country)"
				multiply = 0.04
			}

			if = {
				limit = {
					root = scope:target_country
					NOT = { has_law_or_variant = law_type:law_free_trade } # Everyone else asks for a toll exemption for themselves instead
				}
				value = 0
			}
		}

		inherent_accept_score = {
			value = 0

			if = {
				limit = {
					scope:source_country ?= this
				}

				# We are doing this to be able to reuse loc across treaties and diplomatic actions
				scope:source_country = {
					save_temporary_scope_as = target
				}
				scope:target_country = {
					save_temporary_scope_as = actor
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = -10
				}

				add = {
					desc = "concept_tolls"
					value = {
						scope:target_country = {
							add = "strait_trade_dependence_by(scope:source_country)"
						}
					}
					multiply = -0.5
					min = -50
				}
			}
			else = {
				scope:source_country = {
					save_temporary_scope_as = actor
				}
				scope:target_country = {
					save_temporary_scope_as = target
				}

				if = {
					limit = {
						scope:target_country = {
							"strait_trade_dependence_by(scope:source_country)" < define:NDiplomacy|STRAIT_GRIEVANCE_MIN_DEPENDENCE
						}
					}

					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_NO_STRAIT_DEPENDENCE"
						value = -10
					}
				}
				else = {
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_STRAIT_DEPENDENCE"
						value = {
							scope:target_country = {
								add = "strait_trade_dependence_by(scope:source_country)"
							}
						}
						multiply = 0.15
						max = 15
					}
				}
			}
		}

		wargoal_score_multiplier = {
			value = 0
			if = {
				limit = {
					has_law_or_variant = law_type:law_free_trade # Everyone else demands a toll exemption for themselves instead
					"strait_toll_grievance_against(scope:source_country)" > 0
				}
				value = 1
				add = {
					value = "strait_toll_grievance_against(scope:source_country)"
					multiply = 0.15
				}
			}
		}
	}

	wargoal = {
		execution_priority = 21

		contestion_type = control_any_target_country_state

		maneuvers = {
			value = 0

			add = {
				desc = "MANEUVERS_BASE_VALUE"
				value = 1
			}

			scope:target_country = {
				multiply = {
					desc = "MANEUVERS_TARGET_COUNTRY_POPULATION_FACTOR"
					value = total_population
					divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
					multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
                    max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_TOTAL
				}
			}
		}

		infamy = {
			value = 0
		}
	}
	
}
