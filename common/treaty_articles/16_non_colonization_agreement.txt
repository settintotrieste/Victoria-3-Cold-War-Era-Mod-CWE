non_colonization_agreement = {
    kind = directed
    cost = 10
	
	max_target_involvement = 1000
	target_involvement_applies_to = source_country

    relations_progress_per_day = -0.25
    relations_improvement_min = 0

	icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/non_colonization_agreement_treaties.dds"

	flags = {
		is_non_colonization_agreement
		can_be_enforced
		can_be_renegotiated
		hostile
		ai_consider_threaten_naval_hostilities
	}

	usage_limit = once_per_side_with_same_inputs

	maintenance_paid_by = target_country

    required_inputs = {
        strategic_region
    }

    unlocked_by_technologies = {
        
    }

	visible = {
		is_not_revolutionary_or_decentralised = yes
		is_sufficiently_independent = yes
	}

	possible = {
		
		custom_tooltip = {
			text = has_colonial_growth_tt
			has_colonial_growth = yes
		}
		
		#Proper countries
		is_not_revolutionary_or_decentralised = yes
		scope:other_country = { is_not_revolutionary_or_decentralised = yes }
		
		#Both parties are sufficiently independent
		is_sufficiently_independent = yes
		scope:other_country = { is_sufficiently_independent = yes }
		
		#Neither parties are subjects of the other
		scope:other_country = { NOT = { is_subject_of = ROOT } } 
		NOT = { is_subject_of = scope:other_country }
		
	}

	can_ratify = {
        custom_tooltip = {
            text = DUPLICATE_ARTICLE_SAME_INPUTS_TREATY
            NOT = {
                scope:source_country = {
                    any_scope_treaty = {
						hidden_trigger = {
							OR = {
								scope:treaty = {
									is_renegotiation = no
								}
								scope:treaty.amended_treaty != this
							}
						}
                        any_scope_article = {
                            has_type = non_colonization_agreement
                            input_strategic_region = root.input_strategic_region
							source_country = scope:source_country
                        }
                    }
                }
            }
        }
		custom_tooltip = {
            text = DUPLICATE_ARTICLE_SAME_INPUTS
            scope:treaty ?= {
                NOT = {
                    any_scope_article_option = {
                        count > 1
                        has_type = non_colonization_agreement
                        source_country = scope:source_country
                        input_strategic_region = root.input_strategic_region
                    }
                }
            }
        }
	}

    goods_valid_trigger = {
        root.market = {
            scope:market_goods = {
                market_goods_sell_orders >= 10
            }
        }
        scope:input = {
            is_tradeable = yes
        }
    }

    on_entry_into_force = {}

	on_withdrawal = {}
	on_break = {
		scope:withdrawing_country = {
			change_infamy = {
				value = {
					add = 1
					multiply = {
						value = scope:treaty_options.remaining_binding_period
						min = 365
						divide = 365
					}
					max = 20
					min = 2
				}
			}
		}
		scope:withdrawing_country = {
			change_relations = { country = scope:non_withdrawing_country value = -10 }
		}
	}

	ai = {

		treaty_categories = {
			expansion
		}

		evaluation_chance = {
			value = 0

			if = {
				limit = {
					country_rank >= rank_value:major_power
				}
				add = 0.25
			}
		}

        article_ai_usage = {
            request
        }

		inherent_accept_score = {
			value = 0

			if = {
				limit = {
					exists = scope:article.input_strategic_region
					scope:article.input_strategic_region = { "ai_colonization_value(scope:source_country)" > 0 }
					scope:article.input_strategic_region = { "ai_colonization_value(scope:target_country)" > 0 }
				}

				if = {
					limit = {
						scope:source_country ?= this
					}

					scope:source_country = {
						save_temporary_scope_as = target
					}
					scope:target_country = {
						save_temporary_scope_as = actor
					}

					scope:article.input_strategic_region = {
						subtract = {
							desc = "DIPLOMATIC_ACCEPTANCE_COLONIAL_INTEREST"
							value = "ai_colonization_value(scope:source_country)"
							multiply = 2
						}
					}

					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_RANK"

						if = {
							limit = {
								country_rank >= rank_value:great_power
							}
							value = -50
						}
						else_if = {
							limit = {
								country_rank >= rank_value:major_power
							}
							value = -25
						}
					}

					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_UNRECOGNIZED"

						if = {
							limit = {
								scope:target_country ?= {
									is_country_type = unrecognized
								}
							}
							value = -100
						}
					}

					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_COLONIAL_EXPANSION"

						if = {
							limit = {
								has_strategy = ai_strategy_colonial_expansion
							}
							add = -25
						}
					}

					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
						if = {
							limit = {
								OR = {
									has_attitude = {
										who = scope:target_country
										attitude = disinterested
									}
									has_attitude = {
										who = scope:target_country
										attitude = wary
									}
								}
							}
							value = -300
						}
					}

					add = {
						desc = "concept_claims"
						if = {
							limit = {
								scope:article.input_strategic_region = {
									any_scope_state = {
										has_claim_by = scope:source_country
									}
								}
							}
							value = -1000
						}
					}
				}
				else_if = {
					limit = {
						scope:target_country ?= this
						OR = {
							has_colonial_growth = yes
							scope:target_country.capital.region = {
								this = scope:article.input_strategic_region
							}
						}
					}

					scope:source_country = {
						save_temporary_scope_as = target
					}
					scope:target_country = {
						save_temporary_scope_as = actor
					}

					scope:article.input_strategic_region = {
						add = {
							desc = "DIPLOMATIC_ACCEPTANCE_COLONIAL_INTEREST"
							value = "ai_colonization_value(scope:target_country)"
						}
					}

					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_COLONIAL_EXPANSION"

						if = {
							limit = {
								NOT = { exists = scope:is_treaty_active }
								has_strategy = ai_strategy_colonial_expansion
							}
							add = 10
						}
					}

					add = {
						desc = "concept_claims"
						if = {
							limit = {
								scope:article.input_strategic_region = {
									any_scope_state = {
										has_claim_by = scope:target_country
									}
								}
							}
							value = 10
						}
					}
				}
				else_if = {
					limit = { NOT = { exists = scope:is_treaty_active } }
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_DOES_NOT_WANT_ARTICLE"
						value = -300
					}
				}
				else = {
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_DOES_NOT_WANT_ARTICLE"
						value = -30
					}
				}
			}
			else = {
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_DOES_NOT_WANT_ARTICLE"
					value = -100
				}
			}
		}
	}

	wargoal = {
		execution_priority = 21
		contestion_type = control_any_target_incorporated_state

		maneuvers = {
			value = 2.5

			scope:target_country = {
				multiply = {
					value = total_population
					divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
					multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
                    max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_TOTAL
				}
			}
		}

		infamy = {
			value = 0

			add = {
				desc = "INFAMY_NON_COLONIZATION_DECENTRALIZED"
				scope:region ?= {
					every_scope_state = {
						limit = {
							owner = {
								is_country_type = decentralized
							}
						}
						add = state_infamy_value
					}
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_NON_COLONIZATION_FRACTION
				min = define:NDiplomacy|WAR_GOAL_INFAMY_MODERATE_DEMAND_MIN
				max = define:NDiplomacy|WAR_GOAL_INFAMY_MODERATE_DEMAND_MAX
			}

			if = {
				limit = {
					scope:region ?= {
						any_scope_state = {
							has_claim_by = scope:target_country
						}
					}
				}
				multiply = {
					desc = "INFAMY_NON_COLONIZATION_TARGET_CLAIM"
					value = define:NDiplomacy|WAR_GOAL_INFAMY_NON_COLONIZATION_CLAIM_MULT
				}
			}
		}
	}
}
