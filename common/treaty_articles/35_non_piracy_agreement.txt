non_piracy_agreement = {
    kind = directed
    cost = 10

	max_target_involvement = 250
	target_involvement_applies_to = source_country

	icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/non_piracy_agreement.dds"

	flags = {
		show_in_lens
		is_non_piracy_agreement
		can_be_enforced
		ai_consider_threaten_naval_hostilities
		can_be_renegotiated
		giftable
	}

	maintenance_paid_by = target_country
	usage_limit = once_per_side

    unlocked_by_technologies = {
        international_relations
    }

	visible = {
		is_not_revolutionary_or_decentralised = yes
		is_sufficiently_independent = yes
	}

	possible = {
		
	}

    can_ratify = {
        custom_tooltip = {
            text = DUPLICATE_ARTICLE_SAME_INPUTS_TREATY
            NOT = {
                scope:source_country = {
                    any_scope_treaty = {
						binds = scope:target_country
                        hidden_trigger = {
							OR = {
								scope:treaty = {
									is_renegotiation = no
								}
								scope:treaty.amended_treaty != this
							}
						}
						any_scope_article = {
							has_type = non_piracy_agreement
							source_country = scope:source_country
						}
                    }
                }
            }
        }
        custom_tooltip = {
            text = DUPLICATE_ARTICLE_SAME_INPUTS_TREATY
            NOT = {
				scope:treaty = {
					any_scope_article_option = {
						has_type = abandon_piracy
						source_country = scope:source_country
					}
				}
            }
        }
    }

    on_entry_into_force = {}

	on_withdrawal = {}
	on_break = {
		scope:withdrawing_country = {
			change_relations = { country = scope:non_withdrawing_country value = -10 }
		}
	}

	ai = {
		treaty_categories = {
			economy
		}

		evaluation_chance = {
			value = 0.05

			if = {
				limit = {
					country_rank < rank_value:great_power
				}
				add = 0.05
			}

			# Any country a pirate is hurting looks for relief, whatever its rank
			if = {
				limit = { NOT = { has_law_or_variant = law_type:law_free_trade } }
				add = {
					value = worst_piracy_grievance
					multiply = 0.005
				}
			}
		}

		article_ai_usage = {
			request
			offer
		}

		proposal_weight = {
			value = 1

			# Pursued harder the more the pirate is hurting us
			add = {
				value = "piracy_grievance_against(scope:source_country)"
				multiply = 0.04
			}

			if = {
				limit = {
					root = scope:target_country
					has_law_or_variant = law_type:law_free_trade # Free traders ask for piracy to be abandoned against everyone instead
				}
				value = 0
			}
		}

		inherent_accept_score = {
			value = 0

			if = {
				limit = {
					scope:source_country ?= this
				}

				# We are doing this to be able to reuse loc across treaties and diplomatic actions
				scope:source_country = {
					save_temporary_scope_as = target
				}
				scope:target_country = {
					save_temporary_scope_as = actor
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = -10
				}

				if = {
					limit = {
						scope:target_country = {
							"pirated_trade_value_by(scope:source_country)" > 0
						}
					}
					subtract = {
						desc = "country_piracy_income_add"
						scope:target_country = {
							add = "pirated_trade_value_by(scope:source_country)"
						}
						divide = 100
						max = 50
					}
				}
			}
			else = {
				scope:source_country = {
					save_temporary_scope_as = actor
				}
				scope:target_country = {
					save_temporary_scope_as = target
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = -10
				}

				if = {
					limit = {
						scope:target_country = {
							"piracy_grievance_against(scope:source_country)" > 0
						}
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_PIRACY_GRIEVANCE"
						value = {
							scope:target_country = {
								add = "piracy_grievance_against(scope:source_country)"
							}
						}
						multiply = 0.3
						max = 30
					}
				}
			}
		}

		wargoal_score_multiplier = {
			value = 0
			if = {
				limit = {
					NOT = { has_law_or_variant = law_type:law_free_trade } # Free traders demand piracy be abandoned against everyone instead
					"piracy_grievance_against(scope:source_country)" > 0
				}
				value = 1
				add = {
					value = "piracy_grievance_against(scope:source_country)"
					multiply = 0.15
				}
			}
		}
	}

	wargoal = {
		execution_priority = 21

		contestion_type = control_any_target_country_state

		maneuvers = {
			value = 0

			add = {
				desc = "MANEUVERS_BASE_VALUE"
				value = 1
			}

			scope:target_country = {
				multiply = {
					desc = "MANEUVERS_TARGET_COUNTRY_POPULATION_FACTOR"
					value = total_population
					divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
					multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
                    max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_TOTAL
				}
			}
		}

		infamy = {
			value = 0
		}
	}
	
}
