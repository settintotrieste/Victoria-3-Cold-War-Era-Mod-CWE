dominate_politics = {
	icon = "gfx/interface/icons/war_goals/dominate_politics.dds"

	kind = custom

	settings = {
		require_target_be_part_of_war
		assent_required

		# Validation
		validate_conflicts_war_goals_all
		validate_conflicts_annex_country
		validate_conflicts_make_subject
		validate_conflicts_existing_subject
	}

	execution_priority = 20

	contestion_type = control_any_target_incorporated_state

	target_type = country

	possible = {
		# trigger to determine if a goal with its target data is listed when selecting a war goal in the diplo play panel
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
		
		#THIS is sufficiently independent
		is_sufficiently_independent = yes
		scope:target_country = { is_sufficiently_independent = yes }
		
		#Not already dominating politics of TARGET
		NOT = { has_diplomatic_pact = { who = scope:target_country type = dominate_politics is_initiator = yes } }
	}

	valid = {
		# trigger in addition to some basic validation code-side
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
		
		#Not already dominating politics of TARGET
		NOT = { has_diplomatic_pact = { who = scope:target_country type = dominate_politics is_initiator = yes } }
	}

	maneuvers = {
		value = 0

		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 10
		}
	}

	infamy = {
		value = 0

		scope:target_country = {
			add = country_infamy_value
		}

		multiply = {
			desc = "INFAMY_WAR_GOAL_SEVERITY"
			value = 0.05
		}

		multiply = {
			desc = "INFAMY_FROM_LOYAL_SUBJECT"
			value = 1

			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
						liberty_desire < define:NDiplomacy|WAR_GOAL_INFAMY_LOW_LIBERTY_DESIRE_THRESHOLD
					}
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_LOW_LIBERTY_DESIRE_SUBJECT_MULT
			}
		}

		multiply = {
			desc = "INFAMY_FROM_REBELLIOUS_SUBJECT" # Rebellious as in they are the aggressor in the play
			value = 1

			if = {
				limit = {
					scope:target_country = {
						is_subject_of = root
						is_diplomatic_play_enemy_of = root
						scope:diplomatic_play ?= {
							initiator_is = scope:target_country
						}
					}
				}
				multiply = define:NDiplomacy|WAR_GOAL_INFAMY_REBELLING_SUBJECT_MULT
			}
		}

		min = {
			desc = "INFAMY_MINIMUM_VALUE"
			value = define:NDiplomacy|WAR_GOAL_INFAMY_MAJOR_DEMAND_MIN
		}

		max = {
			desc = "INFAMY_MAXIMUM_VALUE"
			value = define:NDiplomacy|WAR_GOAL_INFAMY_MAJOR_DEMAND_MAX
		}
	}

	on_enforced = {
		# script effect on top of the predefined code effect
		
		#Create dominate politics
		create_diplomatic_pact = {
			country = scope:target_country
			type = dominate_politics
		}
	}
	
	mirrored_wargoal = {
		method = no_mirror
	}

	ai = {
		is_significant_demand = no
	}
}
