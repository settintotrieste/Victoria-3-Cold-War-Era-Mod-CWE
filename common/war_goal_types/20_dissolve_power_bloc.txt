dissolve_power_bloc = {
	icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/invite_to_power_bloc_treaties.dds"

	kind = custom

	settings = {
		require_target_be_part_of_war
		assent_required

		# Validation
		validate_conflicts_war_goals_all
		validate_conflicts_annex_country
		validate_conflicts_make_subject
		validate_conflicts_existing_subject
	}

	execution_priority = 21

	contestion_type = control_target_country_capital

	target_type = country

	possible = {
		# trigger to determine if a goal with its target data is listed when selecting a war goal in the diplo play panel
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
		
		#THIS are proper countries
		is_not_revolutionary_or_decentralised = yes
		scope:target_country = { is_not_revolutionary_or_decentralised = yes }
		
		#THIS is sufficiently independent
		is_sufficiently_independent = yes
		scope:target_country = { is_subject = no }
		
		#TARGET is a power bloc leader
		scope:target_country = { 
			is_power_bloc_leader = yes
		}
		
	}

	valid = {
		# trigger in addition to some basic validation code-side
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
		
		#THIS is a power bloc leader
		scope:target_country = { 
			is_power_bloc_leader = yes
			is_subject = no
		}
	}

	maneuvers = {
		value = 0

		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 10
		}
	}

	infamy = {
		value = 0

		scope:target_country = {
			add = country_infamy_value
		}

		multiply = {
			desc = "INFAMY_WAR_GOAL_SEVERITY"
			value = 0.05
		}

		min = {
			desc = "INFAMY_MINIMUM_VALUE"
			value = define:NDiplomacy|WAR_GOAL_INFAMY_MODERATE_DEMAND_MIN
		}

		max = {
			desc = "INFAMY_MAXIMUM_VALUE"
			value = define:NDiplomacy|WAR_GOAL_INFAMY_MODERATE_DEMAND_MAX
		}
	}

	on_enforced = {
		# script effect on top of the predefined code effect
		
		#Loser dissolves power bloc
		scope:target_country = {
			every_country = {
				limit = {
					is_in_same_power_bloc = scope:target_country
					is_power_bloc_leader = no
					is_subject = no
				}
				add_modifier = {
					name = forced_to_dissolve_power_bloc_modifier 
					years = 5 
				}
				save_scope_as = enemy_pb_member
				ROOT = {
					create_diplomatic_pact = {
						country = scope:enemy_pb_member
						type = confederal_state
					}
				}
				make_independent = yes
			}
			add_modifier = { 
				name = forced_to_dissolve_power_bloc_modifier 
				years = 5 
			}
		}
		create_diplomatic_pact = {
			country = scope:target_country
			type = confederal_state
		}
		scope:target_country = { make_independent = yes }
		
	}

	ai = {
		is_significant_demand = no
	}
}
