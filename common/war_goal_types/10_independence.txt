independence = {
	icon = "gfx/interface/icons/war_goals/independence.dds"

	kind = independence

	settings = {
		require_target_be_part_of_war
		can_add_for_other_country
		preserve_on_switching_sides

		# Validation
		validate_conflicts_war_goals_all
		validate_independence
	}

	execution_priority = 100

	contestion_type = control_all_own_states

	target_type = country

	possible = {
		# trigger to determine if a goal with its target data is listed when selecting a war goal in the diplo play panel
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
	}

	valid = {
		# trigger in addition to some basic validation code-side
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
		root.overlord = scope:target_country
	}

	fill_per_week = 2
	deplete_per_week = 2

	maneuvers = {
		value = 0

		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 20
		}
	}

	infamy = {
		value = 0
	}

	on_enforced = {
		# script effect on top of the predefined code effect
		
		#Subject becomes a republic
		scope:target_country = {
			if = {
				limit = {
					has_law_or_variant = law_type:law_HOS_monarch
					is_a_subsidiary_monarchy = no
					shares_same_monarch = yes
				}
				activate_law = law_type:law_HOS_president
				activate_law = law_type:law_HOG_prime_minister
			}
		}
		
	}

	mirrored_wargoal = {
		method = restore_subject
	}

	ai = {
		is_significant_demand = yes
	}
}
