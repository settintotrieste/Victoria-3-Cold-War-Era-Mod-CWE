proclaim_regent_monarch = {
    icon = "gfx/interface/icons/ideology_icons/ideology_leader/ideology_leader_royalist.dds"
	potential = {
		has_role = character_role_regent
		owner ?= { has_law_or_variant = law_type:law_HOS_monarch }
	}

	possible = {
       
		#No Monarchs
		custom_tooltip = {
            text = no_heir_tt
			OR = {
				NOT = { exists = owner.heir }
				owner.heir ?= { NOT = { has_role = character_role_monarch } }
			}
        }
		
		#Nobles need less time
		trigger_if = {
			limit = {
				OR = {
					AND = { 
						is_noble = yes
						has_role_of_type = magnate
					}
					has_role = character_role_prince
				}
			}
			custom_tooltip = {
				text = eterna_regency_5_tt
				has_variable = regency_years
				var:regency_years > 5
			}
		}
		
		#Commoners need time to gain legitimacy
		trigger_else = {
			custom_tooltip = {
				text = eterna_regency_tt
				has_variable = regency_years
				var:regency_years > 10
			}
		}
		
		#Parliament support
		owner ?= {
			custom_tooltip = {
				text = government_has_legislative_super_majority.tt
				government_has_legislative_super_majority = yes
			}
		}
		
	}

	effect = {
		
		#No longer becomes a regent
		hidden_effect = {
           clean_up_regent_character = yes
        }
		
		#Becomes a monarch
		if = {
			limit = {
				is_female = yes
			}
			custom_tooltip = {
				text = remove_primogeniture_system.tt
				remove_variable = primogeniture_system
			}
		}
		set_character_as_monarch = yes
		
		#Set popularity
        owner ?= {
            if = { # Sorts an edge case
                limit = {
                    NAND = {
                        root.popularity < 27
                        root.popularity > 23
                    }
                }
                custom_tooltip = effects_depend_on_popularity_tt
                add_modifier = {
                    name = modifier_new_royal_dynasty
                    days = long_modifier_time
                    is_decaying = yes
                    multiplier = {
                        value = root.popularity
                        subtract = 25
                        divide = 2
                        min = -25
                        max = 25
                    }
                }
            }
            remove_regency_modifier = yes
        }
		
	}

	cooldown = { days = long_modifier_time }
	show_confirmation_box = yes

	ai_chance = {
		value = 1
	}
	
}

restore_abdicated_monarch = {
    icon = "gfx/interface/icons/ideology_icons/ideology_leader/ideology_leader_royalist.dds"
	potential = {
		has_variable = prince_renounced_throne
		OR = {
			has_role = character_role_monarch
			has_role = character_role_prince
		}
		NOT = { has_role = character_role_regent }
		owner ?= { has_law_or_variant = law_type:law_HOS_monarch }
	}

	possible = {
		
		#Parliament support
		owner ?= {
			custom_tooltip = {
				text = government_has_legislative_super_majority.tt
				government_has_legislative_super_majority = yes
			}
		}
		
		#No other monarch
		trigger_if = {
			limit = { has_role = character_role_prince }
			owner ?= {
				custom_tooltip = {
					text = there_is_no_monarch.tt
					NOT = { any_scope_character = { has_role = character_role_monarch } }
				}
			}
		}
		
		#Not recently suspended monarch
		trigger_if = {
			limit = { has_role = character_role_monarch }
			custom_tooltip = {
				text = suspended_monarch_timer.tt
				NOT = { has_variable = suspended_monarch_timer }
			}
		}
		
	}

	effect = {
		
		#Become monarch
		if = {
			limit = { NOT = { has_role = character_role_monarch } }
			set_character_as_monarch = yes
		}
		
		#Set popularity
		if = {
			limit = { NOT = { has_role = character_role_monarch } }
			owner ?= {
				if = { # Sorts an edge case
					limit = {
						NAND = {
							root.popularity < 27
							root.popularity > 23
						}
					}
					custom_tooltip = effects_depend_on_popularity_tt
					add_modifier = {
						name = modifier_new_royal_dynasty
						days = long_modifier_time
						is_decaying = yes
						multiplier = {
							value = root.popularity
							subtract = 25
							divide = 2
							min = -25
							max = 25
						}
					}
				}
				remove_regency_modifier = yes
			}
		}
		else = {
			owner ?= {
				every_interest_group = {
					limit = {
						law_stance = {
							law = law_type:law_HOS_monarch
							value < neutral
						}
					}
					save_scope_as = THIS_IG
					owner ?= { 
						add_radicals = {
							value = large_radicals
							interest_group = scope:THIS_IG
						}
					}
				}
			}
		}
		
		#Restore to throne
		remove_variable = prince_renounced_throne
		
	}
	
	show_confirmation_box = yes
	
	should_ai_evaluate = {
		always = no
	}

	ai_chance = {
		value = 1
	}
	
}

suspend_monarch = {
    #icon = "gfx/interface/icons/ideology_icons/ideology_leader/ideology_leader_royalist.dds"
	potential = {
		has_role = character_role_monarch
		NOT = { has_variable = prince_renounced_throne }
		owner ?= { has_law_or_variant = law_type:law_HOS_monarch }
	}

	possible = {
		
		#Has debilitating illness
		OR = {
			has_trait = psychological_affliction
			has_trait = senile
			has_trait = cancer
			age >= 80
		}
		
		#Parliament support
		owner ?= {
			custom_tooltip = {
				text = government_has_legislative_super_majority.tt
				government_has_legislative_super_majority = yes
			}
		}
		
		#Not recently suspended monarch
		trigger_if = {
			limit = { has_role = character_role_monarch }
			custom_tooltip = {
				text = suspended_monarch_timer.tt
				NOT = { has_variable = suspended_monarch_timer }
			}
		}
		
	}

	effect = {
		
		#Anger Monarch's interest group
		interest_group ?= { save_scope_as = THIS_IG }
		owner ?= { 
			add_radicals = {
				value = large_radicals
				interest_group = scope:THIS_IG
			}
		}
		
		#Mark
		set_variable = prince_renounced_throne
		owner ?= { trigger_event = { id = regency.1 } }
		
		#Timer
		set_variable = { name = suspended_monarch_timer years = 1 }
		
	}
	
	show_confirmation_box = yes
	
	should_ai_evaluate = {
		always = no
	}

	ai_chance = {
		value = 1
	}
	
}
