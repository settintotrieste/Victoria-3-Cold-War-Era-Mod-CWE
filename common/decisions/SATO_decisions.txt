form_sato_decision = {

	is_shown = {
		
		#Date of establishment
		has_technology_researched = tech_geopolitics_3
		
		#SATO does not already exist
		NOR = { has_modifier = sato_member has_modifier = sato_leader }
		NOT = { any_country = { has_modifier = sato_leader } }
		
		#Cold War
		cold_war_ongoing = yes
		
		#THIS is a democracy
		is_a_democracy = yes
		
		#THIS must not be a revolutionary
		is_revolutionary = no is_secessionist = no
		
		#South American & African countries only
		OR = { 
			country_is_in_south_america = yes
			country_is_in_africa = yes
		}
		
	}

	possible = {
	
		#Within the top 2 countries in the world
		country_rank = rank_value:super_power
	
		#Country is not at war
		is_at_war = no
		
		#Country is not neutral
		is_not_neutral = yes
		
		#There is someone to create SATO with
		any_country = {
			NOT = { THIS = ROOT }
			OR = { 
				country_is_in_south_america = yes
				country_is_in_africa = yes
			}
			is_revolutionary = no is_secessionist = no
			
			is_a_democracy = yes
			is_sufficiently_independent = yes
			is_not_neutral = yes
			is_at_war = no
			
			relations:root >= relations_threshold:amicable
		}
		
	}
	
	when_taken = {
	
		#THIS forms SATO by being the first member & leader
		add_modifier = { name = sato_leader }
		add_journal_entry = { type = je_sato_system }
		
	}

	ai_chance = {
		value = 0
		
		if = {
			limit = {
				#There is a communist hostile superpower
				any_country = {
					NOT = { THIS = ROOT }
					country_rank = rank_value:super_power
					is_a_communist_government = yes
					relations:root < relations_threshold:neutral
				}
				
			}
			add = 100
		}
		
		if = { limit = { historical_railroading_active = yes } add = 100 }
		
		
	}
	
}
