construct_great_man_made_river_decision = {

	is_shown = {
		
		#Has Required techs
		has_technology_researched = tech_construction_infrastructure_4
		
		#Has a desert state
		any_scope_state = { 
			NOT = { has_variable = great_man_made_river_constructed }
			is_a_desert_state = yes
			trigger_if = {
				limit = { is_split_state = yes }
				is_largest_state_in_region = yes
			}
		}
		
	}

	possible = {
		
		#Has Required techs
		has_technology_researched = tech_manufacturing_5
		
		#Country is not at war
		is_at_war = no
		
		#Country has enough money
		scaled_debt = 0
		
	}
	
	when_taken = {
		
		#Effects
		set_variable = {
			name = great_man_made_river_states
			value = 0
		}
		every_scope_state = {
			limit = {
				NOT = { has_variable = great_man_made_river_constructed }
				is_a_desert_state = yes
				trigger_if = {
					limit = { is_split_state = yes }
					is_largest_state_in_region = yes
				}
			}
			state_region = {
				add_arable_land = 10
			}
			set_variable = great_man_made_river_constructed
			owner ?= {
				change_variable = {
					name = great_man_made_river_states
					add = 1
				}
			}
		}
		custom_tooltip = {
			text = construct_great_man_made_river_expenses.tt
			if = {
				limit = { has_variable = great_man_made_river_states }
				add_modifier = {
					name = construct_great_man_made_river_expenses
					multiplier = var:great_man_made_river_states
					years = 10
				}
			}
		}
		remove_variable = great_man_made_river_states
		
		#Gain support of farmers
		add_loyalists = {
			value = very_large_radicals
			interest_group = ig:ig_rural_folk
		}
		
	}

	ai_chance = {
		value = 1
	}
	
}
