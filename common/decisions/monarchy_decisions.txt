enact_primogeniture_system_decision = {

	is_shown = {
		
		is_a_monarchy = yes
		ruler_is_governor_general = no
		#shares_same_monarch = no
		
		NOT = { has_variable = primogeniture_system }
		
	}

	possible = {
		
		is_sufficiently_independent = yes
		
		#Timer
		custom_tooltip = {
			text = primogeniture_system_timer.tt
			NOT = { has_variable = primogeniture_system_timer }
		}
		
		#Sufficient Government support
		government_has_legislative_super_majority = yes
		
	}
	
	when_taken = {
		
		every_interest_group = {
			limit = {
				has_ideology = ideology:ideology_progressive
			}
			owner ?= {
				add_radicals = {
					value = large_radicals
					interest_group = prev
				}
			}
		}
		
		custom_tooltip = {
			text = enact_primogeniture_system.tt
			set_variable = primogeniture_system
		}
		
		set_variable = {
			name = primogeniture_system_timer
			years = 5
		}
		
	}

	ai_chance = {
		value = 0
	}
	
}

remove_primogeniture_system_decision = {

	is_shown = {
		
		is_a_monarchy = yes
		ruler_is_governor_general = no
		#shares_same_monarch = no
		
		has_variable = primogeniture_system
		
	}

	possible = {
		
		is_sufficiently_independent = yes
		
		#Timer
		custom_tooltip = {
			text = primogeniture_system_timer.tt
			NOT = { has_variable = primogeniture_system_timer }
		}
		
		#Sufficient Government support
		government_has_legislative_super_majority = yes
		
	}
	
	when_taken = {
		
		every_interest_group = {
			limit = {
				has_ideology = ideology:ideology_traditionalist
			}
			owner ?= {
				add_radicals = {
					value = large_radicals
					interest_group = prev
				}
			}
		}
		every_interest_group = {
			limit = {
				has_ideology = ideology:ideology_conservative
			}
			owner ?= {
				add_radicals = {
					value = medium_radicals
					interest_group = prev
				}
			}
		}
		
		custom_tooltip = {
			text = remove_primogeniture_system.tt
			remove_variable = primogeniture_system
		}
		
		set_variable = {
			name = primogeniture_system_timer
			years = 5
		}
		
	}

	ai_chance = {
		value = 0
	}
	
}
