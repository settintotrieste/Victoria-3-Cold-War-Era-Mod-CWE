form_afto_decision = {

	is_shown = {
		
		#Date of establishment
		has_technology_researched = tech_geopolitics_4
		
		#ADTO does not already exist
		NOR = { has_modifier = afto_member has_modifier = afto_leader }
		NOT = { any_country = { has_modifier = afto_leader } }
		
		#THIS is a democracy
		is_a_democracy = yes
		
		#THIS must not be a revolutionary
		is_revolutionary = no is_secessionist = no
		
		#South American & African countries only
		country_is_in_africa = yes
		
	}

	possible = {
	
		#Within the top 2 countries in the world
		country_rank >= rank_value:great_power
		
		is_sufficiently_independent = yes
	
		#Country is not at war
		is_at_war = no
		
		#Country is not neutral
		is_not_neutral = yes
		
		#There is someone to create ADTO with
		any_country = {
			NOT = { THIS = ROOT }
			country_is_in_africa = yes
			is_revolutionary = no is_secessionist = no
			
			is_a_democracy = yes
			is_sufficiently_independent = yes
			is_not_neutral = yes
			is_at_war = no
			
			relations:root >= relations_threshold:amicable
		}
		
	}
	
	when_taken = {
	
		#THIS forms ADTO by being the first member & leader
		add_modifier = { name = afto_leader }
		add_journal_entry = { type = je_afto_system }
		
	}

	ai_chance = {
		value = 0
		
		if = {
			limit = {
				#There is a hostile superpower
				any_country = {
					NOT = { THIS = ROOT }
					country_rank = rank_value:super_power
					relations:root < relations_threshold:neutral
				}
				
			}
			add = 100
		}
		
		if = { limit = { historical_railroading_active = yes } add = 100 }
		
		
	}
	
}
