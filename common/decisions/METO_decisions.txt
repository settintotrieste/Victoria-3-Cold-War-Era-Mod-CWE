form_meto_decision = {

	is_shown = {
		
		#Date of establishment
		has_technology_researched = tech_geopolitics_3
		
		#SEATO does not already exist
		NOR = { has_modifier = meto_member has_modifier = meto_leader }
		NOT = { any_country = { has_modifier = meto_leader } }
		
		#THIS is a democracy
		is_a_democracy = yes
		
		#THIS must not be a revolutionary
		is_revolutionary = no is_secessionist = no
		
		#South American & African countries only
		country_is_in_middle_east = yes
		
	}

	possible = {
	
		#Within the top 2 countries in the world
		country_rank >= rank_value:great_power
		
		is_sufficiently_independent = yes
	
		#Country is not at war
		is_at_war = no
		
		#Country is not neutral
		is_not_neutral = yes
		
		#There is someone to create SEATO with
		any_country = {
			NOT = { THIS = ROOT }
			country_is_in_middle_east = yes
			is_revolutionary = no is_secessionist = no
			
			is_a_democracy = yes
			is_sufficiently_independent = yes
			is_not_neutral = yes
			is_at_war = no
			
			relations:root >= relations_threshold:amicable
		}
		
	}
	
	when_taken = {
	
		#THIS forms SEATO by being the first member & leader
		add_modifier = { name = meto_leader }
		add_journal_entry = { type = je_meto_system }
		
	}

	ai_chance = {
		value = 0
		
		if = {
			limit = {
				#There is a hostile superpower
				any_country = {
					NOT = { THIS = ROOT }
					country_rank = rank_value:super_power
					relations:root < relations_threshold:neutral
				}
				
			}
			add = 100
		}
		
		if = { limit = { historical_railroading_active = yes } add = 100 }
		
		
	}
	
}
