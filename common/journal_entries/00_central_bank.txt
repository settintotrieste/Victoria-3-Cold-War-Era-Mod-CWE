je_central_bank = {
	icon = "gfx/interface/icons/invention_icons/central_banking.dds"

	group = je_group_political_organs

	#scripted_button = je_central_bank_give_independence_button
	#scripted_button = je_central_bank_remove_independence_button
	
	#scripted_button = je_central_bank_quantitative_easing_button
	
	scripted_button = je_central_bank_increase_CBIR_button
	scripted_button = je_central_bank_decrease_CBIR_button
	
	scripted_button = je_central_bank_increase_monetary_policy_button
	scripted_button = je_central_bank_decrease_monetary_policy_button
	
	scripted_button = je_central_bank_revalue_currency_system_button
	scripted_button = je_central_bank_devalue_currency_system_button

	possible = {
		
		#Not part of a Currency Union
		custom_tooltip = {
			text = je_bretton_woods_system_member.4.tt
			NOT = {
				any_scope_diplomatic_pact = {
					is_diplomatic_action_type = currency_union
					second_country = ROOT
				}
			}
		}
		
	}

	immediate = {
		
		#Calculate Core interest Rate
		determine_core_interest_rate = yes
		
		#Default is medium CBIR
		if = { limit = { has_modifier = very_high_cb_interest_rate } remove_modifier = very_high_cb_interest_rate }
		if = { limit = { has_modifier = high_cb_interest_rate } remove_modifier = high_cb_interest_rate }
		if = { limit = { NOT = { has_modifier = medium_cb_interest_rate } } add_modifier = { name = medium_cb_interest_rate } }
		if = { limit = { has_modifier = low_cb_interest_rate } remove_modifier = low_cb_interest_rate }
		if = { limit = { has_modifier = very_low_cb_interest_rate } remove_modifier = very_low_cb_interest_rate }
		
		#Default is gold standard
		if = {
			limit = {
				NOR = {
					has_modifier = law_gold_standard
					has_modifier = law_semi_gold_standard
					has_modifier = law_fiat_standard
				}
			}
			if = {
				limit = { NOT = { has_technology_researched = tech_economy_3 } }
				activate_law_law_gold_standard = yes
			}
			else_if = {
				limit = { NOT = { has_technology_researched = tech_economy_4 } }
				activate_law_law_semi_gold_standard = yes
			}
			else = {
				activate_law_law_fiat_standard  = yes
			}
		}
		
	}
	
	invalid = {
		
		#THIS is part of a Currency Union
		custom_tooltip = {
			text = je_central_bank_system_1.tt
			any_scope_diplomatic_pact = {
				is_diplomatic_action_type = currency_union
				second_country = ROOT
			}
		}
		
	}
	
	on_invalid = {
		
		hidden_effect = {
			remove_variable = core_interest_rate_raw
			remove_variable = core_interest_rate_total_factors
			remove_variable = core_interest_rate
			
			remove_variable = very_high_core_inflation
			remove_variable = high_core_inflation
			remove_variable = moderate_core_inflation
			remove_variable = low_core_inflation
			remove_variable = very_low_core_inflation
			
			remove_modifier = very_high_cb_interest_rate
			remove_modifier = high_cb_interest_rate
			remove_modifier = medium_cb_interest_rate
			remove_modifier = low_cb_interest_rate
			remove_modifier = very_low_cb_interest_rate
		}
		
	}

	on_monthly_pulse = {
		
		effect = {
			
			determine_core_interest_rate = yes
			
			random_list = {
				1 = { set_central_bank_interest_rate = yes }
				5 = { }
			}
			
			#set_domestic_reaction_to_interest_rate = yes
			
			if = {
				limit = {
					NOR = {
						has_modifier = very_high_cb_interest_rate
						has_modifier = high_cb_interest_rate
						has_modifier = medium_cb_interest_rate
						has_modifier = low_cb_interest_rate
						has_modifier = very_low_cb_interest_rate
					}
				}
				add_modifier = { name = medium_cb_interest_rate }
			}
			
			#Set appropriate minting level
			if = {
				limit = {
					OR = {
						has_law_or_variant = law_type:law_independent_central_bank
						is_player = no
					}
					NOR = {
						AND = { 
							has_variable = very_low_core_inflation
							NOT = { has_modifier = law_very_expansionary_monetary_policy }
						}
						AND = { 
							has_variable = low_core_inflation
							NOT = { has_modifier = law_expansionary_monetary_policy }
						}
						AND = { 
							has_variable = moderate_core_inflation
							NOT = { has_modifier = law_neutral_monetary_policy }
						}
						AND = { 
							has_variable = high_core_inflation
							NOT = { has_modifier = law_deflationary_monetary_policy }
						}
						AND = { 
							has_variable = very_high_core_inflation
							NOT = { has_modifier = law_very_deflationary_monetary_policy }
						}
					}
				}
				if = {
					limit = { has_variable = very_low_core_inflation }
					activate_law_law_very_expansionary_monetary_policy = yes
				}
				else_if = {
					limit = { has_variable = low_core_inflation }
					activate_law_law_expansionary_monetary_policy = yes
				}
				else_if = {
					limit = { has_variable = moderate_core_inflation }
					activate_law_law_neutral_monetary_policy = yes
				}
				else_if = {
					limit = { has_variable = high_core_inflation }
					activate_law_law_deflationary_monetary_policy = yes
				}
				else_if = {
					limit = { has_variable = very_high_core_inflation }
					activate_law_law_very_deflationary_monetary_policy = yes
				}
				else = {
					activate_law_law_neutral_monetary_policy = yes
				}
			}
			
		}
		
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = law_very_expansionary_monetary_policy_status
				trigger = {
					has_modifier = law_very_expansionary_monetary_policy
				}
			}
			triggered_desc = {
				desc = law_expansionary_monetary_policy_status
				trigger = {
					has_modifier = law_expansionary_monetary_policy
				}
			}
			triggered_desc = {
				desc = law_neutral_monetary_policy_status
				trigger = {
					has_modifier = law_neutral_monetary_policy
				}
			}
			triggered_desc = {
				desc = law_deflationary_monetary_policy_status
				trigger = {
					 has_modifier = law_deflationary_monetary_policy
				}
			}
			triggered_desc = {
				desc = law_very_deflationary_monetary_policy_status
				trigger = {
					has_modifier = law_very_deflationary_monetary_policy
				}
			}
		}
	}

	weight = 3

	should_be_pinned_by_default_uninvolved_or_context = no

	transferable = no

	progressbar = no
}

		#Determine foreign central bank

				trigger = {
