je_renovate_capital_system = {
	icon = "gfx/interface/icons/invention_icons/steel_frame_buildings.dds"

	group = je_group_technology
	
	is_shown_when_inactive = {
		
		
	}
	
	possible = {
		
		#Added by decision
		
	}

	immediate = {
		
		#Construction time
		set_variable = { 
			name = renovate_capital_progress
			value = 0 
		}
		
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_renovate_capital_system_completed_desc
		color = "gold_positive"
		trigger = {
			#Capital renovated
			scope:journal_entry = {
				is_goal_complete = yes
			}
		}
		effect = {
			capital ?= {
				add_modifier = { name = grand_capital }
			}
		}
	}
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_renovate_capital_system_failed_desc
		color = "blue_negative"
		trigger = {
			OR = {
				in_default = yes
				has_revolution = yes
				is_not_sufficiently_independent = yes
				capital ?= { is_split_state = yes }
			}
		}
		effect = {
			#Clear variables
			remove_variable = renovate_capital_progress
			#Domestic reaction
			add_radicals = {
				value = large_radicals
				strata = upper
			}
			add_radicals = {
				value = medium_radicals
				strata = middle
			}
		}
	}
	

	
	# zero or more static modifiers that will be applied to the Journal Entry when it activates, where they will propagate to the country
	modifiers_while_active = {
		renovate_capital_dev_cost
	}

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes

	transferable = no

	progressbar = yes
	
	current_value = {
		value = root.var:renovate_capital_progress
	}

	goal_add_value = {
		add = 120
	}
	
	on_monthly_pulse = {
		
		#Add progress
		effect = {
		
			if = {
				limit = { in_default = no }
				if = {
					limit = { government_wage_level = very_high }
					change_variable = {
						name = renovate_capital_progress
						add = 5
					}
				}
				else_if = {
					limit = { government_wage_level = high }
					change_variable = {
						name = renovate_capital_progress
						add = 4
					}
				}
				else_if = {
					limit = { government_wage_level = medium }
					change_variable = {
						name = renovate_capital_progress
						add = 3
					}
				}
				else_if = {
					limit = { government_wage_level = low }
					change_variable = {
						name = renovate_capital_progress
						add = 2
					}
				}
				else = {
					change_variable = {
						name = renovate_capital_progress
						add = 1
					}
				}
				clamp_variable = {
					name = renovate_capital_progress
					max = 120
					min = 0
				}
			}
			
		}
		
	}
	
}
