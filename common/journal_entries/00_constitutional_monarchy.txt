je_constitutional_monarchy = {
	icon = "gfx/interface/icons/law_icons/monarchy.dds"

	group = je_group_political_organs

	scripted_button = je_enact_symbolic_monarchy_button
	scripted_button = je_enact_elective_monarchy_button
	scripted_button = je_issue_royal_decree_button
	scripted_button = je_enact_no_nobility_button

	is_shown_when_inactive = {
		
		has_law_or_variant = law_type:law_HOS_monarch
		is_a_constitutional_monarchy = yes
		
		ruler_is_governor_general = no
		
		NOT = { has_variable = block_constitutional_monarchy_generation }
		
		#Not has regency
		NOR = {
			has_journal_entry = je_regency_system
			any_scope_character = { has_role = character_role_regent }
		}
		
	}

	possible = {
		
		trigger_if = {
			limit = { 
				OR = { 
					is_a_colonial_government = yes
					shares_same_monarch = yes
				}
				is_subject = yes
			}
			overlord ?= {
				any_scope_character = {
					OR = {
						has_role = character_role_monarch
						has_role = character_role_regent
					}
				}
			}
		}
		
	}

	immediate = {
		
		if = {
			limit = { 
				OR = { 
					is_a_colonial_government = yes
					shares_same_monarch = yes
				}
				is_subject = yes
			}
			if = {
				limit = { 
					overlord ?= {
						any_scope_character = {
							OR = {
								has_role = character_role_monarch
								has_role = character_role_regent
							}
						}
					}
				}
				overlord ?= {
					if = {
						limit = {
							any_scope_character = {
								has_role = character_role_monarch
							}
						}
						random_scope_character = {
							limit = { 
								has_role = character_role_monarch
							}
							save_scope_as = constitutional_monarch_scope
						}
					}
					else_if = {
						limit = {
							any_scope_character = {
								has_role = character_role_regent
							}
						}
						random_scope_character = {
							limit = { 
								has_role = character_role_regent
							}
							save_scope_as = constitutional_monarch_scope
						}
					}
				}
			}
			else_if = {
				limit = { 
					top_overlord ?= {
						any_scope_character = {
							OR = {
								has_role = character_role_monarch
								has_role = character_role_regent
							}
						}
					}
				}
				top_overlord ?= {
					if = {
						limit = {
							any_scope_character = {
								has_role = character_role_monarch
							}
						}
						random_scope_character = {
							limit = { 
								has_role = character_role_monarch
							}
							save_scope_as = constitutional_monarch_scope
						}
					}
					else_if = {
						limit = {
							any_scope_character = {
								has_role = character_role_regent
							}
						}
						random_scope_character = {
							limit = { 
								has_role = character_role_regent
							}
							save_scope_as = constitutional_monarch_scope
						}
					}
				}
			}
			else = {
				#No existing monarch
				if = {
					limit = {
						NOT = {
							any_scope_character = {
								has_role = character_role_monarch
							}
						}
					}
					generate_monarch_for_country = yes
				}
				#Mark Monarch
				random_scope_character = {
					limit = { 
						has_role = character_role_monarch
					}
					save_scope_as = constitutional_monarch_scope
				}
			}
			
			#If there is no Viceroy
			if = {
				limit = {
					exists = scope:constitutional_monarch_scope
					OR = {
						is_a_colonial_government = yes
						shares_same_monarch = yes
					}
					is_subject = yes
					NOT = {
						any_scope_character = {
							OR = { 
								has_role = character_role_monarch
								has_role = character_role_viceroy
							}
						}
					}
				}
				generate_viceroy_for_country = yes
			}
			
		}
		else_if = {
			limit = { 
				any_scope_character = {
					has_role = character_role_monarch
				}
			}
			#Mark Monarch
			random_scope_character = {
				limit = { 
					has_role = character_role_monarch
				}
				save_scope_as = constitutional_monarch_scope
			}
		}
		else = {
		
			#No existing monarch
			if = {
				limit = {
					NOT = {
						any_scope_character = {
							has_role = character_role_monarch
						}
					}
				}
				if = {
					limit = { 
						NOT = {
							active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy }
						}
					}
					generate_monarch_for_country = yes
				}
				else_if = {
					limit = { 
						active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy }
						any_scope_character = {
							has_role = character_role_monarch_elective
							NOT = { has_variable = elected_monarch_term_cooldown }
						}
					}
					random_scope_character = {
						limit = { 
							has_role = character_role_monarch_elective
							NOT = { has_variable = elected_monarch_term_cooldown }
						}
						set_character_as_monarch = yes
					}
				}
				else_if = {
					limit = { 
						active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy }
					}
					generate_monarch_for_country = yes
				}
			}
			
			#Mark Monarch
			random_scope_character = {
				limit = { 
					has_role = character_role_monarch
				}
				save_scope_as = constitutional_monarch_scope
			}
			
			#There is local monarchy when monarchy should be foreign
			if = {
				limit = {
					OR = { 
						ruler_is_governor_general = yes
						shares_same_monarch = yes
					}
					any_scope_character = {
						OR = {
							has_role = character_role_monarch 
							has_role = character_role_monarch_elective
							has_role = character_role_regent
							has_role = character_role_prince
						}
					}
				}
				every_scope_character = {
					limit = { has_role = character_role_monarch }
					remove_character_role = character_role_monarch
				}
				every_scope_character = {
					limit = { has_role = character_role_monarch_elective }
					remove_character_role = character_role_monarch_elective
				}
				every_scope_character = {
					limit = { has_role = character_role_regent }
					remove_character_role = character_role_regent
				}
				every_scope_character = {
					limit = { has_role = character_role_prince }
					remove_character_role = character_role_prince
				}
			}
			
		}
		
		#Clear variables
		remove_variable = elective_monarchy_system
		remove_variable = elective_monarchy_system_timer
		remove_variable = female_monarch_var
		
		#Generate elected monarchs
		if = {
			limit = {
				active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy }
			}
			set_variable = {
				name = elective_monarchy_system_timer
				years = 5
			}
			scope:constitutional_monarch_scope ?= {
				set_variable = {
					name = elected_monarch_term
					years = 5
				}
			}
			generate_elective_monarchs_for_country = yes
		}
		
		if = {
			limit = { 
				active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy }
			}
			save_scope_as = constitutional_monarch_country_scope
			post_notification = new_elective_monarch
		}
		else_if = {
			limit = {
				scope:constitutional_monarch_scope ?= {
					is_adult = yes
					NAND = {
						owner ?= { has_variable = primogeniture_system }
						is_female = yes
					}
				}
			}
			save_scope_as = constitutional_monarch_country_scope
			post_notification = new_monarch
		}
		
	}

	invalid = {
		
		OR = {
		
			#No longer a constitutional monarchy
			NAND = {
				has_law_or_variant = law_type:law_HOS_monarch
				is_a_constitutional_monarchy = yes
			}
			
			#Ruler is a governor general
			ruler_is_governor_general = yes
			
			#Monarch no longer exists
			NOT = { exists = scope:constitutional_monarch_scope }
			
			#Monarch no longer a monarch
			scope:constitutional_monarch_scope ?= {
				NOR = {
					has_role = character_role_monarch
					has_role = character_role_regent
				}
			}
			
			#Foreign Monarch is no longer monarch of THIS
			AND = {
				scope:constitutional_monarch_scope ?= {
					NOT = { owner ?= ROOT }
				}
				OR = {
					is_subject = no
					shares_same_monarch = no
				}
			}
			
			#Regency
			scope:constitutional_monarch_scope ?= {
				owner ?= ROOT
				has_role = character_role_monarch
				OR = {
					is_adult = no
					AND = {
						is_female = yes
						owner ?= { has_variable = primogeniture_system }
					}
					AND = {
						has_role = character_role_regent
						has_role = character_role_monarch
					}
				}
			}
			
		}
		
	}
	
	on_invalid = {
		
		#Remove all viceroys
		hidden_effect = {
			if = {
				limit = { 
					is_a_foreign_regime_monarchy = no
					NAND = {
						is_subject = yes
						shares_same_monarch = yes
					}
				}
				every_scope_character = {
					limit = { 
						has_role = character_role_viceroy
					}
					retire_character = yes
				}
			}
		}
		
		#Clear variables
		remove_variable = elective_monarchy_system
		remove_variable = elective_monarchy_system_timer
		remove_variable = female_monarch_var
		
		#Clear Elective Monarchs
		if = {
			limit = { 
				NOT = { active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy } }
			}
			every_scope_character = {
				limit = { has_role = character_role_monarch_elective }
				remove_character_role = character_role_monarch_elective
				set_character_as_royal = yes
			}
		}
		
		#Clear monarchy
		if = {
			limit = {
				OR = {
					NOT = { has_law_or_variant = law_type:law_HOS_monarch }
					ruler_is_governor_general = yes
					shares_same_monarch = yes
				}
				any_scope_character = {
					OR = {
						has_role = character_role_monarch 
						has_role = character_role_monarch_elective
						has_role = character_role_regent
						has_role = character_role_prince
					}
				}
			}
			every_scope_character = {
				limit = { has_role = character_role_monarch }
				remove_character_role = character_role_monarch
			}
			every_scope_character = {
				limit = { has_role = character_role_monarch_elective }
				remove_character_role = character_role_monarch_elective
			}
			every_scope_character = {
				limit = { has_role = character_role_regent }
				remove_character_role = character_role_regent
			}
			every_scope_character = {
				limit = { has_role = character_role_prince }
				remove_character_role = character_role_prince
			}
		}
		
		#Create Regency if applicable
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				scope:constitutional_monarch_scope ?= {
					owner ?= ROOT
					has_role = character_role_monarch
					OR = {
						is_adult = no
						AND = {
							owner ?= { has_variable = primogeniture_system }
							is_female = yes
						}
					}
				}
				NOT = {
					any_scope_character = {
						has_role = character_role_regent
					}
				}
			}
			if = {
				limit = {
					any_scope_character = {
						has_role = character_role_prince
						NOT = { has_variable = prince_consort_character }
						NAND = {
							owner ?= { has_variable = primogeniture_system }
							is_female = yes
						}
					}
				}
				random_scope_character = {
					limit = { 
						has_role = character_role_prince
						NOT = { has_variable = prince_consort_character }
						NAND = {
							owner ?= { has_variable = primogeniture_system }
							is_female = yes
						}
					}
					set_character_as_monarch = yes
				}
			}
			else = {
				if = {
					limit = {
						any_scope_character = {
							has_role_of_type = magnate
							is_noble = yes
							is_adult = yes
						}
					}
					ordered_scope_character = {
						limit = {
							has_role_of_type = magnate
							is_noble = yes
							is_adult = yes
						}
						order_by = popularity
						position = 0
						set_character_as_regent = yes
					}
				}
				else_if = {
					limit = {
						any_scope_character = {
							OR = { 
								has_role_of_type = general
								has_role_of_type = admiral
							}
						}
					}
					ordered_scope_character = {
						limit = {
							OR = { 
								has_role_of_type = general
								has_role_of_type = admiral
							}
						}
						order_by = commander_rank
						position = 0
						set_character_as_regent = yes
					}
				}
				else = {
					create_character = {
						role = character_role_regent noble = yes
						on_created = { set_character_as_regent = yes }
					}
				}
			}
		}
		
	}
	
	on_complete = {
		
		#Mark Monarch under Regency
		if = {
			limit = {
				scope:constitutional_monarch_scope ?= {
					has_role = character_role_monarch
					owner ?= {
						is_a_regency_government = yes
					}
				}
			}
			scope:constitutional_monarch_scope ?= {
				owner ?= {
					random_scope_character = {
						limit = { 
							has_role = character_role_regent
						}
						save_scope_as = regent_character_label
						custom_tooltip = {
							text = regent_character_label.tt
						}
					}
				}
			}
		}
		
		#Mark elective monarchs
		if = {
			limit = {
				active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy }
			}
			every_scope_character = {
				limit = { has_role = character_role_monarch_elective }
				save_scope_as = monarch_elective_character_label
				custom_tooltip = {
					text = monarch_elective_character_label.tt
				}
			}
		}
		
		#Mark viceroy
		random_scope_character = {
			limit = { 
				has_role = character_role_viceroy
			}
			save_scope_as = viceroy_character_label
			custom_tooltip = {
				text = governor_general_character_label.tt
			}
		}
	
	}
	
	on_monthly_pulse = {
	
		effect = {
			
			#If monarch is ruler OR an IG leader
			if = {
				limit = {
					ruler ?= {
						OR = {
							has_role = character_role_monarch
							has_role = character_role_monarch_elective
							has_role = character_role_viceroy
						}
					}
				}
				trigger_event = { id = reserve_powers_system.19 }
			}
			every_scope_character = {
				limit = {
					OR = {
						has_role = character_role_monarch
						has_role = character_role_monarch_elective
						has_role = character_role_viceroy
					}
					has_role_of_type = politician
				}
				remove_character_role = politician
			}
			
			#If there is no Viceroy
			if = {
				limit = {
					exists = scope:constitutional_monarch_scope
					OR = {
						is_a_colonial_government = yes
						shares_same_monarch = yes
					}
					is_subject = yes
					NOT = {
						any_scope_character = {
							OR = { 
								has_role = character_role_monarch
								has_role = character_role_viceroy
							}
						}
					}
				}
				generate_viceroy_for_country = yes
			}
			
			#Restore any exiled monarch
			if = {
				limit = { 
					exists = scope:constitutional_monarch_scope
					scope:constitutional_monarch_scope ?= { is_in_exile_pool = yes }
				}
				scope:constitutional_monarch_scope ?= {
					transfer_character = ROOT
					set_character_as_monarch = yes
				}
			}
			
			#Remove Viceroy
			if = {
				limit = { 
					exists = scope:constitutional_monarch_scope
					any_scope_character = {
						has_role = character_role_monarch
					}
					any_scope_character = {
						has_role = character_role_viceroy
					}
				}
				random_scope_character = {
					limit = { 
						has_role = character_role_viceroy
					}
					retire_character_scripted_effect = yes
				}
			}
			
		}
		
	}
	
	on_yearly_pulse = {
	
		effect = {
			
			#Generate elected monarchs
			if = {
				limit = { 
					active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy }
				}
				generate_elective_monarchs_for_country = yes
				if = {
					limit = {
						scope:constitutional_monarch_scope ?= { NOT = { has_variable = elected_monarch_term } }
					}
					scope:constitutional_monarch_scope ?= {
						set_variable = { 
							name = elected_monarch_term_cooldown
							years = 2
						}
						replace_character_roles = { add = { character_role_monarch_elective } remove_all = yes }
					}
				}
			}
			
		}
		
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = monarch_ig_label
				trigger = {
					scope:constitutional_monarch_scope ?= {
						OR = { 
							has_role = character_role_monarch
							has_role = character_role_regent
						}
					}
				}
			}
			triggered_desc = {
				desc = monarch_vacated
				trigger = {
					always = yes
				}
			}
		}
	}
	
	# zero or more static modifiers that will be applied to the Journal Entry when it activates, where they will propagate to the country
	modifiers_while_active = {
		#monarchy_system
	}

	weight = 50

	should_be_pinned_by_default_uninvolved_or_context = no

	transferable = no

	progressbar = no
}
