je_regency_system = {
	icon = "gfx/interface/icons/law_icons/monarchy.dds"

	group = je_group_political_organs
	
	scripted_button = je_dismiss_regent_button 

	is_shown_when_inactive = {
		has_law_or_variant = law_type:law_HOS_monarch
		ruler_is_governor_general = no
		shares_same_monarch = no
		is_a_regency_government = yes
		NOT = { active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy } }
		any_scope_character = { has_role = character_role_regent }
	}

	possible = {
		
	}

	immediate = {
		random_scope_character = {
			limit = { has_role = character_role_regent }
			save_scope_as = regent_character
		}
		random_scope_character = {
			limit = { has_role = character_role_monarch }
			save_scope_as = monarch_character
		}
	}
	
	invalid = {
		OR = {
			NAND = {
				has_law_or_variant = law_type:law_HOS_monarch
				ruler_is_governor_general = no
				shares_same_monarch = no
				is_a_regency_government = yes
			}
			scope:regent_character ?= { NOT = { has_role = character_role_regent } }
			NOT = { exists = scope:regent_character }
			scope:monarch_character ?= { NOT = { has_role = character_role_monarch } }
			active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy }
		}
	}
	
	on_invalid = {
		
		#Coronation Day
		hidden_effect = {
			if = {
				limit = {
					has_law_or_variant = law_type:law_HOS_monarch
					ruler_is_governor_general = no
					shares_same_monarch = no
					any_scope_character = {
						NOT = { has_role = character_role_regent }
						has_role = character_role_monarch
						is_adult = yes
						NAND = {
							owner ?= { has_variable = primogeniture_system }
							is_female = yes
						}
					}
				}
				trigger_event = { id = regency.3 popup = yes }
			}
		}
		
		#Clean up
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				ruler_is_governor_general = no
				shares_same_monarch = no
				scope:monarch_character ?= { has_role = character_role_monarch }
			}
			scope:regent_character ?= {
				remove_character_role = character_role_regent
				clean_up_regent_character = yes
			}
		}
		
		#Detect monarch
		random_scope_character = {
			limit = { has_role = character_role_monarch }
			save_scope_as = constitutional_monarch_scope
		}
		
		#Create Regency if applicable
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				ruler_is_governor_general = no
				shares_same_monarch = no
				NOT = { active_law:lawgroup_governance_principles ?= { has_amendment = amendment_type:amendment_elective_monarchy } }
				any_scope_character = {
					owner ?= ROOT
					has_role = character_role_monarch
					is_eligible_to_become_monarch = no
				}
			}
			trigger_event = { id = regency.2 popup = yes }
		}
		
	}
	
	complete = {
		trigger_if = {
			limit = { exists = scope:monarch_character }
			scope:monarch_character ?= {
				OR = { 
					has_role = character_role_monarch
					AND = { 
						has_role = character_role_prince
						owner ?= { has_law_or_variant = law_type:law_HOG_prime_minister }
					}
				}
				NOT = { has_role = character_role_regent }
				is_eligible_to_become_monarch = yes
			}
		}
		trigger_else = {
			any_scope_character = {
				OR = { 
					has_role = character_role_monarch
					AND = { 
						has_role = character_role_prince
						owner ?= { has_law_or_variant = law_type:law_HOG_prime_minister }
					}
				}
				NOT = { has_role = character_role_regent }
				is_eligible_to_become_monarch = yes
			}
		}
	}
	
	on_complete = {
		
		hidden_effect = {
			
			#Coronation Day
			if = {
				limit = {
					has_law_or_variant = law_type:law_HOS_monarch
					ruler_is_governor_general = no
					shares_same_monarch = no
					any_scope_character = {
						OR = { 
							has_role = character_role_monarch
							AND = { 
								has_role = character_role_prince
								owner ?= { has_law_or_variant = law_type:law_HOG_prime_minister }
							}
						}
						NOT = { has_role = character_role_regent }
						is_eligible_to_become_monarch = yes
					}
				}
				trigger_event = { id = regency.3 popup = yes }
			}
			
			#Clean up
			scope:regent_character ?= {
				remove_character_role = character_role_regent
				clean_up_regent_character = yes
			}
			
		}
		
		#Mark monarchs
		every_scope_character = {
			limit = { has_role = character_role_monarch }
			save_scope_as = monarch_character_label
			custom_tooltip = {
				text = character_role_monarch.tt
			}
		}
		
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = regency_IG
				trigger = {
					scope:regent_character ?= { has_role = character_role_regent }
				}
			}
			triggered_desc = {
				desc = regency_vacant
				trigger = {
					
				}
			}
		}
	}

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = no

	transferable = no

	progressbar = no
}
