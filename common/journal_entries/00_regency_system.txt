je_regency_system = {
	icon = "gfx/interface/icons/law_icons/monarchy.dds"

	group = je_group_political_organs

	is_shown_when_inactive = {
		has_law_or_variant = law_type:law_HOS_monarch
		is_a_colonial_government = no
		shares_same_monarch = no
		OR = {
			is_a_regency_government = yes
			any_scope_character = { has_role = character_role_regent }
		}
	}

	possible = {
		
	}

	immediate = {
		if = {
			limit = {
				is_a_regency_government = yes
				NOT = {
					any_scope_character = {
						has_role = character_role_regent
					}
				}
			}
			ruler ?= { set_character_as_regent = yes }
		}
		random_scope_character = {
			limit = { has_role = character_role_regent }
			save_scope_as = regent_character
		}
		random_scope_character = {
			limit = { has_role = character_role_monarch }
			save_scope_as = monarch_character
		}
	}
	
	invalid = {
		OR = {
			NOT = { has_law_or_variant = law_type:law_HOS_monarch }
			scope:regent_character ?= { NOT = { has_role = character_role_regent } }
			NOT = { exists = scope:regent_character }
		}
	}
	
	on_invalid = {
		
		#Clean up
		every_scope_character = {
			limit = { has_role = character_role_regent }
			remove_character_role = character_role_regent
		}
		every_scope_character = {
			limit = { 
				has_modifier = regent_character
			}
			remove_modifier = regent_character
		}
		remove_regency_modifier = yes
		
		#Detect monarch
		random_scope_character = {
			limit = { has_role = character_role_monarch }
			save_scope_as = constitutional_monarch_scope
		}
		
		#Create Regency if applicable
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				is_a_regency_government = yes
				any_scope_character = {
					owner ?= ROOT
					is_ruler_of_own_country = yes
					has_role = character_role_monarch
					OR = {
						is_adult = no
						AND = {
							owner ?= { has_variable = primogeniture_system }
							is_female = yes
						}
					}
				}
			}
			trigger_event = { id = regency.2 }
		}
		
	}
	
	complete = {
		trigger_if = {
			limit = { exists = scope:monarch_character }
			scope:monarch_character ?= {
				OR = {
					has_role = character_role_monarch
					AND = {
						has_role = character_role_prince
						custom_tooltip = {
							text = prince_consort_character.tt
							NOT = { has_variable = prince_consort_character }
						}
						NAND = {
							custom_tooltip = {
								text = primogeniture_system.tt
								owner ?= { has_variable = primogeniture_system }
							}
							is_female = yes
						}
					}
				}
				is_adult = yes
				NAND = {
					custom_tooltip = {
						text = primogeniture_system.tt
						owner ?= { has_variable = primogeniture_system }
					}
					is_female = yes
				}
			}
		}
		trigger_else = {
			any_scope_character ?= {
				OR = {
					has_role = character_role_monarch
					AND = {
						has_role = character_role_prince
						NOT = {  has_role = character_role_regent }
						custom_tooltip = {
							text = prince_consort_character.tt
							NOT = { has_variable = prince_consort_character }
						}
						NAND = {
							custom_tooltip = {
								text = primogeniture_system.tt
								owner ?= { has_variable = primogeniture_system }
							}
							is_female = yes
						}
					}
				}
				is_adult = yes
				NAND = {
					custom_tooltip = {
						text = primogeniture_system.tt
						owner ?= { has_variable = primogeniture_system }
					}
					is_female = yes
				}
			}
		}
	}
	
	on_complete = {
		
		#Clean up
		every_scope_character = {
			limit = { has_role = character_role_regent }
			remove_character_role = character_role_regent
		}
		hidden_effect = {
			every_scope_character = {
				limit = { 
					has_modifier = regent_character
				}
				remove_modifier = regent_character
			}
			remove_regency_modifier = yes
		}
		
		#Mark monarchs
		every_scope_character = {
			limit = { has_role = character_role_monarch }
			save_scope_as = monarch_character_label
			custom_tooltip = {
				text = character_role_monarch.tt
			}
		}
		
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = regency_IG
				trigger = {
					scope:regent_character ?= { has_role = character_role_regent }
				}
			}
			triggered_desc = {
				desc = regency_vacant
				trigger = {
					
				}
			}
		}
	}

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = no

	transferable = no

	progressbar = no
}
