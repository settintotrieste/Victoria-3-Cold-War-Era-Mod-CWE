je_bretton_woods_system_leader = {
	icon = "gfx/interface/icons/goods_icons/gold.dds"

	group = je_group_foreign_affairs

	possible = {
		
		#Has Gold Standard
		has_modifier = law_gold_standard
		
		#Most powerful country with the gold standard
		custom_tooltip = {
			text = je_bretton_woods_system_leader.3.tt
			NOT = {
				any_country = {
					NOT = { THIS = ROOT }
					has_modifier = law_gold_standard
					country_rank >= rank_value:great_power
					this.gdp > root.gdp
				}
			}
		}
		
		#Not part of a Currency Union
		custom_tooltip = {
			text = je_bretton_woods_system_member.4.tt
			NOT = {
				any_scope_diplomatic_pact = {
					is_diplomatic_action_type = currency_union
					second_country = ROOT
				}
			}
		}
		
		#Any country has its currency pegged to a currency which is in turn pegged to a gold standard
		any_country = {
			NOT = { THIS = ROOT }
			has_modifier = law_semi_gold_standard
		}
		
		country_rank >= rank_value:great_power
		
	}

	immediate = {
		
		#THIS is the Gold Standard Leader
		add_modifier = { name = gold_standard_leader }
		set_variable = gold_standard_leader
		
		save_scope_as = bretton_woods_system_leader
		
		#Calculate number of BWS members
		count_bretton_woods_members = yes
		
	}
	
	on_monthly_pulse = {
		
		effect = {
		
			#Calculate number of BWS members
			count_bretton_woods_members = yes
		
		}
		
	}

	fail = {
	
		OR = { 
		
			#Abolished Gold Standard
			NOT = { has_modifier = law_gold_standard }
		
			#There is a more powerful Gold Standard country
			custom_tooltip = {
				text = je_bretton_woods_system_leader.1.tt
				any_country = {
					NOT = { THIS = ROOT }
					has_modifier = law_gold_standard
					country_rank >= rank_value:great_power
					this.gdp > root.gdp
				}
			}
			
			#Not a single country has a semi gold standard
			custom_tooltip = {
				text = je_bretton_woods_system_leader.2.tt
				NOT = { 
					any_country = {
						NOT = { THIS = ROOT }
						has_modifier = law_semi_gold_standard
					}
				}
			}
			
			#THIS is part of a Currency Union
			custom_tooltip = {
				text = je_central_bank_system_1.tt
				any_scope_diplomatic_pact = {
					is_diplomatic_action_type = currency_union
					second_country = ROOT
				}
			}
			
			country_rank < rank_value:great_power
			
		}
		
	}
	
	on_fail = {
		
		#Remove modifier
		remove_modifier = gold_standard_leader
		remove_variable = gold_standard_leader
		
	}

	weight = 50

	should_be_pinned_by_default_uninvolved_or_context = no

	transferable = no

	progressbar = no
}

je_bretton_woods_system_member = {
	icon = "gfx/interface/icons/goods_icons/gold.dds"

	group = je_group_foreign_affairs

	possible = {
		
		#Has a Semi-Gold Standard
		has_modifier = law_semi_gold_standard
		
		#Most powerful country with the gold standard
		custom_tooltip = {
			text = je_bretton_woods_system_member.2.tt
			any_country = {
				NOT = { THIS = ROOT }
				has_modifier = gold_standard_leader
			}
		}
		
		#Not part of a Currency Union
		custom_tooltip = {
			text = je_bretton_woods_system_member.4.tt
			NOT = {
				any_scope_diplomatic_pact = {
					is_diplomatic_action_type = currency_union
					second_country = ROOT
				}
			}
		}
		
	}
	
	immediate = {
		
		random_country = {
			limit = { has_modifier = gold_standard_leader }
			save_scope_as = bretton_woods_system_leader
		}
		
		#Calculate number of BWS members
		count_bretton_woods_members = yes
		
	}

	fail = {
		
		OR = {
		
			#No longer has a Semi-Gold Standard
			NOT = { has_modifier = law_semi_gold_standard }
		
			#There is no longer a bretton woods leader
			custom_tooltip = {
				text = je_bretton_woods_system_member.1.tt
				NOR = {
					scope:bretton_woods_system_leader ?= {
						has_modifier = gold_standard_leader
					}
					exists = scope:bretton_woods_system_leader
				}
			}
			
			#THIS is part of a Currency Union
			custom_tooltip = {
				text = je_central_bank_system_1.tt
				any_scope_diplomatic_pact = {
					is_diplomatic_action_type = currency_union
					second_country = ROOT
				}
			}
			
		}
		
	}
	
	on_monthly_pulse = {
		
		effect = {
			
			### Synchronise Monetary Policy of TARGET with THIS in reverse ###
			if = {
				limit = {
					scope:bretton_woods_system_leader ?= { has_modifier = gold_standard_leader has_modifier = law_very_deflationary_monetary_policy }
					NOT = { has_modifier = law_very_expansionary_monetary_policy }
				}
				activate_law_law_very_expansionary_monetary_policy = yes
			}
			else_if = {
				limit = {
					scope:bretton_woods_system_leader ?= { has_modifier = gold_standard_leader has_modifier = law_deflationary_monetary_policy }
					NOT = { has_modifier = law_expansionary_monetary_policy }
				}
				activate_law_law_expansionary_monetary_policy = yes
			}
			else_if = {
				limit = {
					scope:bretton_woods_system_leader ?= { has_modifier = gold_standard_leader has_modifier = law_neutral_monetary_policy }
					NOT = { has_modifier = law_neutral_monetary_policy }
				}
				activate_law_law_neutral_monetary_policy = yes
			}
			else_if = {
				limit = {
					scope:bretton_woods_system_leader ?= { has_modifier = gold_standard_leader has_modifier = law_expansionary_monetary_policy }
					NOT = { has_modifier = law_deflationary_monetary_policy }
				}
				activate_law_law_deflationary_monetary_policy = yes
			}
			else_if = {
				limit = {
					scope:bretton_woods_system_leader ?= { has_modifier = gold_standard_leader has_modifier = law_very_expansionary_monetary_policy }
					NOT = { has_modifier = law_very_deflationary_monetary_policy }
				}
				activate_law_law_very_deflationary_monetary_policy = yes
			}
			
			#Calculate number of BWS members
			count_bretton_woods_members = yes
			
		}
		
	}

	weight = 50

	should_be_pinned_by_default_uninvolved_or_context = yes

	transferable = no

	progressbar = no
}
