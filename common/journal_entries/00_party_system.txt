je_party_system = {
	icon = "gfx/interface/icons/invention_icons/corporate_charters.dds"

	group = je_group_political_organs
	
	scripted_button = je_party_system_recruit_politician_button
	
	is_shown_when_inactive = {
		
		NOR = { has_law_or_variant = law_type:law_parties_not_allowed has_law_or_variant = law_type:law_single_party_state }
		
	}
	
	possible = {
		
		ruler.interest_group ?= { has_party = yes }
		
	}

	immediate = {
		
		ruler.interest_group ?= {
			party ?= { save_scope_as = ruling_party_scope }
		}
		
		calculate_party_radicalism = yes
		
	}
	
	on_monthly_pulse = {
		
		effect = {
			
			calculate_party_radicalism = yes
			
		}
		
	}

	invalid = {
		
		OR = {
		
			has_law_or_variant = law_type:law_parties_not_allowed
			has_law_or_variant = law_type:law_single_party_state
			
			ruler.interest_group ?= { has_party = no }
			
			NOT = { exists = scope:ruling_party_scope }
			
		}
		
	}
	
	on_invalid = {
		
		#Clear variables
		remove_variable = party_radicalism
		remove_variable = very_reformist_party
		remove_variable = reformist_party
		remove_variable = moderate_party
		remove_variable = radical_party
		remove_variable = very_radical_party
		
		#Remove effects
		hidden_effect = {
			every_scope_character = {
				limit = { has_modifier = party_outcast }
				remove_modifier = party_outcast
			}
		}
		
		#Remove modifiers
		if = {
			limit = { has_modifier = divided_party }
			remove_modifier = divided_party
		}
		
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = very_reformist_party
				trigger = {
					has_variable = very_reformist_party
				}
			}
			triggered_desc = {
				desc = reformist_party
				trigger = {
					has_variable = reformist_party
				}
			}
			triggered_desc = {
				desc = radical_party
				trigger = {
					has_variable = radical_party
				}
			}
			triggered_desc = {
				desc = very_radical_party
				trigger = {
					has_variable = very_radical_party
				}
			}
			triggered_desc = {
				desc = moderate_party
				trigger = {
					
				}
			}
		}
	}

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = no

	transferable = no

	progressbar = no
	
}
