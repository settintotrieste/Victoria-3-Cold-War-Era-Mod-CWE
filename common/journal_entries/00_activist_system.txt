je_activist_system = {
	icon = "gfx/interface/icons/invention_icons/realism.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		
		#Has oppressive laws
		OR = {
			NOT = { has_law_or_variant = law_type:law_democratic_legislature }
			has_law_or_variant = law_type:law_lenient_policing
			has_law_or_variant = law_type:law_slavery_political
			has_law_or_variant = law_type:law_slavery_cultural
			has_law_or_variant = law_type:law_secret_police
			has_law_or_variant = law_type:law_outlawed_dissent
			has_law_or_variant = law_type:law_censorship
			NOR = { has_law_or_variant = law_type:law_multicultural has_law_or_variant = law_type:law_cultural_exclusion }
			NOR = { has_law_or_variant = law_type:law_freedom_of_conscience has_law_or_variant = law_type:law_total_separation }
		}
		
	}

	possible = {
		
		#There are activists
		any_scope_character = {
			has_role_of_type = agitator
			character_is_peaceful = yes
		}
		
	}

	immediate = {
		
		#Generate activists
		every_scope_character = {
			limit = {
				has_role_of_type = agitator
				character_is_peaceful = yes
				NOT = { has_role = character_role_activist }
			}
			add_character_role = character_role_activist
		}
		
	}

	complete = {
	
		#There are no activists
		NOR = {
			NOT = { has_law_or_variant = law_type:law_democratic_legislature }
			has_law_or_variant = law_type:law_lenient_policing
			has_law_or_variant = law_type:law_slavery_political
			has_law_or_variant = law_type:law_slavery_cultural
			has_law_or_variant = law_type:law_secret_police
			has_law_or_variant = law_type:law_outlawed_dissent
			has_law_or_variant = law_type:law_censorship
			NOR = { has_law_or_variant = law_type:law_multicultural has_law_or_variant = law_type:law_cultural_exclusion }
			NOR = { has_law_or_variant = law_type:law_freedom_of_conscience has_law_or_variant = law_type:law_total_separation }
		}
		
	}
	
	on_complete = {
		
		#Mark activists
		every_scope_character = {
			limit = {
				has_role = character_role_activist
			}
			save_scope_as = activist_character
			custom_tooltip = activist_character_label.tt
			hidden_effect = { remove_character_role = character_role_activist }
		}
		
	}
	
	on_yearly_pulse = {
	
		effect = {
			
			#Generate activists
			every_scope_character = {
				limit = {
					has_role_of_type = agitator
					character_is_peaceful = yes
					NOT = { has_role = character_role_activist }
				}
				add_character_role = character_role_activist
			}
			
		}
		
	}

	weight = 20

	should_be_pinned_by_default_uninvolved_or_context = yes

	transferable = no

	progressbar = no
	
}
