je_federal_administration = {
	icon = "gfx/interface/icons/lens_toolbar_icons/da_appoint_subject_ruler.dds"

	group = je_group_political_organs

	is_shown_when_inactive = {
		
	}

	possible = {
		
		has_law_or_variant = law_type:law_federal_state
		is_a_democracy = yes
		
		#is_at_war = no
		
		hidden_trigger = { NOT = { has_variable = block_constitutional_monarchy_generation } }
		
	}

	immediate = {
		
		#Generate representatives
		if = {
			limit = { NOT = { has_variable = block_representative_generation } }
			hidden_effect = { generate_representatives_for_country_federal = yes }
		}
		
	}
	
	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			NAND = {
				has_law_or_variant = law_type:law_federal_state 
				is_a_democracy = yes
			}
		}
		effect = {
		
			#Remove all representatives
			hidden_effect = { 
				every_scope_character = {
					limit = { has_role = character_role_representative }
					remove_character_role = character_role_representative
				}
			}
			
			#Clear variables
			remove_variable = representatives_quota
			remove_variable = representatives_count
			remove_variable = representative_state
		
		}
	}
	
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_effect_character_list_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = "blue_negative"
		trigger = {
			hidden_trigger = { always = no }
		}
		effect = {
			#Mark Princely Rulers
			every_scope_character = {
				limit = {
					has_role = character_role_representative
				}
				save_scope_as = representative_character
				custom_tooltip = federal_representative_character_label.tt
			}
		}
	}
	
	on_monthly_pulse = {
	
		random_events = {
			2 = 0
			1 = warlord_state_system.3
		}
		
	}
	
	on_yearly_pulse = {
	
		effect = {
			
			generate_representatives_for_country_federal = yes
			
		}
		
	}

	weight = 50

	should_be_pinned_by_default_uninvolved_or_context = no

	transferable = no

	progressbar = no
}
