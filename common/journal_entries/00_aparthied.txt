je_apartheid = {
	icon = "gfx/interface/icons/law_icons/racial_segregation.dds"

	group = je_group_internal_affairs

	possible = {
		
		#Is a Settler Colony
		OR = {
			has_variable = settler_colony_country
			c:SAF ?= this
			c:RHO ?= this
			c:ZAR ?= this
			c:FAG ?= this
		}
		NOT = { has_law_or_variant = law_type:law_multicultural }
		
	}

	immediate = {
		
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = "blue_negative"
		trigger = {
		
			#Abolished Apartheid
			has_law_or_variant = law_type:law_multicultural
			is_sufficiently_independent = yes
			
		}
		effect = {
			
			#Change country tags if applicable
			if = {
				limit = {
					NOT = { exists = c:RSA }
					c:SAF ?= this
				}
				
				change_tag = RSA
				
				add_primary_culture = cu:south_african
				remove_primary_culture = cu:boer
				
			}
			else_if = {
				limit = {
					NOT = { exists = c:ZIM }
					OR = {
						c:RHO ?= this
						c:ZAR ?= this
					}
				}
				
				change_tag = ZIM
				
				add_primary_culture = cu:zimbabwean
				remove_primary_culture = cu:rhodesian
				
			}
			else_if = {
				limit = {
					NOT = { exists = c:ALG }
					c:FAG ?= this
				}
				
				change_tag = ALG
				
				add_primary_culture = cu:algerian
				remove_primary_culture = cu:french
				
			}
			else_if = {
				limit = {
					has_variable = settler_colony_country
				}
				capital ?= {
					save_scope_as = state_in_question
					ordered_scope_pop = {
						limit = {
							culture ?= {
								save_temporary_scope_as = colonial_culture_in_question_temp 
							}
							scope:state_in_question ?= { 
								state_region = { 
									is_homeland = scope:colonial_culture_in_question_temp
								}
							}
						}
						order_by = total_size
						position = 0
						check_range_bounds = no
						culture ?= { save_scope_as = colonial_culture_in_question }
					}
				}
				every_primary_culture = {
					save_scope_as = settler_primary_culture
					ROOT = { remove_primary_culture = scope:settler_primary_culture }
				}
				add_primary_culture = scope:colonial_culture_in_question
				set_variable = dynamic_decolonisation_country
			}
			
			#Remove foreigners
			hidden_effect = {
				every_scope_character = {
					limit = { 
						NOT = { is_primary_culture_of = ROOT }
					}
					retire_character = yes
				}
			}
			
			#National Reconciliation
			add_modifier = {
				name = national_reconciliation
				years = 5
				is_decaying = yes
			}
			
		}
	}
	
	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes

	transferable = no

	progressbar = no
}
