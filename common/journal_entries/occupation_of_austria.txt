je_occupation_of_austria = {
	icon = "gfx/interface/icons/war_goals/unify_germany.dds"

	group = je_group_historical_content
	
	scripted_button = je_occupation_of_austria_end_occupation_button

	is_shown_when_inactive = {
		exists = c:AUS
		OR = { 
			AND = { 
				c:AUS ?= { has_modifier = condominium_modifier }
				has_diplomatic_pact = { who = c:AUS type = condominium is_initiator = yes }
			}
			c:AUS ?= { 
				is_direct_subject_of = ROOT 
				is_sufficiently_independent = no
			}
		}
	}

	possible = {
		always = yes
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_occupation_of_austria_completed_desc
		color = "gold_positive"
		trigger = {
			#Austria exists
			exists = c:AUS
			#Left Austria
			NOR = {
				has_diplomatic_pact = { who = c:AUS type = condominium is_initiator = yes }
				c:AUS ?= {
					is_direct_subject_of = ROOT
					is_sufficiently_independent = no
				}
			}
			#THIS is not at war
			is_at_war = no
		}
	}
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_occupation_of_austria_failed_desc
		color = "blue_negative"
		trigger = {
			#Austria does not exist
			NOT = { exists = c:AUS }
		}
	}
	

	
	# zero or more static modifiers that will be applied to the Journal Entry when it activates, where they will propagate to the country
	modifiers_while_active = {
		mandate_owner_modifier
	}

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
