je_subsidiary_monarchy = {
	icon = "gfx/interface/icons/diplomatic_action_icons/doctrine_of_lapse.dds"

	group = je_group_political_organs
	
	scripted_button = je_consolidate_princely_states_button

	is_shown_when_inactive = {
		
		is_a_subsidiary_monarchy = yes
		
	}

	possible = {
		
		#is_at_war = no
		
		hidden_trigger = { NOT = { has_variable = block_constitutional_monarchy_generation } }
		
	}

	immediate = {
		
		#Generate subsidiary monarchs
		if = {
			limit = { NOT = { has_variable = block_subsidiary_monarch_generation } }
			generate_subsidiary_monarchs_for_country = yes
		}
		
		#Calculate Subsidiary Monarchy ratio
		calculate_subsidiary_monarch_ratio = yes
		
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = "blue_negative"
		trigger = {
			NOT = {
				has_law_or_variant = law_type:law_HOS_monarch
			}
			is_a_subsidiary_monarchy = no
		}
		effect = {
			set_subsidiary_monarchs_amendment = yes
		}
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"
		trigger = {
			active_law:lawgroup_governance_principles ?= {
				has_amendment = amendment_type:amendment_elective_monarchy
			}
		}
		effect = {
			upgrade_subsidiary_monarchs_to_elective_monarchs = yes
		}
	}
	
	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			NOT = {
				is_a_subsidiary_monarchy = yes
			}
		}
		effect = {
		
			#Remove all subsidiary monarchs
			every_scope_character = {
				limit = { has_role = character_role_subsidiary_monarch }
				remove_character_role = character_role_subsidiary_monarch
			}
			
			#Clear variables
			remove_variable = subsidiary_monarchs_quota
			remove_variable = subsidiary_monarchs_count
			remove_variable = subsidiary_monarch_state
		
		}
	}
	
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_effect_character_list_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = "blue_negative"
		trigger = {
			hidden_trigger = { always = no }
		}
		effect = {
			#Mark Princely Rulers
			every_scope_character = {
				limit = {
					has_role = character_role_subsidiary_monarch
				}
				save_scope_as = subsidiary_monarch_character
				custom_tooltip = subsidiary_monarch_character_label.tt
			}
		}
	}
	
	on_yearly_pulse = {
	
		effect = {
			
			if = {
				limit = { 
					has_variable = subsidiary_monarchs_quota
					any_scope_character = { 
						has_role = character_role_subsidiary_monarch
						count < owner.var:subsidiary_monarchs_quota
					}
				}
				generate_subsidiary_monarchs_for_country = yes
			}
			
		}
		
	}
	
	on_monthly_pulse = {
	
		random_events = {
			2 = 0
			1 = warlord_state_system.3
		}
		
		effect = {
			
			#Calculate Subsidiary Monarchy ratio
			calculate_subsidiary_monarch_ratio = yes
			
		}
		
	}
	
	# zero or more blocks, each applying a static modifier while the Journal Entry is active with its effects multiplied by a script value
	scaling_modifiers_while_active = {
		modifier = subsidiary_monarchy_system
		value = {
			value = var:subsidiary_monarch_ratio
			min = 0
		}
	}

	weight = 50

	should_be_pinned_by_default_uninvolved_or_context = no

	transferable = no

	progressbar = no
}
