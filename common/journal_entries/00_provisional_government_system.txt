je_provisional_government_system = {
	icon = "gfx/interface/icons/event_icons/event_carlist_beret.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		
		#Revolutionaries won civil war
		has_variable = civil_war_country_var
		is_at_war = no
		
	}

	possible = {
		
		#Not already has a power struggle
		NOR = {
			has_journal_entry = je_successor_struggle_system
			has_journal_entry = je_power_struggle_system
		}
		
	}

	immediate = {
		
		#Candidate 1
		ruler ?= { 
			save_scope_as = provisional_government_candidate_1
			owner ?= {
				set_variable = {
					name = provisional_government_candidate_1
					value = 0
				}
			}
		}
		
		#Candidate 2
		ordered_interest_group = {
			limit = {
				NOT = { is_same_interest_group_type = root.ruler.interest_group }
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = provisional_government_candidate_2_ig
		}
		ordered_scope_character = {
			limit = {
				is_ruler_of_own_country = no
				is_heir_of_own_country = no
				OR = { 
					has_role_of_type = politician
					has_role_of_type = general
					has_role_of_type = admiral
				}
				interest_group ?= { is_same_interest_group_type = scope:provisional_government_candidate_2_ig }
			}
			order_by = prominence
			position = 0
			check_range_bounds = no
			save_scope_as = provisional_government_candidate_2
			owner ?= {
				set_variable = {
					name = provisional_government_candidate_2
					value = 0
				}
			}
			hidden_effect = { remove_modifier = leadership_contender_modifier }
			add_modifier = { name = leadership_contender_modifier years = 5 }
		}
		
		#Candidate 3
		ordered_interest_group = {
			limit = {
				NOT = { is_same_interest_group_type = root.ruler.interest_group }
			}
			order_by = ig_clout
			position = 1
			check_range_bounds = no
			save_scope_as = provisional_government_candidate_3_ig
		}
		ordered_scope_character = {
			limit = {
				is_ruler_of_own_country = no
				is_heir_of_own_country = no
				OR = { 
					has_role_of_type = politician
					has_role_of_type = general
					has_role_of_type = admiral
				}
				interest_group ?= { is_same_interest_group_type = scope:provisional_government_candidate_3_ig }
			}
			order_by = prominence
			position = 1
			check_range_bounds = no
			save_scope_as = provisional_government_candidate_3
			owner ?= {
				set_variable = {
					name = provisional_government_candidate_3
					value = 0
				}
			}
			hidden_effect = { remove_modifier = leadership_contender_modifier }
			add_modifier = { name = leadership_contender_modifier years = 5 }
		}
		
	}

	complete = {
		
		OR = {
		
			#Won competition
			trigger_if = {
				limit = { has_variable = provisional_government_candidate_1 }
				custom_tooltip = {
					text = provisional_government_candidate_1_completed.tt
					var:provisional_government_candidate_1 >= 100
				}
			}
			trigger_if = {
				limit = { has_variable = provisional_government_candidate_2 }
				custom_tooltip = {
					text = provisional_government_candidate_2_completed.tt
					var:provisional_government_candidate_2 >= 100
				}
			}
			trigger_if = {
				limit = { has_variable = provisional_government_candidate_3 }
				custom_tooltip = {
					text = provisional_government_candidate_3_completed.tt
					var:provisional_government_candidate_3 >= 100
				}
			}
			
			#Ruler resigned
			custom_tooltip = {
				text = ruler_resigned_in_leadership_challenge.tt
				has_variable = ruler_resigned_in_leadership_challenge
			}
			
			#All rivals eliminated
			custom_tooltip = {
				text = no_leadership_challengers_remains.tt
				AND = {
					exists = scope:provisional_government_candidate_1
					OR = {
						NOT = { exists = scope:provisional_government_candidate_2 }
						scope:provisional_government_candidate_2 ?= { has_role = character_role_prisoner }
					}
					OR = {
						NOT = { exists = scope:provisional_government_candidate_3 }
						scope:provisional_government_candidate_3 ?= { has_role = character_role_prisoner }
					}
				}
			}
			
		}
		
	}
	
	on_complete = {
		
		#Remove flags
		remove_variable = civil_war_country_var
		
		#Declare victor
		if = {
			limit = { 
				has_variable = provisional_government_candidate_1
				var:provisional_government_candidate_1 >= 100
			}
			scope:provisional_government_candidate_1 ?= { set_character_as_ruler_cwe = yes }
		}
		else_if = {
			limit = { 
				has_variable = provisional_government_candidate_2
				var:provisional_government_candidate_2 >= 100
			}
			scope:provisional_government_candidate_2 ?= { set_character_as_ruler_cwe = yes }
		}
		else_if = {
			limit = { 
				has_variable = provisional_government_candidate_3
				var:provisional_government_candidate_3 >= 100
			}
			scope:provisional_government_candidate_3 ?= { set_character_as_ruler_cwe = yes }
		}
		
		#Ruler resigned
		if = {
			limit = { 
				has_variable = ruler_resigned_in_leadership_challenge
				var:provisional_government_candidate_2 >= var:provisional_government_candidate_3
			}
			scope:provisional_government_candidate_2 ?= { set_character_as_ruler_cwe = yes }
			if = {
				limit = { exists = scope:provisional_government_candidate_3 }
				set_heir = scope:provisional_government_candidate_3
			}
			remove_variable = ruler_resigned_in_leadership_challenge
		}
		else_if = {
			limit = { 
				has_variable = ruler_resigned_in_leadership_challenge
				var:provisional_government_candidate_3 >= var:provisional_government_candidate_2
			}
			scope:provisional_government_candidate_3 ?= { set_character_as_ruler_cwe = yes }
			if = {
				limit = { exists = scope:provisional_government_candidate_2 }
				set_heir = scope:provisional_government_candidate_2
			}
			remove_variable = ruler_resigned_in_leadership_challenge
		}
		
		#Regime Change from New Ruler
		if = {
			limit = { scope:provisional_government_candidate_1 ?= { is_ruler_of_own_country = no } }
			custom_tooltip = {
				text = form_new_provisional_government.tt
				if = {
					limit = { ruler.interest_group ?= { has_ideology = ideology:ideology_monarchist } }
					activate_law = law_type:law_HOS_monarch
				}
				else_if = {
					limit = { ruler.interest_group ?= { has_ideology = ideology:ideology_republican } }
					activate_law = law_type:law_HOS_president
				}
				else_if = {
					limit = { ruler.interest_group ?= { has_ideology = ideology:ideology_theocrat } }
					activate_law = law_type:law_HOS_religious_leader
				}
				else_if = {
					limit = { ruler.interest_group ?= { has_ideology = ideology:ideology_militarist } }
					activate_law = law_type:law_HOS_nationalist_leader
				}
				else_if = {
					limit = { ruler.interest_group ?= { has_ideology = ideology:ideology_communist } }
					activate_law = law_type:law_HOS_nationalist_leader
				}
				else_if = {
					limit = { ruler.interest_group ?= { has_ideology = ideology:ideology_communist } }
					activate_law = law_type:law_HOS_communist_leader
				}
			}
		}
		
		#Domestic Peace
		hidden_effect = { has_modifier = strengthened_government_authority }
		add_modifier = { 
			name = strengthened_government_authority 
			years = 5
			is_decaying = yes
		}
		
		#Clean up modifiers
		hidden_effect = {
			every_scope_character = {
				limit = { has_modifier = leadership_contender_modifier }
				remove_modifier = leadership_contender_modifier
			}
		}
		
	}
	
	fail = {
		
		#Ruler is no longer Ruler
		NOT = { exists = scope:provisional_government_candidate_1 }
		scope:provisional_government_candidate_1 ?= { is_ruler_of_own_country = no }
		
	}
	
	on_fail = {
	
		#Clean up modifiers
		hidden_effect = {
			every_scope_character = {
				limit = { has_modifier = leadership_contender_modifier }
				remove_modifier = leadership_contender_modifier
			}
		}
		
		#Remove flags
		remove_variable = civil_war_country_var
	
	}
	
	on_monthly_pulse = {
	
		effect = {
			
			#Calculate progress for Candidate 1
			if = {
				limit = { 
					has_variable = provisional_government_candidate_1
					exists = ruler
				}
				
				#Popularity
				if = { limit = { ruler.popularity >= 0.20 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.popularity >= 0.40 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.popularity >= 0.60 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.popularity >= 0.80 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.popularity >= 1.00 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				
				#IG Clout
				if = { limit = { ruler.interest_group.ig_clout >= 0.05 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.interest_group.ig_clout >= 0.10 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.interest_group.ig_clout >= 0.20 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.interest_group.ig_clout >= 0.30 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.interest_group.ig_clout >= 0.40 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.interest_group.ig_clout >= 0.50 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.interest_group.ig_clout >= 0.60 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.interest_group.ig_clout >= 0.70 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.interest_group.ig_clout >= 0.80 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				if = { limit = { ruler.interest_group.ig_clout >= 0.90 } change_variable = { name = provisional_government_candidate_1 add = 1 } }
				
				#Limit
				clamp_variable = {
					name = provisional_government_candidate_1
					max = 100
					min = 0
				}
				
			}
			
			#Calculate progress for Candidate 2
			if = {
				limit = { 
					has_variable = provisional_government_candidate_2
					exists = scope:provisional_government_candidate_2
				}
				
				#Popularity
				if = { limit = { scope:provisional_government_candidate_2.popularity >= 0.20 } change_variable = { name = provisional_government_candidate_2 add = 1 } }
				if = { limit = { scope:provisional_government_candidate_2.popularity >= 0.40 } change_variable = { name = provisional_government_candidate_2 add = 1 } }
				if = { limit = { scope:provisional_government_candidate_2.popularity >= 0.60 } change_variable = { name = provisional_government_candidate_2 add = 1 } }
				if = { limit = { scope:provisional_government_candidate_2.popularity >= 0.80 } change_variable = { name = provisional_government_candidate_2 add = 1 } }
				if = { limit = { scope:provisional_government_candidate_2.popularity >= 1.00 } change_variable = { name = provisional_government_candidate_2 add = 1 } }
				
				#IG Clout
				if = { limit = { scope:provisional_government_candidate_2.interest_group.ig_clout >= 0.05 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.interest_group.ig_clout >= 0.10 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.interest_group.ig_clout >= 0.20 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.interest_group.ig_clout >= 0.30 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.interest_group.ig_clout >= 0.40 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.interest_group.ig_clout >= 0.50 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.interest_group.ig_clout >= 0.60 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.interest_group.ig_clout >= 0.70 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.interest_group.ig_clout >= 0.80 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.interest_group.ig_clout >= 0.90 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				
				#Prominence
				if = { limit = { scope:provisional_government_candidate_2.prominence >= 10 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.prominence >= 20 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.prominence >= 30 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.prominence >= 40 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.prominence >= 50 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.prominence >= 60 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.prominence >= 70 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.prominence >= 80 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.prominence >= 90 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_2.prominence >= 100 } change_variable = { name = provisional_government_candidate_2 add = 0.5 } }
				
				#Limit
				clamp_variable = {
					name = provisional_government_candidate_2
					max = 100
					min = 0
				}
				
			}
			
			#Calculate progress for Candidate 3
			if = {
				limit = { 
					has_variable = provisional_government_candidate_3
					exists = scope:provisional_government_candidate_3
				}
				
				#Popularity
				if = { limit = { scope:provisional_government_candidate_3.popularity >= 0.20 } change_variable = { name = provisional_government_candidate_3 add = 1 } }
				if = { limit = { scope:provisional_government_candidate_3.popularity >= 0.40 } change_variable = { name = provisional_government_candidate_3 add = 1 } }
				if = { limit = { scope:provisional_government_candidate_3.popularity >= 0.60 } change_variable = { name = provisional_government_candidate_3 add = 1 } }
				if = { limit = { scope:provisional_government_candidate_3.popularity >= 0.80 } change_variable = { name = provisional_government_candidate_3 add = 1 } }
				if = { limit = { scope:provisional_government_candidate_3.popularity >= 1.00 } change_variable = { name = provisional_government_candidate_3 add = 1 } }
				
				#IG Clout
				if = { limit = { scope:provisional_government_candidate_3.interest_group.ig_clout >= 0.05 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.interest_group.ig_clout >= 0.10 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.interest_group.ig_clout >= 0.20 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.interest_group.ig_clout >= 0.30 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.interest_group.ig_clout >= 0.40 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.interest_group.ig_clout >= 0.50 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.interest_group.ig_clout >= 0.60 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.interest_group.ig_clout >= 0.70 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.interest_group.ig_clout >= 0.80 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.interest_group.ig_clout >= 0.90 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				
				#Prominence
				if = { limit = { scope:provisional_government_candidate_3.prominence >= 10 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.prominence >= 20 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.prominence >= 30 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.prominence >= 40 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.prominence >= 50 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.prominence >= 60 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.prominence >= 70 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.prominence >= 80 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.prominence >= 90 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				if = { limit = { scope:provisional_government_candidate_3.prominence >= 100 } change_variable = { name = provisional_government_candidate_3 add = 0.5 } }
				
				#Limit
				clamp_variable = {
					name = provisional_government_candidate_3
					max = 100
					min = 0
				}
				
			}
			
		}
		
	}
	
	# zero or more static modifiers that will be applied to the Journal Entry when it activates, where they will propagate to the country
	modifiers_while_active = {
		power_vacuum_modifier
	}

	weight = 20

	should_be_pinned_by_default_uninvolved_or_context = yes

	transferable = no

	progressbar = no
	
}
