je_british_palestine = {
	icon = "gfx/interface/icons/invention_icons/colonization.dds"

	group = je_group_internal_affairs

	possible = {
		
		#Added by script
		
	}

	immediate = {
		
	}

	complete = {
	
		#Independent
		is_subject = no
		is_sufficiently_independent = yes
		
	}
	
	on_complete = {
		
		#Become Israel - if not multicultural
		if = {
			limit = {
				NOT = { exists = c:ISR }
				c:PIR ?= this
				NOT = { has_law_or_variant = law_type:law_multicultural }
			}
			
			#Become Israel
			change_tag = ISR
			remove_primary_culture = cu:palestinian
			set_state_religion = root.country_definition.religion
			
			#Clean up politics
			every_scope_character = {
				limit = {
					culture = cu:palestinian
				}
				retire_character = yes
			}
			
		}
		
	}
	
	modifiers_while_active = {
		colonial_citizenship_modifier_3
	}

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = no

	transferable = no

	progressbar = no
}
