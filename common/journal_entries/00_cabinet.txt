je_cabinet = {
	icon = "gfx/interface/icons/law_icons/single_party_state.dds"

	group = je_group_political_organs

	scripted_button = je_issue_executive_order_button
	#scripted_button = je_issue_royal_decree_button
	scripted_button = je_call_elections_button
	
	scripted_button = je_enact_state_of_emergency_button
	scripted_button = je_lift_state_of_emergency_button
	
	#scripted_button = je_commence_democratic_reform_button
	
	#scripted_button = je_controlled_elections_enabled_button
	#scripted_button = je_controlled_elections_disabled_button
	
	scripted_button = je_establish_republican_dynasty_button

	possible = {
		
		exists = ruler
		exists = root.ruler.interest_group
		
	}

	immediate = {
	
		#Mark ruler
		ruler ?= {
			save_scope_as = ruler_scope
		}
		
		#Initialise Cabinet Unity
		calculate_cabinet_unity = yes
		
		#Mark IGs
		ig:ig_armed_forces ?= { save_scope_as = ig_armed_forces_cabinet_scope }
		ig:ig_intelligentsia ?= { save_scope_as = ig_intelligentsia_cabinet_scope }
		
		#Reset Dynastic Republic
		if = {
			limit = { 
				is_a_dynastic_government = no
			}
			scope:ruler_scope ?= { remove_variable = dynastic_ruler_enabled }
		}
		
		#Check for warlordism
		scope:ruler_scope ?= { evaluate_warlordism_eligibility = yes }
		
	}

	invalid = {
		
		OR = {
			scope:ruler_scope ?= { is_ruler_of_own_country = no }
			NOT = { exists = scope:ruler_scope }
		}
		
	}
	
	on_invalid = {
		
		custom_tooltip = {
			text = government_resigns.tt
		
			#Clear all modifiers
			every_interest_group = {
				limit = { has_modifier = cabinet_minister_ig }
				remove_modifier = cabinet_minister_ig
			}
			every_scope_character = {
				limit = { has_role = character_role_cabinet_minister }
				remove_character_role = character_role_cabinet_minister
			}
		
		}
		
		#Clear variables
		remove_variable = strong_cabinet_unity
		remove_variable = medium_cabinet_unity
		remove_variable = low_cabinet_unity
		remove_variable = cabinet_unity_var
		remove_variable = cabinet_unity_divisor_var
		remove_variable = cabinet_incompetence_var
		
		#Clear modifiers
		remove_modifier = cabinet_incompetence_modifier
		
		#Reset Dynastic Republic
		if = {
			limit = { 
				is_a_dynastic_government = no
			}
			scope:ruler_scope ?= { remove_variable = dynastic_ruler_enabled }
		}
		
		#Bipartisanship
		remove_variable = has_bipartisan_plan
		every_interest_group = {
			remove_variable = bipartisan_participant
		}
		
	}
	
	on_monthly_pulse = {
	
		random_events = {
			2 = 0
			1 = cabinet_system.1
		}
		
		effect = {
			
			#Calculate cabinet unity
			calculate_cabinet_unity = yes
			
			#Domestic discontent with unjustified state of emergency
			if = {
				limit = {
					has_modifier = state_of_emergency_issued
					has_revolution = no
					is_at_war = no
				}
				add_radicals = {
					value = small_radicals
				}
			}
			
			#Clean up any accidental monarchs
			if = {
				limit = {
					is_a_monarchy = no
					any_scope_character = { has_role = character_role_monarch }
				}
				every_scope_character = {
					limit = {
						has_role = character_role_monarch
					}
					remove_character_role = character_role_monarch
					set_character_as_royal = yes
				}
			}
			else_if = {
				limit = {
					shares_same_monarch = yes
					overlord ?= { is_a_monarchy = yes }
					any_scope_character = { has_role = character_role_monarch }
				}
				every_scope_character = {
					limit = {
						has_role = character_role_monarch
					}
					remove_character_role = character_role_monarch
					set_character_as_royal = yes
				}
			}
			
		}
		
	}
	
	on_fail = {
	
		#Mark Cabinet Officials
		every_scope_character = {
			limit = {
				character_is_key_state_officeholder = yes
			}
			save_scope_as = cabinet_character
			custom_tooltip = cabinet_character_label.tt
		}
		
	}

	weight = 2

	should_be_pinned_by_default_uninvolved_or_context = no

	transferable = no

	progressbar = no
}
