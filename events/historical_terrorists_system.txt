#Renato Curcio – Italy
historical_terrorists_events.1 = {
    type = country_event

    title = ITA_terrorist_Renato_Curcio.t
    desc = ITA_terrorist_Renato_Curcio.d
    #flavor = ITA_terrorist_Renato_Curcio.f

    event_image = { texture = "europenorthamerica_political_extremism" }
    icon = "gfx/interface/icons/invention_icons/anarchy.dds"
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    duration = 1

    trigger = {
	
        c:ITA ?= this
		
        game_date >= 1972.1.1
        game_date < 1982.1.1
		
        is_a_communist_government = no
		
        NOT = { has_variable = ITA_terrorist_Renato_Curcio_hte }
		
        any_political_movement = { is_political_movement_type = movement_communist }
		
    }

    immediate = {
        set_variable = ITA_terrorist_Renato_Curcio_hte
    }

    option = {
        name = historical_terrorists_events.option

        create_character = {
            first_name = "Renato_Curcio_first_name"
            last_name = "Renato_Curcio_last_name"
            is_agitator = yes historical = yes
            birth_date = 1941.9.23
            interest_group = ig_trade_unions
            ideology = ideology_agitator_communist
            culture = cu:italian
            religion = rel:catholic
            traits = {
                wrathful
                persistent
                charismatic
            }
            on_created = {
                add_character_role = character_role_terrorist
            }
        }
    }
}
