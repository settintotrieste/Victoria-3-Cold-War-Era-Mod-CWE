namespace = cabinet_system

#Minister Resigns
cabinet_system.1 = {
	type = country_event

	title = cabinet_minister_resigns.t
	desc = cabinet_minister_resigns.d
	flavor = cabinet_minister_resigns.f

	event_image = {
		texture = "unspecific_ruler_speaking_to_people"
	}

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/leader_speaking_to_a_group_of_people"

	duration = 1

	trigger = {
		
		#Angry Cabinet Member
		any_scope_character = {
			has_role = character_role_cabinet_minister
			is_ruler_of_own_country = no
			OR = {
				interest_group ?= { ig_approval <= unhappy }
				AND = {
					owner = { enacting_any_law = yes }
					OR = {
						law_enactment_stance = {
							law = owner.currently_enacting_law.type
							value <= disapprove
						}
						interest_group ?= {
							law_enactment_stance = {
								law = owner.currently_enacting_law.type
								value <= disapprove
							}
						}
					}
				}
			}
		}
		
		NOT = { has_variable = cabinet_system_1_fired }
		
	}

	immediate = {
		
		random_scope_character = {
			limit = {
				has_role = character_role_cabinet_minister
				is_ruler_of_own_country = no
				OR = {
					interest_group ?= { ig_approval <= unhappy }
					AND = {
						owner = { enacting_any_law = yes }
						OR = {
							law_enactment_stance = {
								law = owner.currently_enacting_law.type
								value <= disapprove
							}
							interest_group ?= {
								law_enactment_stance = {
									law = owner.currently_enacting_law.type
									value <= disapprove
								}
							}
						}
					}
				}
			}
			save_scope_as = angry_cabinet_member
		}
		
		set_variable = {
			name = cabinet_system_1_fired 
			days = 32
		}
		
	}

	option = {
		name = cabinet_system_option #Who cares about a loser!
		default_option = yes
		
		#Minister resigns
		if = {
			limit = { exists = scope:angry_cabinet_member }
			scope:angry_cabinet_member ?= {
				if = {
					limit = {
						has_role = character_role_cabinet_minister
					}
					remove_character_role = character_role_cabinet_minister
				}
				interest_group ?= { save_scope_as = angry_cabinet_member_ig }
			}
			add_radicals = {
				value = small_radicals
				interest_group = scope:angry_cabinet_member_ig
			}
		}
		
		#Set law back
		if = {
			limit = {
				enacting_any_law = yes
			}
			add_enactment_setback = 1
		}
		
		remove_variable = cabinet_system_1_fired
		
	}
	
}
