namespace = politicians_system

#Randomly retire rulers
politicians_system.1 = {
	type = country_event
	hidden = yes

	trigger = {
		
		#Not at war
		is_at_war = no
		
	}

	immediate = {
		
		#Retire the elderly
		random_scope_character = {
			limit = {
				OR = {
					AND = {
						age >= 100
						is_historical = yes
					}
					AND = {
						age >= 90
						is_historical = no
					}
					AND = {
						age >= 80
						OR = {
							has_trait = psychological_affliction
							has_trait = cancer
							has_trait = senile
							has_trait = wounded
							has_trait = sickly
							has_trait = alcoholic
							has_trait = opium_addiction
							has_trait = cocaine_addiction
							has_trait = tuberculosis
							has_trait = syphilis
							has_trait = kidney_stones
						}
						is_historical = no
					}
				}
			}
			retire_character = yes
		}
		
	}

}

							set_ruling_party = yes
						}
					}
					else = {
						add_ruling_interest_group = yes
					}
				}
			}
		}

		if = {
			limit = { scope:ig_industrialists_scope_ig ?= { NAND = { is_in_government = yes ig_government_power_share > 0.5 } } }

				value = small_radicals

			add_loyalists = {
				value = small_radicals
				interest_group = scope:most_powerful_gov_ig
			}
		}

			base = 10
			modifier = { 
				trigger = { 
					scope:ig_industrialists_scope_ig ?= {

    flavor = ineligible_ruler_character.f 							# Flavor for event; ADD LOCALISATION, IF REQUIRED

    gui_window = event_window_1char_tabloid

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

    duration = 1 # Duration to fire if triggers are met
	
	trigger = {														# What triggers the event, if using other events or journal entries leave empty
		
		#Triggered by event
		ruler ?= { has_trait = character_ineligible_as_office_holder }
		
    }
	
	immediate = {
		
		random_scope_character = {
			limit = { 
				is_ruler_of_own_country = yes
				has_trait = character_ineligible_as_office_holder
			}
			save_scope_as = ineligible_ruler_character
		}
		
	}
    
	cancellation_trigger = {
		OR = { 
			NOT = { exists = scope:ineligible_ruler_character }
			scope:ineligible_ruler_character ?= { is_ruler_of_own_country = no }
		}
	}
	
	option = {
        name = ineligible_ruler_character.option.1					# Let them be free
		highlighted_option = yes
		
		#Anger every other interest group
		scope:ineligible_ruler_character.interest_group ?= { 
			save_scope_as = THIS_IG_scope 
			owner ?= {
				add_loyalists = {
					value = very_large_radicals
					interest_group = scope:THIS_IG_scope
				}
				every_interest_group = {
					limit = {
						NOT = { this ?= scope:THIS_IG_scope }
					}
					save_scope_as = ig_scope
					owner ?= {
						add_radicals = {
							value = medium_radicals
							interest_group = scope:ig_scope
						}
					}
				}
			}
		}
		
    }

}

			limit = {
				NOT = { has_variable = very_reformist_party }
			}
			every_interest_group = {
				limit = {
					is_in_government = yes
				}
				save_scope_as = THIS_IG_scope
				owner ?= {
					add_radicals = {
						value = small_radicals
						interest_group = scope:THIS_IG_scope
					}
				}
			}
		}
		
    }

        name = recruit_politician_character.option.6				# Diversity
		trigger = {
			any_interest_group = {
				is_in_government = yes
				count > 1
			}
		}
		
		#Select different politician
		random_interest_group = {
			limit = { 
				is_in_government = yes
				NOT = { is_same_interest_group_type = root.ruler.interest_group }
			}
			save_scope_as = THIS_IG_scope
		}
		if = {
			limit = { exists = scope:THIS_IG_scope }
			random_list = {
				1 = {
					create_character = {
						role = politician
						interest_group = scope:THIS_IG_scope
						on_created = {
							add_trait = political_appointee
						}
						save_scope_as = new_politician_character_scope
					}
				}
				1 = {
					create_character = {
						role = politician
						female = yes
						interest_group = scope:THIS_IG_scope
						on_created = {
							add_trait = political_appointee
						}
						save_scope_as = new_politician_character_scope
					}
				}
			}
			post_notification = new_politician_recruited
		}
		
		#Party effects
		if = {
			limit = {
				NOR = {
					has_variable = moderate_party
					has_variable = reformist_party
					has_variable = very_reformist_party
				}
			}
			every_interest_group = {
				limit = {
					is_in_government = yes
				}
				save_scope_as = THIS_IG_scope
				owner ?= {
					add_radicals = {
						value = small_radicals
						interest_group = scope:THIS_IG_scope
					}
				}
			}
		}
