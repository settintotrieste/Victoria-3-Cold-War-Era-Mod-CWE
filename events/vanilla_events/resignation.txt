resignation.1 = {
	type = country_event
	placement = ROOT
	
	title = resignation.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = resignation.1.d1
				trigger = {
					ruler.interest_group = {
						OR = {
							ig_approval <= angry
							ig_counts_as_marginal = yes
						}
					}
				}
			}
			triggered_desc = {
				desc = resignation.1.d2
				trigger = {
					OR = {
						ruler = {
							popularity <= -50
						}
						has_revolution = yes
					}
				}
			}
			triggered_desc = {
				desc = resignation.1.d3
				trigger = {
					ruler = {
						OR = {
							has_trait = psychological_affliction
							has_trait = cancer
							has_trait = senile
							has_trait = wounded
						}
					}
				}
			}
			triggered_desc = {
				desc = resignation.1.d4
				trigger = {
					always = yes
				}
			}
		}
	}
	flavor = resignation.1.f

	gui_window = event_window_1char_tabloid
	
	left_icon = scope:resigning_politician

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	
	duration = 3
	
	trigger = {
		# triggered by resign_from_office interaction
	}

	immediate = {
	
		ruler ?= {
			save_scope_as = resigning_politician
			interest_group ?= {
				save_scope_as = resignation_ig
			}
		}
		
		#Legacy
		if = {
			limit = {
				is_foreign_governor_only_country = yes
			}
			if = {
				limit = { exists = overlord }
				overlord ?= {
					random_primary_culture = { save_scope_as = overlord_primary_culture }
					religion = { save_scope_as = overlord_religion }
				}
			}
			else_if = {
				limit = { has_modifier = condominium_modifier }
				random_country = {
					limit = { has_diplomatic_pact = { who = ROOT type = condominium is_initiator = yes } }
					random_primary_culture = { save_scope_as = overlord_primary_culture }
					religion = { save_scope_as = overlord_religion }
				}
			}
		}
		else = {
			random_primary_culture = {
				save_scope_as = overlord_primary_culture
			}
			religion = {
				save_scope_as = overlord_religion
			}
		}
		
		#Mark candidates
		random_scope_character = {
			limit = {
				has_role_of_type = politician is_ruler_of_own_country = no is_heir_of_own_country = no
				has_ideology = ideology:ideology_autocratic
				interest_group = { is_same_interest_group_type = scope:resignation_ig }
			}
			save_scope_as = replacement_ruler_autocratic
		}
		random_scope_character = {
			limit = {
				has_role_of_type = politician is_ruler_of_own_country = no is_heir_of_own_country = no
				has_ideology = ideology:ideology_power_hungry
				interest_group = { is_same_interest_group_type = scope:resignation_ig }
			}
			save_scope_as = replacement_ruler_power_hungry
		}
		random_scope_character = {
			limit = {
				has_role_of_type = politician is_ruler_of_own_country = no is_heir_of_own_country = no
				has_ideology = ideology:ideology_elitist

			save_scope_as = replacement_ruler_elitist

			limit = {
				has_role_of_type = politician is_ruler_of_own_country = no is_heir_of_own_country = no

		}

			}

		}
		
	}
	
	cancellation_trigger = {
		
		OR = {
			NOT = { exists = scope:resignation_ig }

	}
	
	option = { # Autocratic successor
		name = resignation_1.1.a

			}
		}

					scope:resignation_ig = {
						OR = {
							ig_approval <= angry
							ig_counts_as_marginal = yes
						}
					}

						popularity <= -50
					}
				}
			}
			add_modifier = {
				name = modifier_disastrous_administration
				days = normal_modifier_time
			}
			add_loyalists = {
				value = small_radicals
			}
		}
		
		#Appoint ruler
		scope:replacement_ruler_autocratic ?= {
			set_character_as_ruler_cwe = yes
		}
		
		#Call snap elections where applicable
		if = {
			limit = {

				months = 3

resignation.2 = {
	type = country_event
	placement = ROOT
	
	title = resignation.2.t
	desc = resignation.2.d
	flavor = resignation.2.f

	gui_window = event_window_1char_tabloid
	
	left_icon = scope:resigning_politician

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	
	duration = 3
	
	trigger = {
		# triggered by resign_from_office interaction
	}

	immediate = {
		ruler ?= {
			save_scope_as = resigning_politician
			interest_group = {
				save_scope_as = resignation_ig
			}
		}
		random_civil_war = {
			limit = {
				is_civil_war_type = revolution
			}
			civil_war_save_scopes = yes
		}
	}

	option = {
		name = resignation.2.a
		trigger = {
			exists = heir
			heir ?= { is_adult = yes }
		}
		
		#Resign in favour of successor
		custom_tooltip = {
			text = leader_resigns_from_office_tt
			heir ?= {
				set_character_as_ruler_cwe = yes
			}
			scope:resigning_politician ?= {
				exile_character_with_martyrdom = yes
			}
		}
		if = {
			limit = {
				scope:resigning_politician ?= {
					popularity >= 0
				}
			}
			scope:ongoing_revolution_movement ?= {
				add_modifier = {
					name = modifier_ruler_resigned
					days = long_modifier_time
				}
			}
		}
		else = {
			scope:ongoing_revolution_movement ?= {
				add_modifier = {
					name = modifier_ruler_resigned
					days = long_modifier_time
					multiplier = 3
				}
			}
		}
		
	}

	option = {
		name = resignation.2.b
		default_option = yes
		
		#Leader flees country, successor is kicked out
		custom_tooltip = {
			text = leader_resigns_from_office_tt
			heir ?= {
				disinherit_character = yes
			}
			scope:resigning_politician ?= {
				evaluate_warlordism_eligibility = yes
				exile_character_with_martyrdom = yes
			}
			check_power_struggle_eligibility = yes
		}
		
		#Lose to revolution
		capitulate_to_revolution = yes
		add_radicals = {
			value = large_radicals
			interest_group = scope:resigning_politician.interest_group
		}
		
	}
