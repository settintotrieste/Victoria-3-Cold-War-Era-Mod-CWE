namespace = regency

# Start of the regency
regency.1 = {
	type = country_event
	placement = root

	duration = 1

	title = regency.1.t
	desc = regency.1.d
	flavor = regency.1.f

    event_image = { texture = "unspecific_ruler_speaking_to_people" }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	### No cooldown

	trigger = {
	
		# Is a monarchy, ruler is under age of majority
		has_law_or_variant = law_type:law_HOS_monarch
		ruler_is_governor_general = no
		shares_same_monarch = no
		is_a_regency_government = no
		OR = {
			ruler ?= {
				NOT = { has_role = character_role_regent }
				is_eligible_to_become_monarch = no
			}
			any_scope_character = {
				has_role = character_role_monarch
				is_eligible_to_become_monarch = no
			}
		}
		
		NOT = { has_variable = replacing_regency }
		
	}

	cancellation_trigger = {
		NAND = {
			has_law_or_variant = law_type:law_HOS_monarch
			ruler_is_governor_general = no
			shares_same_monarch = no
			OR = {
				ruler ?= {
					NOT = { has_role = character_role_regent }
					is_eligible_to_become_monarch = no
				}
				any_scope_character = {
					has_role = character_role_monarch
					is_eligible_to_become_monarch = no
				}
			}
		}
	}

	immediate = {
		ordered_interest_group = { # Just the most notable IG in government
			limit = {
				is_in_government = yes
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = ig_candidate_1
		}
		ordered_interest_group = { # Any pro-monarchy IG
			limit = {
				NOT = {
					scope:ig_candidate_1 ?= this
				}
				law_stance = {
					law = law_type::law_HOS_monarch
					value >= approve
				}
				ig_counts_as_marginal = no
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = ig_candidate_2
		}
		ordered_interest_group = {
			limit = {
				NOR = {
					scope:ig_candidate_1 ?= this
					scope:ig_candidate_2 ?= this
				}
				law_stance = {
					law = law_type::law_HOS_monarch
					value >= neutral
				}
				ig_counts_as_marginal = no
				is_insurrectionary = no
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = ig_candidate_3
		}
		ordered_interest_group = {
			limit = {
				NOR = {
					scope:ig_candidate_1 ?= this
					scope:ig_candidate_2 ?= this
					scope:ig_candidate_3 ?= this
				}
				law_stance = {
					law = law_type::law_HOS_monarch
					value >= neutral
				}
				ig_counts_as_marginal = no
				is_insurrectionary = no
			}
			order_by = this.leader.popularity
			position = 0
			check_range_bounds = no
			save_scope_as = ig_candidate_4
		}
		if = {
			limit = {
				NOT = {
					any_scope_character = {
						has_role = character_role_monarch
					}
				}
			}
			ordered_scope_character = {
				limit = {
					has_role = character_role_prince
					is_eligible_to_become_monarch = no
				}
				order_by = age
				position = 0
				check_range_bounds = no
				save_scope_as = monarch_scope
			}
		}
		else = {
			random_scope_character = {
				limit = {
					has_role = character_role_monarch
				}
				save_scope_as = monarch_scope
			}
		}
		set_variable = { name = replacing_regency months = 2 }
	}

	option = {
		name = regency.1.a
		default_option = yes
		set_variable = { name = hide_ruler_change_notification days = 3 }
		scope:ig_candidate_1.leader = {
			set_character_as_regent = yes
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
				exists = scope:monarch_scope
			}
			set_heir = scope:monarch_scope
		}
	}

	option = {
		name = regency.1.b
		trigger = {
			exists = scope:ig_candidate_2
		}
		set_variable = { name = hide_ruler_change_notification days = 3 }
		scope:ig_candidate_2.leader = {
			set_character_as_regent = yes
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
				exists = scope:monarch_scope
			}
			set_heir = scope:monarch_scope
		}
	}

	option = {
		name = regency.1.c
		trigger = {
			exists = scope:ig_candidate_3
		}
		set_variable = { name = hide_ruler_change_notification days = 3 }
		scope:ig_candidate_3.leader = {
			set_character_as_regent = yes
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
				exists = scope:monarch_scope
			}
			set_heir = scope:monarch_scope
		}
	}

	option = {
		name = regency.1.e
		trigger = {
			exists = scope:ig_candidate_4
		}
		set_variable = { name = hide_ruler_change_notification days = 3 }
		scope:ig_candidate_4.leader = {
			set_character_as_regent = yes
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_HOS_monarch
				NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
				exists = scope:monarch_scope
			}
			set_heir = scope:monarch_scope
		}
	}
}

# New regent - for cases in which the current regent is immortal, but clearly needs to die.
regency.2 = {
	type = country_event
	placement = root

	duration = 1

	title = regency.2.t
	desc = regency.2.d
	flavor = regency.2.f

	event_image = { texture = "unspecific_ruler_speaking_to_people" }

	left_icon = scope:regent_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/2Characters"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	trigger = {
		
		#Regency
		is_a_regency_government = yes
		
		#Regent conditions
		OR = {
			
			#Regent has been removed
			NOT = { any_scope_character = { has_role = character_role_regent } }
		
			#Regent is incapacitated
			any_scope_character = {
				has_role = character_role_regent
				OR = {
					age >= 80
					has_trait = psychological_affliction
					has_trait = cancer
					has_trait = senile
				}
			}
			
			#Bypass
			has_variable = dismissing_regent
			
		}
		any_scope_character = {
			has_role = character_role_monarch
			is_eligible_to_become_monarch = no
		}
		
		NOT = { has_variable = replacing_regency }
		
	}

	#cooldown = { months = short_modifier_time }

	cancellation_trigger = {
		NAND = {
			
			has_law_or_variant = law_type:law_HOS_monarch
			ruler_is_governor_general = no
			shares_same_monarch = no
			
			any_scope_character = {
				has_role = character_role_monarch
				is_eligible_to_become_monarch = no
			}
			
		}
	}

	immediate = {
		random_scope_character = {
			limit = { has_role = character_role_regent }
			save_scope_as = regent_scope
		}
		random_scope_character = {
			limit = { has_role = character_role_monarch }
			save_scope_as = monarch_scope
		}
		random_scope_character = {
			limit = { 
				is_ruler_of_own_country = yes
				NOT = { has_role = character_role_monarch }
			}
			interest_group ?= {
				save_scope_as = ig_candidate_1
				if = {
					limit = {
						NOT = {
							any_scope_character = {
								is_ruler_of_own_country = no
								is_heir_of_own_country = no
							}
						}
					}
					ROOT = {
						create_character = {
							interest_group = scope:ig_candidate_1
							role = character_role_politician
							save_scope_as = backup_regent_scope
						}
					}
				}
			}
		}
		ordered_interest_group = { # Just the most notable IG in government
			limit = {
				NOT = {
					scope:ig_candidate_1 ?= this
				}
				is_in_government = yes
				ig_counts_as_marginal = no
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = ig_candidate_2
		}
		ordered_interest_group = { # Any pro-monarchy IG
			limit = {
				NOR = {
					scope:ig_candidate_1 ?= this
					scope:ig_candidate_2 ?= this
				}
				law_stance = {
					law = law_type::law_HOS_monarch
					value >= approve
				}
				ig_counts_as_marginal = no
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = ig_candidate_3
		}
		ordered_interest_group = {
			limit = {
				NOR = {
					scope:ig_candidate_1 ?= this
					scope:ig_candidate_2 ?= this
					scope:ig_candidate_3 ?= this
				}
				law_stance = {
					law = law_type::law_HOS_monarch
					value >= neutral
				}
				is_insurrectionary = no
				ig_counts_as_marginal = no
			}
			order_by = this.leader.popularity
			position = 0
			check_range_bounds = no
			save_scope_as = ig_candidate_4
		}
		set_variable = { name = replacing_regency months = 2 }
	}

	option = {
		name = regency.2.a
		trigger = {
			exists = scope:ig_candidate_1
		}
		default_option = yes
		if = {
			limit = {
				scope:ig_candidate_1 ?= {
					any_scope_character = {
						is_ruler_of_own_country = no
						is_heir_of_own_country = no
					}
				}
			}
			scope:ig_candidate_1 ?= {
				ordered_scope_character = {
					limit = {
						is_ruler_of_own_country = no
						is_heir_of_own_country = no
					}
					order_by = loyalty
					position = 0
					if = {
						limit = {
							has_role_of_type = agitator
						}
						replace_character_roles = { remove = { agitator } add = { politician } }
					}
					hidden_effect = { set_character_as_regent = yes }
				}
			}
		}
		else = {
			scope:backup_regent_scope ?= {
				hidden_effect = { set_character_as_regent = yes }
			}
		}
		scope:regent_scope ?= {
			hidden_effect = {
				if = {
					limit = {
						is_immortal = yes
						has_variable = regent_immortality_var
					}
					set_character_immortal = no
				}
				retire_character = yes
			}
		}
		if = {
			limit = {
				exists = scope:monarch_scope
				NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			}
			set_heir = scope:monarch_scope
		}
	}

	option = {
		name = regency.2.b
		trigger = {
			exists = scope:ig_candidate_2
		}
		scope:ig_candidate_2.leader = {
			hidden_effect = { set_character_as_regent = yes }
		}
		scope:regent_scope ?= {
			hidden_effect = {
				if = {
					limit = {
						is_immortal = yes
						has_variable = regent_immortality_var
					}
					set_character_immortal = no
				}
				retire_character = yes
			}
		}
		if = {
			limit = {
				exists = scope:monarch_scope
				NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			}
			set_heir = scope:monarch_scope
		}
	}

	option = {
		name = regency.2.c
		trigger = {
			exists = scope:ig_candidate_3
		}
		scope:ig_candidate_3.leader = {
			hidden_effect = { set_character_as_regent = yes }
		}
		scope:regent_scope ?= {
			hidden_effect = {
				if = {
					limit = {
						is_immortal = yes
						has_variable = regent_immortality_var
					}
					set_character_immortal = no
				}
				retire_character = yes
			}
		}
		if = {
			limit = {
				exists = scope:monarch_scope
				NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			}
			set_heir = scope:monarch_scope
		}
	}

	option = {
		name = regency.2.e
		trigger = {
			exists = scope:ig_candidate_4
		}
		scope:ig_candidate_4.leader = {
			hidden_effect = { set_character_as_regent = yes }
		}
		scope:regent_scope ?= {
			hidden_effect = {
				if = {
					limit = {
						is_immortal = yes
					}
					set_character_immortal = no
				}
				retire_character = yes
			}
		}
		if = {
			limit = {
				exists = scope:monarch_scope
				NOT = { has_law_or_variant = law_type:law_HOG_prime_minister }
			}
			set_heir = scope:monarch_scope
		}
	}
	
}

# End of the regency
regency.3 = {
	type = country_event
	placement = root

	duration = 1

	title = regency.3.t
	desc = regency.3.d
	flavor = regency.3.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:monarch_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/2Characters"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	#cooldown = { days = 100 }

	trigger = {
		
		#Monarchy
		is_a_regency_government = yes
		
		#Heirs comes of age while regent is still alive
		any_scope_character = {
			NOT = { has_role = character_role_regent }
			has_role = character_role_monarch
			is_eligible_to_become_monarch = yes
		}
		
	}

	cancellation_trigger = {
		NAND = {
		
			#Monarchy
			has_law_or_variant = law_type:law_HOS_monarch
			
			#Heirs comes of age while regent is still alive
			any_scope_character = {
				has_role = character_role_monarch
				is_eligible_to_become_monarch = yes
			}
			
		}
	}

	immediate = {
		random_scope_character = {
			limit = {
				has_role = character_role_regent
			}
			save_scope_as = regent_scope
		}
		random_scope_character = {
			limit = {
				NOT = { has_role = character_role_regent }
				has_role = character_role_monarch
				is_eligible_to_become_monarch = yes
			}
			save_scope_as = monarch_scope
		}
	}

	option = {
		name = regency.3.a
		default_option = yes
		
		#Crown monarch
		scope:monarch_scope ?= {
			set_variable = regency_complete
			set_character_as_monarch = yes
		}
		if = {
			limit = {
				scope:monarch_scope ?= { 
					has_variable = regency_complete 
					is_adult = no
				}
			}
			scope:monarch_scope ?= { remove_variable = regency_complete }
		}
		
		#End Regency
		remove_regency_modifier = yes
		scope:regent_scope ?= {
			remove_character_role = character_role_regent
			clean_up_regent_character = yes
		}
		
	}
	
}
