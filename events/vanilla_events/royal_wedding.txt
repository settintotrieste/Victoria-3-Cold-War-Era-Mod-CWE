	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	
	duration = 1

		# triggered by marriage.2
		exists = scope:royal_marriage_character
		exists = scope:marriage_country
	}

	immediate = {
		capital = {
			save_scope_as = capital_scope
		}
		scope:royal_marriage_character ?= {
			save_scope_as = royal_marriage_character
			set_variable = getting_married # for clothing script
		}
		scope:foreign_ruler_1 ?= {
			save_scope_as = foreign_ruler
		}
	}

	option = { # private wedding for international nobility
		name = wedding.1.a
		default_option = yes
		add_modifier = { # influence mult
			name = royal_marriage
			days = normal_modifier_time
		}
		scope:royal_marriage_character ?= {
			hidden_effect = {
				remove_variable = getting_married
			}
		}
		
		#Transfer character
		if = {
			limit = {
				is_subject_of = scope:foreign_ruler_1.owner
				scope:foreign_ruler_1.owner = { NOT = { THIS = ROOT } }
			}
			scope:royal_marriage_character ?= {
				set_variable = prince_consort_character
				set_variable = is_married
				transfer_character = scope:foreign_ruler_1.owner
			}
		}
		else_if = {
			limit = { 
				country_rank > scope:foreign_ruler_1.owner.country_rank
				scope:foreign_ruler_1.owner = { NOT = { THIS = ROOT } }
			}
			scope:foreign_ruler_1 ?= { 
				transfer_character = ROOT
				set_variable = prince_consort_character
				set_variable = is_married
				if = { limit = { has_role = character_role_monarch } remove_character_role = character_role_monarch }
				set_character_as_royal = yes
			}
		}
		else_if = {
			limit = { 
				scope:foreign_ruler_1 ?= { is_female = yes }
				scope:foreign_ruler_1.owner = { NOT = { THIS = ROOT } }
			}
			scope:foreign_ruler_1 ?= { 
				transfer_character = ROOT
				set_variable = prince_consort_character
				set_variable = is_married
				if = { limit = { has_role = character_role_monarch } remove_character_role = character_role_monarch }
				set_character_as_royal = yes
			}
		}
		else_if = {
			limit = { 
				scope:foreign_ruler_1 ?= { is_female = no }
				scope:foreign_ruler_1.owner = { NOT = { THIS = ROOT } }
			}
			scope:royal_marriage_character ?= {
				set_variable = prince_consort_character
				set_variable = is_married
				if = { limit = { has_role = character_role_monarch } remove_character_role = character_role_monarch }
				set_character_as_royal = yes
				transfer_character = scope:foreign_ruler_1.owner 
			}
		}
		else = {
			if = {
				limit = {
					has_variable = female_princess_lose_title_upon_marriage
					scope:royal_marriage_character ?= {
						is_female = yes
						has_role = character_role_prince
					}
				}
				scope:royal_marriage_character ?= {
					custom_tooltip = {
						text = female_princess_lose_title_upon_marriage.tt
						if = {
							limit = { is_heir_of_own_country = yes }
							disinherit_character = yes
						}
						remove_character_role = character_role_prince
						set_variable = former_royal_character
						if = {
							limit = { NOT = { has_role_of_type = magnate } }
							add_character_role = magnate
						}
					}
				}
				scope:foreign_ruler_1 ?= {
					set_variable = prince_consort_character
					set_variable = is_married
					hidden_effect = { set_is_noble = yes }
				}
			}
			else = {
				scope:foreign_ruler_1 ?= {
					set_variable = prince_consort_character
					set_variable = is_married
					set_character_as_royal = yes
				}
			}
		}
		
	}
