wedding.1 = {

		capital ?= {

			save_scope_as = foreign_ruler

	option = { # private wedding for international nobility
		name = wedding.1.a
		default_option = yes
		add_modifier = { # influence mult
			name = royal_marriage
			days = normal_modifier_time
		}
		scope:royal_marriage_character ?= {
			hidden_effect = {
				remove_variable = getting_married
			}
		}
		
		#Transfer character
		if = {
			limit = {
				exists = scope:foreign_ruler_1
				is_subject_of = scope:foreign_ruler_1.owner
				scope:foreign_ruler_1.owner = { NOT = { THIS = ROOT } }
			}
			scope:royal_marriage_character ?= {
				set_variable = prince_consort_character
				set_variable = is_married
				set_potential_personal_union_royal_consort = yes
				transfer_character = scope:foreign_ruler_1.owner
			}
		}
		else_if = {
			limit = { 
				exists = scope:foreign_ruler_1
				country_rank > scope:foreign_ruler_1.owner.country_rank
				scope:foreign_ruler_1.owner = { NOT = { THIS = ROOT } }
			}
			scope:foreign_ruler_1 ?= { 
				set_potential_personal_union_royal_consort = yes
				transfer_character = ROOT
				set_variable = prince_consort_character
				set_variable = is_married
				if = { limit = { has_role = character_role_monarch } remove_character_role = character_role_monarch }
				set_character_as_royal = yes
			}
		}
		else_if = {
			limit = { 
				exists = scope:foreign_ruler_1
				scope:foreign_ruler_1 ?= { is_female = yes }
				scope:foreign_ruler_1.owner = { NOT = { THIS = ROOT } }
			}
			scope:foreign_ruler_1 ?= { 
				set_potential_personal_union_royal_consort = yes
				transfer_character = ROOT
				set_variable = prince_consort_character
				set_variable = is_married
				if = { limit = { has_role = character_role_monarch } remove_character_role = character_role_monarch }
				set_character_as_royal = yes
			}
		}
		else_if = {
			limit = { 
				exists = scope:foreign_ruler_1
				scope:foreign_ruler_1 ?= { is_female = no }
				scope:foreign_ruler_1.owner = { NOT = { THIS = ROOT } }
			}
			scope:royal_marriage_character ?= {
				set_variable = prince_consort_character
				set_variable = is_married
				if = { limit = { has_role = character_role_monarch } remove_character_role = character_role_monarch }
				set_character_as_royal = yes
				set_potential_personal_union_royal_consort = yes
				transfer_character = scope:foreign_ruler_1.owner 
			}
		}
		else = {
			if = {
				limit = {
					has_variable = female_princess_lose_title_upon_marriage
					scope:royal_marriage_character ?= {
						is_female = yes
						has_role = character_role_prince
					}
				}
				scope:royal_marriage_character ?= {
					custom_tooltip = {
						text = female_princess_lose_title_upon_marriage.tt
						if = {
							limit = { is_heir_of_own_country = yes }
							disinherit_character = yes
						}
						remove_character_role = character_role_prince
						set_variable = former_royal_character
						if = {
							limit = { NOT = { has_role_of_type = magnate } }
							add_character_role = magnate
						}
					}
				}
				scope:foreign_ruler_1 ?= {
					set_variable = prince_consort_character
					set_variable = is_married
					hidden_effect = { set_is_noble = yes }
				}
			}
			else = {
				scope:foreign_ruler_1 ?= {
					set_variable = prince_consort_character
					set_variable = is_married
					set_character_as_royal = yes
				}
			}
		}
		
	}

	option = { # public wedding for the masses
		name = wedding.1.b
		scope:royal_marriage_character ?= {
			add_modifier = { # popularity
				name = marriage_popularity
				days = very_long_modifier_time
			}
		}
		
		#Transfer character
		if = {
			limit = {
				exists = scope:foreign_ruler_1
				is_subject_of = scope:foreign_ruler_1.owner
				scope:foreign_ruler_1.owner ?= { NOT = { THIS = ROOT } }
			}

wedding.2 = {
	type = country_event
	placement = ROOT
	
	title = wedding.2.t
	desc = wedding.2.d
	flavor = wedding.2.f
	
	gui_window = event_window_1char_tabloid

	left_icon = scope:royal_marriage_character

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	
	duration = 1
	
	trigger = {
		# triggered by marry_ruler_or_heir interaction
	}

	immediate = {
	
		#Mark Royal Bachelor
		random_scope_character = {
			limit = {
				has_variable = spouse_seeker
			}
			save_scope_as = royal_marriage_character
			set_variable = is_married
			remove_variable = spouse_seeker
		}

	}
	
	cancellation_trigger = {
		NOT = { exists = scope:royal_marriage_character }
	}
	
	option = { # marry noble 3
		name = wedding.2.d
		default_option = yes
		
		random_scope_character = {
			limit = {
			
				has_role_of_type = magnate
				is_noble = yes
				
				#Is of a marryable age
				is_adult = yes
				trigger_if = {
					limit = { scope:THIS_character_scope ?= { is_female = no } }
					age < 50
				}
				trigger_else_if = {
					limit = { scope:THIS_character_scope ?= { is_female = yes } }
					age < 40
				}
				
				#Is of correct gender or gender orientation
				trigger_if = {
					limit = { scope:THIS_character_scope ?= { is_female = no } }
					is_female = yes
					NOT = { has_trait = homosexual_character }
				}
				trigger_else_if = {
					limit = { scope:THIS_character_scope ?= { is_female = yes } }
					is_female = no
					NOT = { has_trait = homosexual_character }
				}
				
				#Shares same religion
				religion ?= { is_state_religion = scope:actor }
				
				#Shares similar culture
				culture ?= { shares_heritage_trait_group_with_any_primary_culture = scope:actor }
				
				#Not already a target
				custom_tooltip = {
					text = is_not_married.tt
					NOT = { has_variable = spouse_seeker }
					NOT = { has_variable = is_married }
					NOT = { has_variable = marriage_target }
				}
				
			}
			set_variable = marriage_target
			save_scope_as = foreign_ruler_1
			owner ?= {
				save_scope_as = marriage_country 
			}
		}
		trigger_event = {
			id = wedding.1
			popup = yes
		}
		
	}
	
}

wedding.3 = {
	type = country_event
	placement = ROOT
	
	title = wedding.3.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = wedding.3.d1
				trigger = {
					any_scope_character = {
						has_role = character_role_monarch
						interest_group ?= {
							OR = {
								ig_approval <= angry
								ig_counts_as_marginal = yes
							}
						}
					}
				}
			}
			triggered_desc = {
				desc = wedding.3.d2
				trigger = {
					any_scope_character = {
						has_role = character_role_monarch
						popularity <= -50
					}
				}
			}
			triggered_desc = {
				desc = wedding.3.d3
				trigger = {
					any_scope_character = {
						has_role = character_role_monarch
						OR = {
							has_trait = psychological_affliction
							age >= 75
						}
					}
				}
			}
			triggered_desc = {
				desc = wedding.3.d4
				trigger = {
					
				}
			}
		}
	}
	flavor = wedding.3.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:abdicating_monarch
	right_icon = scope:heir_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/2Characters"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	
	duration = 1

	#dlc = dlc004
	
	trigger = {
		# triggered by abdicate_monarch interaction
	}

	immediate = {
		random_scope_character = {
			limit = { has_role = character_role_monarch }
			save_scope_as = abdicating_monarch
			interest_group ?= {
				save_scope_as = monarch_ig
			}
		}
		ordered_scope_character = {
			limit = {
				has_role = character_role_prince
				NOT = { has_variable = prince_consort_character }
				NAND = {
					owner ?= { has_variable = primogeniture_system }
					is_female = yes
				}
			}
			order_by = age
			position = 0
			check_range_bounds = no
			save_scope_as = heir_scope
		}
		law_type:law_HOS_monarch ?= {
			save_scope_as = monarchy_law
		}
	}

	option = { # reassure the aristocracy
		name = wedding.3.a
		trigger = {
			exists = scope:heir_scope
		}
		default_option = yes
		add_loyalists = {
			value = 0.1
			pop_type = aristocrats
		}
		scope:heir_scope ?= {
			set_character_as_monarch = yes
		}
		hidden_effect = {
			scope:abdicating_monarch ?= {
				remove_character_role = character_role_monarch
				set_character_as_royal = yes
				set_variable = prince_renounced_throne
				set_variable = former_elective_monarch
			}
		}
	}

	option = { # reassure the people
		name = wedding.3.b
		trigger = {
			exists = scope:heir_scope
		}
		add_loyalists = {
			value = 0.01
		}
		scope:heir_scope ?= {
			set_character_as_monarch = yes
		}
		hidden_effect = {
			scope:abdicating_monarch ?= {
				remove_character_role = character_role_monarch
				set_character_as_royal = yes
				set_variable = prince_renounced_throne
				set_variable = former_elective_monarch
			}
		}
	}
	
	option = { # No heir - Regency
		name = wedding.3.c
		trigger = {
			NOT = {
				exists = scope:heir_scope
			}
		}
		scope:abdicating_monarch ?= {
			remove_character_role = character_role_monarch
			set_character_as_royal = yes
			set_variable = prince_renounced_throne
			set_variable = former_elective_monarch
		}
	}
