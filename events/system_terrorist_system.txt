namespace = terrorist_system

#Generate terrorists
terrorist_system.1 = {
    type = country_event

	hidden = yes
	
	trigger = {											# What triggers the event, if using other events or journal entries leave empty
		
		#Faithless Agitator exists to become a terrorist
		any_scope_character = {
			has_role_of_type = agitator
			NOR = {
				has_role_of_type = general
				has_role_of_type = admiral
				has_role = character_role_terrorist
			}
			is_a_special_agitator = no
			character_is_peaceful = no
			OR = {
				culture = { NOT = { is_primary_culture_of = owner } }
				AND = {
					interest_group ?= { is_interest_group_type = ig_devout }
					NOT = { religion = owner.religion }
				}
			}
			loyalty < define:NCharacters|LOYALTY_UNCERTAIN_THRESHOLD
		}
		
    }
	
	immediate = {
		
		#Generate terrorist
		every_scope_character = {
			limit = {
				has_role_of_type = agitator
				NOR = {
					has_role_of_type = general
					has_role_of_type = admiral
					has_role = character_role_terrorist
				}
				is_a_special_agitator = no
				character_is_peaceful = no
				OR = {
					culture = { NOT = { is_primary_culture_of = owner } }
					AND = {
						interest_group ?= { is_interest_group_type = ig_devout }
						NOT = { religion = owner.religion }
					}
				}
				loyalty < define:NCharacters|LOYALTY_UNCERTAIN_THRESHOLD
			}
			add_character_role = character_role_terrorist
		}
		
	}

}

# Assassination
terrorist_system.2 = {
    type = country_event

    title = terrorist_assassination.t 								# Title of the event; ADD LOCALISATION
    desc = terrorist_assassination.d 								# Description of the event; ADD LOCALISATION
    #flavor = terrorist_assassination.f 							# Flavor for event; ADD LOCALISATION, IF REQUIRED

    event_image = { texture = "southamerica_public_figure_assassination" }

    icon = "gfx/interface/icons/invention_icons/anarchy.dds" 		# Icon that shows on the map or the outliner
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear" 	# Sound effect if needed

    duration = 1 # Duration to fire if triggers are met
	
	trigger = {														# What triggers the event, if using other events or journal entries leave empty
		
		#Someone to assassinate
		any_scope_character = {
			OR = { 
				character_is_key_state_officeholder = yes
				AND = { 
					has_role_of_type = politician
					interest_group ?= { is_in_government = yes }
				}
			}
			NAND = {
				is_historical = yes
				owner ?= { historical_railroading_active = yes }
			}
		}
		
    }
	
	immediate = {
	
		#Mark target
		random_scope_character = {
			limit = {
				OR = { 
					character_is_key_state_officeholder = yes
					AND = { 
						has_role_of_type = politician
						interest_group ?= { is_in_government = yes }
					}
				}
				NAND = {
					is_historical = yes
					owner ?= { historical_railroading_active = yes }
				}
			}
			save_scope_as = assassination_target
			interest_group ?= { save_scope_as = assassination_target_IG }
		}
		
		#Mark assassin
		random_scope_character = {
			limit = {
				has_role = character_role_terrorist
			}
			save_scope_as = terror_attack_leader
		}
		
	}
	
	cancellation_trigger = {
		OR = {
			NOT = { exists = scope:assassination_target }
			NOT = { exists = scope:terror_attack_leader }
		}
	}
    
    option = {
        name = terrorist_assassination.option					# RIP
		
		#Kill character
		scope:assassination_target ?= {
			kill_character = yes
		}
		
		#Domestic reaction
		add_radicals = {
			value = very_large_radicals
			interest_group = scope:assassination_target_IG
		}
		
    }

}

#Generate guerrilla
terrorist_system.3 = {
    type = country_event

	hidden = yes
	
	trigger = {											# What triggers the event, if using other events or journal entries leave empty
		
		#Generate guerrilla
		any_scope_character = {
			has_role = character_role_terrorist
			NOT = { has_variable = guerrilla_character }
			interest_group.ig_clout >= 0.10
		}
		
    }
	
	immediate = {
		
		#Generate guerrilla
		every_scope_character = {
			limit = {
				has_role = character_role_terrorist
				NOT = { has_variable = guerrilla_character }
				interest_group.ig_clout >= 0.10
			}
			set_variable = guerrilla_character
		}
		
	}

}

# Break out Prisoners
terrorist_system.4 = {
    type = country_event

    title = terrorist_prison_breakout.t 							# Title of the event; ADD LOCALISATION
    desc = terrorist_prison_breakout.d 								# Description of the event; ADD LOCALISATION
    flavor = terrorist_prison_breakout.f 							# Flavor for event; ADD LOCALISATION, IF REQUIRED

    event_image = { texture = "europenorthamerica_political_extremism" }

    icon = "gfx/interface/icons/invention_icons/anarchy.dds" 		# Icon that shows on the map or the outliner
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear" 	# Sound effect if needed

    duration = 1 # Duration to fire if triggers are met
	
	trigger = {														# What triggers the event, if using other events or journal entries leave empty
		
		#Has a prisoner to break out
		any_scope_character = {
			has_role = character_role_terrorist
			has_variable = guerrilla_character
			interest_group ?= { save_temporary_scope_as = terrorist_IG }
			owner ?= {
				any_scope_character = {
					has_role = character_role_prisoner
					interest_group ?= { is_same_interest_group_type = scope:terrorist_IG }
				}
			}
		}
		
    }
	
	immediate = {
	
		#Mark Guerilla
		random_scope_character = {
			limit = {
				has_role = character_role_terrorist
				has_variable = guerrilla_character
			}
			save_scope_as = terrorist_scope
			interest_group ?= { save_scope_as = terrorist_IG_scope }
		}
		
		#Mark Prisoner
		random_scope_character = {
			limit = {
				has_role = character_role_prisoner
				interest_group ?= { is_same_interest_group_type = scope:terrorist_IG_scope }
			}
			save_scope_as = prisoner_scope
		}
		
	}
	
	cancellation_trigger = {
		OR = {
			NOT = { exists = scope:terrorist_scope }
			NOT = { exists = scope:prisoner_scope }
		}
	}
    
    option = {
        name = terrorist_prison_breakout.option					# Oh no
		
		#Free Prisoner
		scope:prisoner_scope ?= {
			replace_character_roles = { add = { character_role_agitator character_role_terrorist } remove_all = yes }
			if = {
				limit = { has_modifier = convicted_character }
				remove_modifier = convicted_character
			}
			set_variable = guerrilla_character
		}
		
		#Domestic reaction
		add_radicals = {
			value = small_radicals
			interest_group = scope:terrorist_IG_scope
		}
		
    }

}
