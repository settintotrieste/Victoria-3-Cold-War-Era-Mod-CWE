# The New Monarch
constitutional_monarchy_system.1 = {
    type = country_event

    title = new_monarch_crowned.t 										# Title of the event; ADD LOCALISATION
    desc = new_monarch_crowned.d 										# Description of the event; ADD LOCALISATION
    flavor = new_monarch_crowned.f 										# Flavor for event; ADD LOCALISATION, IF REQUIRED

    gui_window = event_window_1char_tabloid

	left_icon = scope:monarch_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/2Characters"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

    duration = 1 # Duration to fire if triggers are met
	
	trigger = {															# What triggers the event, if using other events or journal entries leave empty
		
		any_scope_character = {
			has_role = character_role_monarch
			is_eligible_to_become_monarch = yes
		}
		
    }
	
	immediate = {
		
		#Mark monarch
		random_scope_character = {
			limit = { has_role = character_role_monarch }
			save_scope_as = monarch_scope
		}
		
	}
	
	cancellation_trigger = {
		has_variable = coronation_day_event_ongoing
	}
    
    option = { # reassure the aristocracy
		name = wedding.3.a
		default_option = yes
		add_loyalists = {
			value = 0.1
			pop_type = aristocrats
		}
	}

	option = { # reassure the people
		name = wedding.3.b
		add_loyalists = {
			value = 0.01
		}
	}

}
